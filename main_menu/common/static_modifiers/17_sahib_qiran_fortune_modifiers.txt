# The Ṣāḥib-qirān's fortune, by band
# (in_game/common/scripted_effects/18_sahib_qiran_fortune_effects.txt).
# Kept by sq_fortune_refresh: one of the two, or neither.

# High: the amirs follow a man whose star is up.
sq_fortune_high_mod = {
	game_data = {
		category = country
		decaying = no
	}
	# 20, not 10: the sworn amirs (subject_types/sworn_amir.txt) carry no
	# loyalty of their own; this is what holds them.
	subject_loyalty = 20
	monthly_horde_unity = 0.10
	land_morale_modifier = 0.05
	# THE CONQUEST MACHINE. Base gives auto-conquer to the man for life
	# (character modifier timmy_wants_to_play). Here it belongs to the
	# fortune: while his star is up, what his armies occupy is his; when
	# it is not, he besieges and treats like anyone else.
	auto_conquer_at_war = yes
	# At the height of his fortune rulers send embassies and gifts, not
	# armies against him (see sq_name_goes_before_him_mod below).
	antagonism_received_modifier = -0.30
	# He lives on what he takes: without this the host bankrupted him
	# (author's test). Base's own plunder bonuses are 0.33 and 0.5.
	amount_looted_modifier = 1.0
	# Newly taken country does not rise behind a conqueror whose star is
	# up; it rises when the star falls. (Base scales: -0.05 / -0.1.)
	pop_join_rebel_threshold = -0.15
	global_separatism = -0.10
}

# Failing: he has stopped winning, or stopped giving.
sq_fortune_low_mod = {
	game_data = {
		category = country
		decaying = no
	}
	subject_loyalty = -15
	monthly_horde_unity = -0.15
}

# The Ṣāḥib-qirān's standing among other courts, whatever his fortune.
# Temür took Delhi, Damascus, Baghdad and the Ottoman sultan himself and
# no league ever formed against him: Cairo, Constantinople, Castile and
# the Ming court each dealt with him singly, by embassy, submission or
# avoidance. In the author's test (2026-11-14) the antagonism of his
# conquests shut him out of diplomacy "for centuries". Given by
# sq_fortune_refresh (scripted_effects/18) to whoever carries sq_fortune.
sq_name_goes_before_him_mod = {
	game_data = {
		category = country
		decaying = no
	}
	# -0.40 still raised all of Hindustan in one league (author's test):
	# the rulers of the 1390s dealt with him one by one.
	antagonism_received_modifier = -0.60
	# A following, not an administration: he rules through men who must
	# be given things, like the keeper of a khan in a single realm.
	monthly_political_influence_gain_modifier = -0.20
}

# The keeper of the khan INSIDE the ulus (scripted_effects/15,
# ulus_regency_stage_tick). The beglerbeg reform's -20% is the price a
# beg pays for ruling a unitary khanate through a puppet. An amir of
# the ulus who takes the khan into keeping has GAINED something: every
# other house must now deal with him. That gain is greatest while the
# peace holds and is spent as the amirs fall out.
# This one undoes the reform's -20% for him:

ulus_khan_keeper_influence = {

}
