# Spec: docs/design/eranzamin_plan.md §19.8 (Suppress / Concede / Co-opt
# baseline mods) and §20.5 (final victory-state mods).
#
# Pattern: the Matter resolution event applies one of these modifiers
# permanently. The END condition in §20 then reads the axis state and
# matter-resolved flag to determine which IO-transform victory variant
# fires.

# ---§19.8 CONCEDE resolution modifier -----------------------------------

dual_realm_of_eranzamin = {
	# Concede exit (either Matter). Polity persists with the breakaway
	# as a permanent vassal or independent neighbor. Estate balance
	# forcibly recentered; axis pinned within ±15 for 50y via the
	# Concede resolution event applying a periodic axis-correction.
	game_data = {
		category = country
		decaying = no
	}

	# Tax + manpower split modestly between sides — no big swing.
	tax_income_efficiency = 0.05
	global_manpower_modifier = 0.05
	# Diplomatic prestige hit for failing to fully reintegrate.
	diplomatic_reputation = -1
	# Stability premium — dual sovereignty is functional.
	stability_investment = 0.10
}

# ---§19.8 CO-OPT resolution modifier ------------------------------------

convergent_eranzamin = {
	# Co-opt exit. Mid-war "Marriage at" / "Investiture at" event,
	# requires co_opt_track flag set at 25 or 50 threshold AND war
	# duration >= 36 months. Annexes the breakaway peacefully.
	game_data = {
		category = country
		decaying = no
	}

	# Modest bonuses both sides, no penalties anywhere.
	tax_income_efficiency = 0.05
	global_manpower_modifier = 0.05
	stability_investment = 0.15
	monthly_legitimacy = 0.15
	# Cultural-tradition bonus — the polity has demonstrated synthesis
	# capacity, which is rare and remembered.
	cultural_tradition_modifier = 0.25
	# Axis is pinned to 0 ± 5 via the resolution event applying a
	# stronger periodic axis-correction.
}
