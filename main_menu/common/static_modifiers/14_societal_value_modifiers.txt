# Cabinet "Change Societal Values" requires a <side>_progress_cabinet_efficiency
# static modifier to exist for each side of a societal value, or it refuses to
# push that side. But the block is not a bare existence marker: it IS the push.
# The cabinet applies this modifier scaled by cabinet efficiency, and the
# monthly_towards_<side> line inside is the coefficient the engine reads -
# every base block (societal_values.txt) carries monthly_towards_<side> = 1.
# Our old blocks held country_cabinet_efficiency = 0 instead: the action
# seated, worked, and pushed exactly nothing (playtest 2026-08-27).
turk_progress_cabinet_efficiency = {
	game_data = { category = country }
	monthly_towards_turk = 1
}

tazik_progress_cabinet_efficiency = {
	game_data = { category = country }
	monthly_towards_tazik = 1
}
