# ---25 THRESHOLD: A petition from the umarā ---------------------------

matter_of_tork.25 = {
	type = country_event
	category = disaster_event
	title = matter_of_tork.25.title
	desc = matter_of_tork.25.desc
	outcome = neutral

	# Receive them with honor — concedes
	option = {
		name = matter_of_tork.25.a
		add_estate_satisfaction = { type = estate_type:amirs_estate value = 10 }
		add_estate_satisfaction = { type = estate_type:diwan_estate value = -5 }

		set_variable = co_opt_track

	option = {
		name = matter_of_tork.25.b
		change_variable = { name = matter_progress add = -5 }

			limit = { is_tribal_pastoral_location = yes }
			add_unrest = 3
		}
	}

	# Refuse to seat them — provokes
	option = {
		name = matter_of_tork.25.c
		change_variable = { name = matter_progress add = 5 }
		set_variable = petition_refused
	}
}

# ---75 THRESHOLD: The tribes are assembling --------------------------

matter_of_tork.75 = {
	type = country_event
	category = disaster_event
	title = matter_of_tork.75.title
	desc = matter_of_tork.75.desc
	outcome = neutral

	# March the standing army — accelerate to 100, fire disaster
	option = {
		name = matter_of_tork.75.a
		set_variable = { name = matter_progress value = 100 }
		set_variable = matter_chose_suppression
	}

	# Convene a Great Diwan — mass concession, rollback to 30
	option = {
		name = matter_of_tork.75.b
		set_variable = { name = matter_progress value = 30 }
		add_stability = -20
		add_legitimacy = -25
	}

	# Yarliq of Amnesty — full co-opt, disaster averted
	# Available only if co_opt_track is set
	option = {
		name = matter_of_tork.75.c
		trigger = { has_variable = co_opt_track }
		set_variable = { name = matter_progress value = 0 }
		add_country_modifier = {
			modifier = pastures_of_the_sword
			years = -1
			mode = add_and_extend
		}
		set_variable = matter_of_tork_resolved
		# Disaster will end via can_end on next tick.
	}
}

# ---100: DISASTER FIRES (civil war structural event) -----------------

matter_of_tork.100 = {
	type = country_event
	category = disaster_event
	title = matter_of_tork.100.title
	desc = matter_of_tork.100.desc
	outcome = neutral

	immediate = {
		spawn_turkmen_breakaway = yes
		# Wide-area amir defection roll + unrest cascade on non-rollers
		every_province = {
			limit = { is_tribal_pastoral_location = yes }
			# Non-rolling provinces (those that didn't transfer pops in
			# spawn_turkmen_breakaway) get unrest debuff.
			add_unrest = 5
			every_location_in_province = {
				change_control = control_severe_penalty
			}
		}
		# Player buffs during war
		add_estate_satisfaction = { type = estate_type:diwan_estate value = 20 }
	}

	option = {
		name = matter_of_tork.100.a
		# Acknowledge — the war is on.
	}
}

# ---RESOLUTION EVENTS (war ended; route by player flags) ------------

# Resolution events are triggered by an on_action listener on
# `on_civil_war_ended` (or wired via on_monthly polling). For first
# implementation, the eranzamin situation's on_monthly polls
# `in_civil_war = no` + has_variable = matter_of_tork_fired to
# route into one of these.

# Suppress — player won the war
matter_of_tork.500 = {
	type = country_event
	category = disaster_event
	title = matter_of_tork.500.title
	desc = matter_of_tork.500.desc
	outcome = good

	trigger = {
		has_variable = matter_of_tork_fired
		in_civil_war = no
		# Won — breakaway tag was annexed
		NOT = { exists = var:matter_breakaway_ref }
	}

	immediate = {
		matter_of_tork_apply_suppress = yes
	}

	option = {
		name = matter_of_tork.500.a
	}
}

# Co-opt — mid-war "Marriage at" event. Available only if
# co_opt_track was set AND war has lasted ≥36 months.
matter_of_tork.502 = {
	type = country_event
	category = disaster_event
	title = matter_of_tork.502.title
	desc = matter_of_tork.502.desc
	outcome = good

	trigger = {
		has_variable = matter_of_tork_fired
		has_variable = co_opt_track
		in_civil_war = yes
		exists = var:matter_breakaway_ref
		# ≥ 36mo since war started
		var:matter_war_months >= 36
	}

	# Player choice: accept the marriage or refuse
	option = {
		name = matter_of_tork.502.a
		matter_of_tork_apply_co_opt = yes
		# Peacefully annex the breakaway
		var:matter_breakaway_ref ?= {
			destroy_rebel = this
		}
	}

	option = {
		name = matter_of_tork.502.b
		# Refused — war continues. -10 progress as a small consolation.
		change_variable = { name = matter_progress add = -10 }
	}
}
