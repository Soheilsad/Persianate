namespace = amirs_breaking

# .1 — THE PRIZE. To the outsider who has come to hold the pastures of
# the broken ulus (on_action/36_chaghatay_breaking_hooks.txt). Once in a
# game. The Chaghatay were an army with households; the army is gone and
# the households are his to dispose of.
amirs_breaking.1 = {
	type = country_event
	title = amirs_breaking.1.title
	desc = amirs_breaking.1.desc
	outcome = positive
	major = yes
	major_trigger = {
		capital ?= {
			OR = {
				sub_continent = sub_continent:central_asia
				sub_continent = sub_continent:middle_east
				sub_continent = sub_continent:south_asia
			}
		}
	}

	illustration_tags = {
		10 = exterior
		10 = happy
	}

	# Distribute them: the practice of the steppe. The tümens' people in
	# our pastures become ours at once.
	option = {
		name = amirs_breaking.1.a
		custom_tooltip = amirs_breaking_distribute_tt
		hidden_effect = {
			ulus_distribute_the_tumens = yes
		}
		add_country_modifier = {
			modifier = ulus_tumens_distributed_mod
			years = 30
		}
		add_prestige = ulus_breaking_prestige
		# A steppe court does what the steppe does with a beaten tribe;
		# anyone short of men wants the riders as his own now.
		ai_chance = {
			factor = 1
			modifier = {
				factor = 3
				government_type = government_type:steppe_horde
			}
			modifier = {
				factor = 1.5
				manpower_percentage < 0.5
			}
			modifier = {
				factor = 1.5
				OR = {
					ai_personality = ai_personality:ai_aggressive
					ai_personality = ai_personality:ai_expansionist
				}
			}
		}
	}
	# Take them into service: they keep their name and their captains,
	# and ride for us.
	option = {
		name = amirs_breaking.1.b
		custom_tooltip = amirs_breaking_service_tt
		add_country_modifier = {
			modifier = ulus_tumens_in_service_mod
			years = -1
		}
		add_prestige = ulus_breaking_service_prestige
		# A settled court has no tribes of its own to divide them among,
		# and gains a culture it can rule through; a careful ruler keeps
		# what works.
		ai_chance = {
			factor = 1
			modifier = {
				factor = 2.5
				NOT = { government_type = government_type:steppe_horde }
			}
			modifier = {
				factor = 1.5
				OR = {
					ai_personality = ai_personality:ai_cautious
					ai_personality = ai_personality:ai_defensive
					ai_personality = ai_personality:ai_friendly
				}
			}
		}
	}
}
