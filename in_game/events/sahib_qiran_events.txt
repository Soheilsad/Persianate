# ============================================================================
# THE ṢĀḤIB-QIRĀN'S FORTUNE — events
# (scripted_effects/18_sahib_qiran_fortune_effects.txt; fired from
# on_action/33_sahib_qiran_hooks.txt and the monthly tick).
#
#   .1   The Sons and the Amirs — a new man on the seat; the fortune
#        was his father's.
#   .2   The Amirs Ride Away — the fortune empty sq_empty_years
#        running.
#   .3   The Conjunction Holds; .4 An Ambition Made Good.
#   .5   The Army Is Lost — a lost war with the fortune already low;
#        .6 the victor's notice.
#   .7   An Embassy from the Ṣāḥib-qirān — to a player he is courting.
# ============================================================================

# ----------------------------------------------------------------------------
# .1 — THE SONS AND THE AMIRS. Temür died at Otrar in February 1405
# having named Pir Muhammad. Khalil Sultan took Samarkand and the
# treasury within weeks; Shah Rukh held Herat; it was 1409 before one
# of them ruled the whole. The effect (the fortune reset to
# sq_fortune_heir) has already run in the hook.
# ----------------------------------------------------------------------------
sahib_qiran.1 = {
	type = country_event
	title = sahib_qiran.1.title
	desc = sahib_qiran.1.desc
	outcome = negative

	illustration_tags = {
		10 = interior
		10 = angry
	}

	trigger = {
		has_variable = sq_fortune
	}

	option = {
		name = sahib_qiran.1.a
		custom_tooltip = sahib_qiran_succession_tt
	}
}

# ----------------------------------------------------------------------------
# .7 — AN EMBASSY FROM THE ṢĀḤIB-QIRĀN. To a PLAYER realm an AI
# Ṣāḥib-qirān is courting (sq_send_embassy). scope:sq_courting_realm is
# his realm; the gift he offers is on it as sq_gift_offered. Nothing is
# done to the player unless he accepts.
# ----------------------------------------------------------------------------
sahib_qiran.7 = {
	type = country_event
	title = sahib_qiran.7.title
	desc = sahib_qiran.7.desc
	outcome = positive

	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		exists = scope:sq_courting_realm
		scope:sq_courting_realm = { has_variable = sq_gift_offered }
	}

	option = {
		name = sahib_qiran.7.a
		custom_tooltip = sahib_qiran_embassy_accept_tt
		add_gold = scope:sq_courting_realm.var:sq_gift_offered
		scope:sq_courting_realm = {
			add_gold = {
				value = var:sq_gift_offered
				multiply = -1
			}
			remove_variable = sq_gift_offered
		}
		add_opinion_mutual_effect = {
			modifier = opinion_sq_embassy_gifts
			target = scope:sq_courting_realm
		}
	}
	option = {
		name = sahib_qiran.7.b
		scope:sq_courting_realm = {
			remove_variable = sq_gift_offered
		}
	}
}


# ----------------------------------------------------------------------------
# sahib_qiran.8 — THE CORE. Hidden; five days after the founding. The
# sworn amirs that border him and have their seat in the same AREA as
# his own are taken into his realm outright, with their men, treasury
# and people (transfer = yes). Temür's Barlas and the houses around
# Kesh and Samarkand were his household, not his following. The
# periphery stays sworn.
# ----------------------------------------------------------------------------
sahib_qiran.8 = {
	type = country_event
	hidden = yes
	outcome = neutral
	title = empty_text
	desc = empty_text

	trigger = {
		has_variable = sq_fortune
		exists = capital
	}

	immediate = {
		every_subject = {
			limit = {
				is_subject_type = sworn_amir
				has_variable = sq_sworn_amir
				capital ?= { area = root.capital.area }
				any_neighbor_country = { this = root }
			}
			add_to_list = sq_core_houses
		}
		every_in_list = {
			list = sq_core_houses
			save_scope_as = sq_core_house
			root = {
				annex_country = {
					country = scope:sq_core_house
					reason = Diplomatic
					transfer = yes
				}
			}
		}
	}
}
