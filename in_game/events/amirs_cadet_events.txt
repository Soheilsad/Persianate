namespace = amirs_cadet

# ============================================================================
# THE CADET TAKES THE TÜMEN — events. History and effects:
# common/scripted_effects/17_amirs_cadet_effects.txt. Hooks:
# common/on_action/32_amirs_cadet_hooks.txt.
#
#   .1   the invader is upon the house, OR the house's own bid for the
#        throne has failed: the chief stands or rides out
#        (to the house; scope:ulus_cadet_invader; the text follows
#        ulus_cadet_bid_failed and whether the victor is khan or regent)
#   .2   a kinsman of the house has come to our camp   (to the invader)
#   .3   a tümen has a new chief by another's yarlıq   (to the leader)
#   .10  the old chief comes back for his tümen        (to the house)
#   .11  the old chief is dead on the road             (to the house)
#   .20  stage 2: a kinsman who is the better soldier  (to the house)
# ============================================================================

# ----------------------------------------------------------------------------
# .1 — root = the amir house. scope:ulus_cadet_invader = the realm that
# beat it or took its camp.
# ----------------------------------------------------------------------------
amirs_cadet.1 = {
	type = country_event
	title = amirs_cadet.1.title
	desc = {
		first_valid = {
			# Our own bid failed, against the state that holds the khan.
			triggered_desc = {
				trigger = {
					has_variable = ulus_cadet_bid_failed
					scope:ulus_cadet_invader ?= { is_ulus_regent = yes }
				}
				desc = amirs_cadet.1.desc_bid_regent
			}
			# Our own bid failed, against the khan himself.
			triggered_desc = {
				trigger = { has_variable = ulus_cadet_bid_failed }
				desc = amirs_cadet.1.desc_bid_khan
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = amirs_cadet.1.desc
			}
		}
	}
	outcome = negative

	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		exists = scope:ulus_cadet_invader
		is_ulus_chaghatay_amir = yes
		NOT = { has_variable = ulus_fled_chief }
		ulus_house_has_cadet = yes
	}

	immediate = {
		ulus_cadet_find = yes
	}

	# ----- A: The chief rides out; the kinsman goes to the camp -----
	option = {
		name = amirs_cadet.1.a
		trigger = { exists = scope:ulus_cadet }
		# THE CHIEF RIDES OUT when the house cannot win and the tümen has
		# a better man to keep: outmatched, a kinsman who is the abler
		# soldier, a tümen too brittle for a stand, an old chief, a
		# careful one, and above all a chief the victor has a reason to
		# kill (his own bid for the throne has just failed).
		ai_chance = {
			factor = 1
			modifier = {
				factor = 3
				military_strength < {
					value = scope:ulus_cadet_invader.military_strength
					multiply = ulus_cadet_outmatched_share
				}
			}
			modifier = {
				factor = 0.4
				military_strength >= {
					value = scope:ulus_cadet_invader.military_strength
					multiply = ulus_cadet_ai_match_share
				}
			}
			modifier = {
				factor = 1.5
				ulus_cadet_outsoldiers_chief = { EDGE = ulus_cadet_ai_mil_edge }
			}
			modifier = {
				factor = 1.5
				ulus_cadet_tumen_is_brittle = yes
			}
			modifier = {
				factor = 1.5
				ruler ?= { age_in_years >= 55 }
			}
			modifier = {
				factor = 2
				has_variable = ulus_cadet_bid_failed
			}
			modifier = {
				factor = 1.5
				OR = {
					ai_personality = ai_personality:ai_cautious
					ai_personality = ai_personality:ai_defensive
				}
			}
			modifier = {
				factor = 0.6
				ai_personality = ai_personality:ai_aggressive
			}
		}

		custom_tooltip = amirs_cadet_flee_tt
		if = {
			limit = {
				scope:ulus_cadet_invader = { NOT = { is_ulus_chaghatay_member = yes } }
			}
			custom_tooltip = amirs_cadet_tributary_tt
		}
		else_if = {
			limit = {
				scope:ulus_cadet_invader = { is_ulus_regent = yes }
			}
			custom_tooltip = amirs_cadet_sworn_tt
		}
		else = {
			custom_tooltip = amirs_cadet_sworn_khan_tt
		}
		hidden_effect = {
			ulus_cadet_takes_tumen = yes
			ulus_cadet_submits = yes
			scope:ulus_cadet_invader = {
				trigger_event_non_silently = amirs_cadet.2
			}
			international_organization:ulus_chaghatay ?= {
				leader_country ?= {
					if = {
						limit = { NOT = { this = scope:ulus_cadet_invader } }
						trigger_event_non_silently = amirs_cadet.3
					}
				}
			}
		}
	}

	# ----- B: The chief stands -----
	option = {
		name = amirs_cadet.1.b
		# THE CHIEF STANDS when he is a soldier himself, when the house is
		# a match for the invader, and when no kinsman is plainly the
		# better man.
		ai_chance = {
			factor = 1
			modifier = {
				factor = 2
				military_strength >= {
					value = scope:ulus_cadet_invader.military_strength
					multiply = ulus_cadet_ai_match_share
				}
			}
			modifier = {
				factor = 1.5
				ruler ?= { mil >= ulus_cadet_ai_able_chief }
			}
			modifier = {
				factor = 0.5
				ulus_cadet_outsoldiers_chief = { EDGE = ulus_cadet_ai_mil_edge_great }
			}
			modifier = {
				factor = 1.5
				OR = {
					ai_personality = ai_personality:ai_aggressive
					ai_personality = ai_personality:ai_expansionist
				}
			}
		}

		add_prestige = ulus_cadet_stand_prestige
		add_horde_unity = ulus_cadet_stand_unity
	}
}

# ----------------------------------------------------------------------------
# .10 — the old chief comes back. root = the house. Fires one to three
# years after he rode out, if he and the kinsman both still live and
# the kinsman still leads the tümen.
# ----------------------------------------------------------------------------
amirs_cadet.10 = {
	type = country_event
	title = amirs_cadet.10.title
	desc = amirs_cadet.10.desc
	outcome = neutral

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		has_variable = ulus_fled_chief
		has_variable = ulus_cadet_chief
		var:ulus_fled_chief ?= { is_alive = yes }
		var:ulus_cadet_chief ?= {
			is_alive = yes
			is_ruler = yes
		}
	}

	immediate = {
		var:ulus_fled_chief = { save_scope_as = ulus_fled_chief }
		var:ulus_cadet_chief = { save_scope_as = ulus_cadet }
	}

	# ----- A: The tümen is his; I held it for him -----
	option = {
		name = amirs_cadet.10.a
		# HE YIELDS when the old chief is the abler man, when the tümen is
		# too brittle for two chiefs, and when he is not the kind to
		# gamble (Temür yielded Kish to Hajji Beg in 1361, and lost
		# nothing by it).
		ai_chance = {
			factor = 1
			modifier = {
				factor = 2
				scope:ulus_fled_chief = {
					mil >= {
						value = scope:ulus_cadet.mil
						add = ulus_cadet_ai_mil_edge
					}
				}
			}
			modifier = {
				factor = 1.5
				ulus_cadet_tumen_is_brittle = yes
			}
			modifier = {
				factor = 1.5
				scope:ulus_cadet = { age_in_years < 25 }
			}
			modifier = {
				factor = 1.5
				OR = {
					ai_personality = ai_personality:ai_cautious
					ai_personality = ai_personality:ai_friendly
				}
			}
		}

		custom_tooltip = amirs_cadet_yield_tt
		add_horde_unity = ulus_cadet_yield_unity
		hidden_effect = {
			set_variable = { name = ulus_succession_settled value = 1 months = 6 }
			set_new_ruler = scope:ulus_fled_chief
			# The name he made is his own and stays with him.
			scope:ulus_cadet = {
				if = {
					limit = { has_variable = ulus_renown }
					change_variable = { name = ulus_renown add = ulus_cadet_yield_renown }
				}
				else = {
					set_variable = { name = ulus_renown value = ulus_cadet_yield_renown }
				}
			}
			ulus_cadet_clear_exile = yes
		}
	}

	# ----- B: He left it on the ground; I picked it up -----
	option = {
		name = amirs_cadet.10.b
		# HE KEEPS IT when he is the abler man, has made a name since, the
		# regency has reached the years when no one waits for old men,
		# and he is the kind who takes what is in his hand.
		ai_chance = {
			factor = 1
			modifier = {
				factor = 2
				scope:ulus_cadet = {
					mil >= {
						value = scope:ulus_fled_chief.mil
						add = ulus_cadet_ai_mil_edge
					}
				}
			}
			modifier = {
				factor = 1.5
				scope:ulus_cadet = {
					has_variable = ulus_renown
					var:ulus_renown >= 10
				}
			}
			modifier = {
				factor = 1.5
				ulus_regency_stage_at_least = { STAGE = 3 }
			}
			modifier = {
				factor = 1.5
				OR = {
					ai_personality = ai_personality:ai_aggressive
					ai_personality = ai_personality:ai_expansionist
					ai_personality = ai_personality:ai_opportunistic
				}
			}
			modifier = {
				factor = 0.5
				ulus_cadet_tumen_is_brittle = yes
			}
		}

		add_horde_unity = ulus_cadet_refuse_unity
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = ulus_cadet_refuse_nobles
		}
		hidden_effect = {
			ulus_cadet_clear_exile = yes
		}
	}
}

# ----------------------------------------------------------------------------
# .11 — the old chief dies on the road. root = the house.
# ----------------------------------------------------------------------------
amirs_cadet.11 = {
	type = country_event
	title = amirs_cadet.11.title
	desc = amirs_cadet.11.desc
	outcome = neutral

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		has_variable = ulus_fled_chief
		var:ulus_fled_chief ?= { is_alive = yes }
	}

	immediate = {
		var:ulus_fled_chief = { save_scope_as = ulus_fled_chief }
		hidden_effect = {
			kill_character = {
				target = scope:ulus_fled_chief
				reason = assassination
			}
		}
	}

	option = {
		name = amirs_cadet.11.a
		add_horde_unity = ulus_cadet_chief_dead_unity
		hidden_effect = {
			remove_variable = ulus_fled_chief
			remove_variable = ulus_cadet_chief
		}
	}
}

# ----------------------------------------------------------------------------
# .20 — stage 2, the amirs falling out: a kinsman who is plainly the
# better soldier, and men in the tümen who say so. root = the house.
# Fired by ulus_cadet_coup_pulse (rare) or, once a coup has succeeded
# somewhere in the ulus, by ulus_cadet_coup_pulse_contagion (four times
# likelier). A coup that succeeds starts or renews the contagion and,
# if it is the first of the regency, makes the man's name
# (ulus_cadet_coup_succeeded).
# ----------------------------------------------------------------------------
amirs_cadet.20 = {
	type = country_event
	title = amirs_cadet.20.title
	desc = amirs_cadet.20.desc
	outcome = negative

	illustration_tags = {
		10 = interior
		10 = angry
	}

	trigger = {
		is_ulus_chaghatay_amir = yes
		ulus_house_has_ambitious_cadet = yes
	}

	immediate = {
		ulus_cadet_find = yes
	}

	# ----- A: The tümen follows the kinsman -----
	option = {
		name = amirs_cadet.20.a
		trigger = { exists = scope:ulus_cadet }
		# THE TÜMEN FOLLOWS THE KINSMAN when he is far the better soldier,
		# the chief is old, other houses have already done it, and the
		# chief is no soldier to stop him.
		ai_chance = {
			factor = 1
			modifier = {
				factor = 2
				ulus_cadet_outsoldiers_chief = { EDGE = ulus_cadet_ai_mil_edge_great }
			}
			modifier = {
				factor = 1.5
				ruler ?= { age_in_years >= 55 }
			}
			modifier = {
				factor = 1.5
				international_organization:ulus_chaghatay ?= { has_variable = ulus_coup_contagion }
			}
			modifier = {
				factor = 1.5
				ruler ?= { mil < 30 }
			}
			modifier = {
				factor = 0.5
				ulus_cadet_tumen_is_brittle = yes
			}
		}

		custom_tooltip = amirs_cadet_coup_tt
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= {
					NOT = { has_variable = ulus_first_coup_done }
				}
			}
			custom_tooltip = amirs_cadet_first_coup_tt
		}
		custom_tooltip = amirs_cadet_contagion_tt
		add_horde_unity = ulus_cadet_coup_unity
		hidden_effect = {
			ulus_cadet_takes_tumen = yes
			ulus_cadet_coup_succeeded = yes
		}
	}

	# ----- B: The chief has him seized -----
	option = {
		name = amirs_cadet.20.b
		trigger = { exists = scope:ulus_cadet }
		# THE CHIEF HAS HIM SEIZED when he is a soldier himself and still
		# in his strength, when the tümen is too brittle to survive a
		# change of chiefs, and when he is the kind who strikes first.
		ai_chance = {
			factor = 1
			modifier = {
				factor = 2
				ruler ?= { mil >= ulus_cadet_ai_able_chief }
			}
			modifier = {
				factor = 1.5
				ruler ?= { age_in_years < 40 }
			}
			modifier = {
				factor = 1.5
				ulus_cadet_tumen_is_brittle = yes
			}
			modifier = {
				factor = 1.5
				OR = {
					ai_personality = ai_personality:ai_aggressive
					ai_personality = ai_personality:ai_defensive
				}
			}
			modifier = {
				factor = 0.5
				ulus_cadet_outsoldiers_chief = { EDGE = ulus_cadet_ai_mil_edge_great }
			}
		}

		custom_tooltip = amirs_cadet_seized_tt
		add_horde_unity = ulus_cadet_seized_unity
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = ulus_cadet_seized_nobles
		}
		set_variable = { name = ulus_cadet_recent value = 1 years = 10 }
		hidden_effect = {
			kill_character = {
				target = scope:ulus_cadet
				reason = execution
			}
		}
	}

	# Fallback: the man died between the pulse and the event.
	option = {
		name = amirs_cadet.20.c
		trigger = { NOT = { exists = scope:ulus_cadet } }
	}
}
