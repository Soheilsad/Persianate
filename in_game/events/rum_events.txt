# =====================================================================
# rum.2 — THE FORK (mirrors ilkhanate_flavor.1)
#
# Fires once on every eligible beylik at situation start, five days
# after the government switch. The constitutional choice of Shattered
# Rum, up front:
#   A. Ghazi Sultanate — the western-march default. The Burden of
#      the Sword disaster fires on adoption.
#   B. Claimant of the Seljuk Inheritance — Konya's owner only
#      (Karaman, unless the map has already moved). The Seljuk
#      Inheritance disaster fires on adoption.
#   C. Remain a march — no claim, no audit, no crisis (the
#      anatolian_beylik exemption). Both reforms stay pickable
#      later through the government screen; deferral is a choice,
#      not a dead end.
# =====================================================================
rum.2 = {
	type = country_event
	outcome = neutral
	title = rum.2.title
	desc = rum.2.desc

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		has_reform = government_reform:anatolian_beylik
		NOT = { has_reform = government_reform:ghazi_sultanate }
		NOT = { has_reform = government_reform:seljuk_claimant }
	}

	# ----- A: the sword of the faith -----
	option = {
		name = rum.2.a
		ai_chance = {
			factor = 40
			# The doctrine is a frontier doctrine: a beg with infidel
			# neighbors can feed it; an inland beg cannot.
			modifier = {
				factor = 2
				any_neighbor_country = {
					NOT = { religion.group = religion_group:muslim }
				}
			}
			modifier = {
				factor = 0.25
				NOT = {
					any_neighbor_country = {
						NOT = { religion.group = religion_group:muslim }
					}
				}
			}
		}

		add_reform = government_reform:ghazi_sultanate
		# The adoption handshake: the qadis read the khutba gladly.
		add_estate_satisfaction = { type = estate_type:clergy_estate value = 0.10 }
		add_legitimacy = 5
	}

	# ----- B: the empty seat at Konya -----
	# Open to every beylik: the claim is a program, not a possession
	# — taking Konya is its first step. Holding it already just makes
	# the word far more natural in your mouth.
	option = {
		name = rum.2.b
		trigger = {
			capital ?= { region = region:anatolia_region }
		}
		ai_chance = {
			factor = 15
			# Konya's owner declaring is the Karamanid century begun.
			modifier = {
				factor = 10
				owns = location:konya
			}
			# A beg with infidel neighbors has a cheaper road to a
			# crown — the frontier is right there.
			modifier = {
				factor = 0.5
				any_neighbor_country = {
					NOT = { religion.group = religion_group:muslim }
				}
			}
		}

		add_reform = government_reform:seljuk_claimant
	}

	# ----- C: the tents are enough -----
	option = {
		name = rum.2.c
		ai_chance = {
			factor = 40
			# The strong dare; the small keep their valleys.
			modifier = {
				factor = 2
				num_locations < 15
			}
		}

		add_stability = 5
	}
}
