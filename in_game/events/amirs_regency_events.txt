# ============================================================================
# THE AMIRS' REGENCY — events.
#
# Succession and the onset (this half):
#   .1   the amirs raise a khan            (to the new regent)
#   .2   a child khan                      (to the khan's realm)
#   .10  the old seat answers: acquiesce or break east
#   .12  a player amir nearer Almaliq chooses whom to follow
#   .30  the puppet khan is dead           (to the regent)
#   .40  the regent is dead: the amirs are polled (hidden)
#   .41  the heir keeps the khan
#   .42  another amir takes the khan       (to the new regent)
#   .43  the heir loses the khan           (to the old regent's house)
#   .45  the regent is murdered
#   .50  an amir house's succession: confirm or impose (to the leader)
#   .51  a kinsman is set over the house   (to the member)
#   .52  the house refuses the kinsman     (to the leader)
# The disaster (common/disasters/05_amirs_regency.txt):
#   .100 stage 2, the amirs fall out       (to the regent)
#   .110 stage 3, the shock from outside   (to the regent)
#   .120 stage 4: who is the Ṣāḥib-qirān?  (hidden router)
#   .121 stage 4, the Ṣāḥib-qirān          (to whoever holds the khan)
#   .130 the khan taken by force           (to the captor)
#   .200 the khan's purge succeeds: the Khan Triumphant
#   .201 the khan's purge fails
#   .205 the Khan Triumphant by the sword  (to the victor)
#   .210 the Ṣāḥib-qirān                   (ending)
#   .220 Bulqaq                            (ending)
#
# Effects: common/scripted_effects/15_amirs_regency_effects.txt.
# Numbers: common/script_values/05_amirs_regency_values.txt.
# Source for the history throughout: Beatrice Forbes Manz, The Rise and
# Rule of Tamerlane (1989), chs. 2-3.
# ============================================================================

# ----------------------------------------------------------------------------
# .10 — THE OLD SEAT ANSWERS. The eastern tribes did not accept
# Qazaghan's khans: in 1347 the Dughlat amir Bulaji raised Tughluq
# Temür at Aqsu, and from then on there were two khans of the ulus of
# Chaghatay and, in time, two countries. Here the choice is the khan's
# own: sit under the amirs' khan as one member among others, or leave
# with the east and keep the claim.
#
# The members who follow are those whose seat lies nearer the old
# khan's capital than the regent's (the distance test of base
# flavor_tim.8); from the 1337 roll that is the Dughlats, Issyk-Kul and
# Dzungaria. The AI defies when it is the stronger and yields otherwise.
# ----------------------------------------------------------------------------
amirs_regency.10 = {
	type = country_event
	title = amirs_regency.10.title
	desc = amirs_regency.10.desc
	outcome = negative

	illustration_tags = {
		10 = interior
		10 = angry
	}

	immediate = {
		international_organization:ulus_chaghatay ?= {
			leader_country ?= { save_scope_as = ulus_regent }
		}
	}

	# ----- A: Acquiesce -----
	option = {
		name = amirs_regency.10.a
		ai_chance = {
			factor = 50
			modifier = {
				factor = 0
				military_strength > scope:ulus_regent.military_strength
			}
		}

		add_horde_unity = -10
		custom_tooltip = amirs_regency_acquiesce_tt
		# The tributary dynasties nearer the regent go over to him
		# (Badakhshan, Karategin); Qocho stays with Almaliq.
		hidden_effect = {
			every_subject = {
				limit = {
					NOT = { is_subject_type = ulus }
					capital ?= {
						"this.distance_to(scope:ulus_regent.capital)" < "this.distance_to(root.capital)"
					}
				}
				add_to_list = ulus_moving_tributaries
			}
			every_in_list = {
				list = ulus_moving_tributaries
				save_scope_as = ulus_tributary
				root = { cancel_subject = scope:ulus_tributary }
				make_subject_of = {
					target = scope:ulus_regent
					type = subject_type:vassal
				}
			}
		}
	}

	# ----- B: Defy, and break east -----
	option = {
		name = amirs_regency.10.b
		trigger = { exists = scope:ulus_regent }
		ai_chance = {
			factor = 50
			modifier = {
				factor = 0
				military_strength <= scope:ulus_regent.military_strength
			}
		}

		save_scope_as = ulus_eastern_khan
		set_variable = ulus_eastern_khanate_realm
		add_horde_unity = 10
		custom_tooltip = amirs_regency_defy_tt
		add_casus_belli = {
			type = casus_belli:cb_restore_the_ulus
			target = scope:ulus_regent
		}
		add_opinion = {
			modifier = opinion_ulus_two_khans
			target = scope:ulus_regent
		}
		scope:ulus_regent = {
			add_opinion = {
				modifier = opinion_ulus_two_khans
				target = root
			}
		}
		hidden_effect = {
			international_organization:ulus_chaghatay ?= {
				set_variable = { name = ulus_eastern_khanate value = root }
				every_international_organization_member = {
					limit = {
						NOT = { this = root }
						NOT = { has_variable = ulus_regent_seat }
						capital ?= {
							"this.distance_to(root.capital)" < "this.distance_to(scope:ulus_regent.capital)"
						}
					}
					add_to_list = ulus_eastern_followers
				}
				remove_country_from_international_organization = {
					country = root
					reason = DecidedToLeave
				}
			}
			every_in_list = {
				list = ulus_eastern_followers
				if = {
					limit = { is_ai = yes }
					ulus_follow_east = yes
				}
				else = {
					trigger_event_non_silently = amirs_regency.12
				}
			}
		}
	}
}

# ----------------------------------------------------------------------------
# .30 — THE PUPPET KHAN IS DEAD. Danishmendcha lasted two years before
# Qazaghan replaced him with Bayan Quli; Bayan Quli outlived Qazaghan
# and was killed by his son. Every funeral makes the fiction thinner.
# Fired by on_borjigin_puppet_death (on_action/
# 04_borjigin_puppet_death.txt) on the patron when it holds
# beglerbeg_regency. root = the regent's state.
# ----------------------------------------------------------------------------
amirs_regency.30 = {
	type = country_event
	title = amirs_regency.30.title
	desc = amirs_regency.30.desc
	outcome = negative

	illustration_tags = {
		10 = interior
		10 = angry
	}

	immediate = {
		save_scope_as = ulus_kingmaker
		save_scope_as = ulus_new_regent
		international_organization:ulus_chaghatay ?= {
			remove_variable = ulus_held_khan
			remove_variable = ulus_held_khan_age
		}
	}

	# ----- A: Enthrone another -----
	option = {
		name = amirs_regency.30.a
		trigger = {
			ulus_khan_material_exists = yes
			political_influence >= ulus_enthrone_khan_cost
		}
		ai_chance = { factor = 60 }

		add_political_influence = ulus_enthrone_khan_pi
		custom_tooltip = amirs_regency_enthrone_tt
		hidden_effect = {
			ulus_pick_puppet_khan = yes
			ulus_seat_khan = yes
			international_organization:ulus_chaghatay ?= {
				ulus_change_progress = { AMOUNT = ulus_regency_progress_khan_death }
				ulus_chaghatay_rebuild_ledger = yes
			}
		}
	}

	# ----- B: A khan of our own daughter's line -----
	option = {
		name = amirs_regency.30.b
		trigger = {
			ulus_house_has_inlaw_tie = yes
			political_influence >= ulus_enthrone_inlaw_cost
		}
		ai_chance = { factor = 80 }

		add_political_influence = ulus_enthrone_inlaw_pi
		custom_tooltip = amirs_regency_enthrone_inlaw_tt
		set_variable = ulus_inlaw_house
		hidden_effect = {
			if = {
				limit = { exists = scope:ulus_puppet }
				clear_saved_scope = ulus_puppet
			}
			create_character = {
				dynasty = dynasty:borjigin_dynasty
				estate = estate_type:nobles_estate
				religion = root.religion
				culture = root.culture
				age = { 4 12 }
				female = no
				save_scope_as = ulus_puppet
			}
			ulus_seat_khan = yes
			ruler ?= { set_variable = ulus_gurkan }
			international_organization:ulus_chaghatay ?= {
				ulus_chaghatay_rebuild_ledger = yes
			}
		}
	}

	# ----- C: Rule without a khan -----
	option = {
		name = amirs_regency.30.c
		ai_chance = {
			factor = 5
			modifier = {
				factor = 10
				NOT = { ulus_khan_material_exists = yes }
			}
		}

		custom_tooltip = amirs_regency_no_khan_tt
		add_horde_unity = -20
		hidden_effect = {
			international_organization:ulus_chaghatay ?= {
				set_variable = ulus_no_khan
				if = {
					limit = { has_variable = ulus_gap }
					set_variable = {
						name = ulus_gap
						value = {
							value = var:ulus_gap
							divide = 2
						}
					}
				}
				ulus_change_progress = { AMOUNT = ulus_regency_progress_ruling_alone }
				ulus_chaghatay_rebuild_ledger = yes
			}
		}
	}
}

# ----------------------------------------------------------------------------
# .40 — THE REGENT IS DEAD: THE AMIRS ARE POLLED. Hidden.
#
# 1358. Qazaghan's son ʿAbdallāh inherited the Qara'unas and assumed he
# had inherited the ulus. He moved the court to Samarkand, killed Bayan
# Quli Khan, and within the year Bayan Suldus and Hajji Barlas had
# driven him out. The house's own succession gives the heir the STATE;
# it does not give him the khan. He keeps the regency only if his
# standing is at least ulus_regent_heir_ratio of the best rival's.
#
# Fired 30-80 days after a new ruler is seated in the regent's state
# (on_action/31_amirs_regency_hooks.txt), by which time root's ruler IS
# the heir and ulus_kingmaker_score reads him.
# ----------------------------------------------------------------------------
amirs_regency.40 = {
	type = country_event
	hidden = yes
	outcome = neutral
	title = empty_text
	desc = empty_text

	trigger = {
		is_ulus_regent = yes
	}

	immediate = {
		save_scope_as = ulus_old_regent
		set_local_variable = { name = ulus_heir_score value = 0 }
		set_local_variable = { name = ulus_rival_score value = 0 }
		if = {
			limit = { exists = scope:ulus_best_rival }
			clear_saved_scope = ulus_best_rival
		}
		international_organization:ulus_chaghatay ?= {
			ulus_change_progress = { AMOUNT = ulus_regency_progress_regent_death }
			ulus_chaghatay_rebuild_ledger = yes
			save_temporary_scope_as = ulus_io
			ordered_international_organization_member = {
				limit = {
					NOT = { is_leader_of_international_organization = scope:ulus_io }
					has_ruler = yes
				}
				order_by = ulus_kingmaker_score
				max = 1
				check_range_bounds = no
				save_scope_as = ulus_best_rival
				set_local_variable = { name = ulus_rival_score value = ulus_kingmaker_score }
			}
		}
		set_local_variable = { name = ulus_heir_score value = ulus_kingmaker_score }

		if = {
			limit = {
				exists = scope:ulus_best_rival
				local_var:ulus_heir_score < {
					value = local_var:ulus_rival_score
					multiply = ulus_regent_heir_ratio
				}
			}
			scope:ulus_best_rival = { save_scope_as = ulus_new_regent }
			ulus_transfer_regency = yes
			# No single beg replaces him: the amirs fall out (stage 2),
			# if they had not already.
			international_organization:ulus_chaghatay ?= {
				ulus_regency_advance_to_stage_2 = yes
			}
			scope:ulus_new_regent = { trigger_event_non_silently = amirs_regency.42 }
			# The dispossessed house may fight to take the khan back.
			add_casus_belli = {
				type = casus_belli:cb_take_the_khan
				target = scope:ulus_new_regent
			}
			trigger_event_non_silently = amirs_regency.43
		}
		else = {
			trigger_event_non_silently = amirs_regency.41
		}
	}
}

# ----------------------------------------------------------------------------
# .50 — AN AMIR HOUSE'S SUCCESSION. The khan's right over his tümens
# was to confirm who led them, and a strong khan used it: Kebek and
# Tarmashirin reshuffled commands. Confirming the house's own heir is
# free and remembered; setting another kinsman over it costs political
# influence and the house's goodwill. Under a regent the cost doubles;
# once the amirs have fallen out (stage 2) a house may refuse and take
# up arms; once the ulus is under attack from outside (stage 3) no
# regent can impose anyone.
#
# root = the leader of the ulus. scope:ulus_amir_house = the member
# whose ruler has just changed (on_action/31_amirs_regency_hooks.txt).
# ----------------------------------------------------------------------------
amirs_regency.50 = {
	type = country_event
	title = amirs_regency.50.title
	desc = amirs_regency.50.desc
	outcome = neutral

	illustration_tags = { 10 = interior }

	trigger = {
		exists = scope:ulus_amir_house
	}

	immediate = {
		if = {
			limit = { exists = scope:ulus_kinsman }
			clear_saved_scope = ulus_kinsman
		}
		scope:ulus_amir_house = {
			ordered_character = {
				limit = {
					is_alive = yes
					is_adult = yes
					is_female = no
					is_ruler = no
					owner ?= scope:ulus_amir_house
					dynasty ?= scope:ulus_amir_house.ruler.dynasty
				}
				order_by = total_abilities
				max = 1
				check_range_bounds = no
				save_scope_as = ulus_kinsman
			}
		}
	}

	# ----- A: Confirm the heir -----
	option = {
		name = amirs_regency.50.a
		ai_chance = { factor = 85 }

		scope:ulus_amir_house = {
			add_opinion = {
				modifier = opinion_ulus_heir_confirmed
				target = root
			}
		}
	}

	# ----- B: Impose a kinsman -----
	option = {
		name = amirs_regency.50.b
		trigger = {
			exists = scope:ulus_kinsman
			NOT = { ulus_regency_stage_at_least = { STAGE = 3 } }
			trigger_if = {
				limit = { is_ulus_regent = yes }
				political_influence >= ulus_impose_kinsman_regent_cost
			}
			trigger_else = {
				political_influence >= ulus_impose_kinsman_cost
			}
		}
		ai_chance = {
			factor = 15
			modifier = {
				factor = 0
				ulus_regency_stage_at_least = { STAGE = 2 }
			}
		}

		if = {
			limit = { is_ulus_regent = yes }
			add_political_influence = ulus_impose_kinsman_regent_pi
		}
		else = {
			add_political_influence = ulus_impose_kinsman_pi
		}
		scope:ulus_amir_house = {
			add_opinion = {
				modifier = opinion_ulus_kinsman_imposed
				target = root
			}
		}
		if = {
			limit = { ulus_regency_stage_at_least = { STAGE = 2 } }
			custom_tooltip = amirs_regency_may_refuse_tt
			hidden_effect = {
				random_list = {
					50 = {
						ulus_impose_kinsman = yes
					}
					50 = {
						scope:ulus_amir_house = {
							add_casus_belli = {
								type = casus_belli:cb_take_the_khan
								target = root
							}
						}
						trigger_event_non_silently = amirs_regency.52
					}
				}
			}
		}
		else = {
			custom_tooltip = amirs_regency_impose_tt
			hidden_effect = {
				ulus_impose_kinsman = yes
			}
		}
	}
}

# ----------------------------------------------------------------------------
# .110 — STAGE 3, THE SHOCK FROM OUTSIDE. In 1360 Tughluq Temür, khan
# of the eastern half since 1347, crossed the Syr Darya. The western
# amirs did not combine against him: Hajji Barlas fled to Khurasan and
# was killed there, and a young kinsman of his, Temür, made his
# submission and was given Kesh. The khan came again in 1361.
#
# The regent's three strongest neighbours outside the ulus are handed
# a claim on it for ten years; the eastern khanate, if there is one,
# gets it beside its standing claim. Members may now leave, and
# whoever takes the regent's capital takes the khan.
# ----------------------------------------------------------------------------
amirs_regency.110 = {
	type = country_event
	title = amirs_regency.110.title
	desc = amirs_regency.110.desc
	outcome = negative

	illustration_tags = {
		10 = exterior
		10 = angry
	}

	immediate = {
		save_scope_as = ulus_regent
		hidden_effect = {
			ordered_neighbor_country = {
				limit = {
					NOT = { is_ulus_chaghatay_member = yes }
					is_subject = no
					is_rebel_country = no
				}
				order_by = military_strength
				max = 3
				check_range_bounds = no
				add_casus_belli = {
					type = casus_belli:cb_ulus_in_disarray
					target = scope:ulus_regent

# ----------------------------------------------------------------------------
# .120 — STAGE 4: WHO IS THE ṢĀḤIB-QIRĀN? Hidden.
#
# By 1370 the man who held the khan was not the heir of the man who
# first took one. He was the amir who had come out of the wars of the
# 1360s with the greatest name. So at the turn into the last stage the
# amirs are measured once more, renown and all: if a rival now stands
# higher than the regent, the khan goes to him.
# ----------------------------------------------------------------------------
amirs_regency.120 = {
	type = country_event
	hidden = yes
	outcome = neutral
	title = empty_text
	desc = empty_text

	trigger = {
		is_ulus_regent = yes
	}

	immediate = {
		set_local_variable = { name = ulus_my_score value = ulus_kingmaker_score }
		set_local_variable = { name = ulus_rival_score value = 0 }
		if = {
			limit = { exists = scope:ulus_best_rival }
			clear_saved_scope = ulus_best_rival
		}
		international_organization:ulus_chaghatay ?= {
			save_temporary_scope_as = ulus_io
			ordered_international_organization_member = {
				limit = {
					NOT = { is_leader_of_international_organization = scope:ulus_io }
					has_ruler = yes
					# A khan of the line does not become a beg's
					# successor; his road is the Khan Triumphant.
					NOT = { ulus_ruler_is_of_the_line = yes }
				}
				order_by = ulus_kingmaker_score
				max = 1
				check_range_bounds = no
				save_scope_as = ulus_best_rival
				set_local_variable = { name = ulus_rival_score value = ulus_kingmaker_score }
			}
		}
		if = {
			limit = {
				exists = scope:ulus_best_rival
				local_var:ulus_rival_score > local_var:ulus_my_score
			}
			scope:ulus_best_rival = { save_scope_as = ulus_new_regent }
			ulus_transfer_regency = yes
			scope:ulus_new_regent = { trigger_event_non_silently = amirs_regency.121 }
			trigger_event_non_silently = amirs_regency.43
		}
		else = {
			trigger_event_non_silently = amirs_regency.121
		}
	}
}
