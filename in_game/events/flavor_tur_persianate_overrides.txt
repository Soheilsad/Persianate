namespace = flavor_tur_persianate

# =====================================================================
# BASE OTTOMAN FLAVOR — persianate twin
#
# Under Shattered Rum, TUR runs government_type persianate_sultanate
# for the whole campaign, which kills base's monarchy-gated
#   flavor_tur.95 — The One True Emperor (1537+)
# Events cannot be REPLACE:d (1.4.7 error.log: "'REPLACE:flavor_tur.95'
# does not have a valid namespace"), and overriding one event would mean
# shipping all of base's 18,900-line flavor_TUR.txt. So this is a TWIN in
# its own namespace: base 1.4.7's event verbatim, gated to
# persianate_sultanate instead of monarchy (the two never both fire), and
# reusing base's flavor_tur.95.* loc keys.
#
# flavor_tur.88 (The Imperial Harem) was dropped: base 1.4 removed the
# event and its loc.
# On a patch: re-copy base's flavor_tur.95 and reapply the one-line gate.
# =====================================================================

flavor_tur_persianate.95 = { #The One True Emperor
	type = country_event
	title = flavor_tur.95.title
	desc = flavor_tur.95.desc
	outcome = neutral

	fire_only_once = yes
	dynamic_historical_event = {
		tag = TUR
		from = 1537.1.1
		to = 1737.1.1
		monthly_chance = 15
	}

	historical_info = flavor_tur.95.historical_info

	trigger = {

		has_ruler = yes

		government_power < 70

		country_rank = country_rank:rank_empire

		government_type = government_type:persianate_sultanate

		"religion_group_percentage_in_country(religion_group:christian)" > 0

		custom_tooltip = {
			text = variable_dilsiz_tt
			has_variable = dilsiz
		}

		owns = location:constantinople

	}

	immediate =  {

		root.ruler = { save_scope_as = target_ruler }
	}

	option = {
		name = flavor_tur.95.a
		historical_option = yes

		change_gold_effect = { scale = -3 }

		custom_tooltip = {
			text = christian_support.tt
			every_owned_location = {
				every_pop = {
					limit = {
						owner = root
						religion.group = religion_group:christian
					}
					add_pop_satisfaction = pop_satisfaction_mild_bonus
				}
			}
		}

		random_list = {

			40 = {
				add_government_power = government_power_mild_bonus
			}
			60 = {
				add_government_power = government_power_severe_bonus
			}
		}
	}

	option = {
		name = flavor_tur.95.b

		change_gold_effect = { scale = -1 }

		add_government_power = government_power_mild_bonus
	}

	option = {
		name = flavor_tur.95.c

		add_estate_satisfaction = { type = estate_type:clergy_estate value = 0.15 }
	}
}
