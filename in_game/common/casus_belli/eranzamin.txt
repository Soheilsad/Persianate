# Erānzamīn succession CB.
#
# The primary contender-vs-contender CB. Usable when both actor and target
# are members of the eranzamin IO. The wargoal is region-limited
# (annex/subjugate only locations in the contested region), so two
# contenders fighting over Erānzamīn don't end up carving up Egypt.
#
# Conquer/subjugate cost on the wargoal stays at base-ilkhanate values
# (0.75); the IO's war_score_cost_modifier provides the additional
# discount on top — same arrangement as base cb_unify_ilkahante.

cb_eranzamin_succession = {

	create_visible = {
		is_member_of_international_organization = international_organization:eranzamin
		scope:target = {
			is_member_of_international_organization = international_organization:eranzamin
		}
	}

	declare_enabled = {
		is_member_of_international_organization = international_organization:eranzamin
		scope:target = {
			is_member_of_international_organization = international_organization:eranzamin
		}
	}

	speed = 10

	war_goal_type = eranzamin_succession_wargoal
}


cb_ghozva = {
	create_visible = {
		OR = {
			has_reform = government_reform:ghazi_sultanate
			has_reform = government_reform:dar_al_jihad
		}
		NOT = { religion.group = scope:target.religion.group } 
	}
	declare_enabled = {
		NOT = { religion.group = scope:target.religion.group } 
	}
	
	#scopes are actor/recipient for countries, while root = province checked.
	province = {
		any_location_in_province = {
			always = yes
		}
	}
	custom_tags = { religious_casus_belli }

	war_goal_type = religious_conquer_province
}
