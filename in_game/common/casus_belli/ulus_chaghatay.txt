# Casus belli of the Amirs' Regency. root = the country holding the CB,
# scope:target = its target (base casus_belli/readme.txt). All are
# handed out by script (add_casus_belli), never fabricated: speed is
# left at its default and create_visible / create_enabled are closed
# (base casus_belli/01_event_triggered.txt).

# THE KHAN'S CLAIM. Held by a realm of the line against the beg who
# rules the ulus through a khan of his own making: Tughluq Temür's
# title to Transoxiana in 1360 was exactly this, that he was a
# Chaghatayid khan and the men holding it were servants. Standing: it
# does not lapse while the regency lasts (can_expire = no; the
# declare gate closes when the target no longer holds the seat).
cb_restore_the_ulus = {
	create_visible = {
		always = no
	}
	create_enabled = {
		always = no
	}
	declare_enabled = {
		scope:target = {
			OR = {
				has_variable = ulus_regent_seat
				has_variable = ulus_sahib_qiran
				# After the Bulqaq: every tümen that went its own way.
				has_variable = ulus_bulqaq_orphan
			}
		}
	}
	can_expire = no
	war_goal_type = restore_the_ulus_wargoal
}

# TAKING THE KHAN. One member of the ulus against the member that
# holds the khan: Husayn and Temür against each other after 1366, and
# every amir against ʿAbdallāh b. Qazaghan in 1358. The prize is the
# regent's ordu, where the khan is kept (the capture itself is the
# siege hook in on_action/31_amirs_regency_hooks.txt).
cb_take_the_khan = {
	create_visible = {
		always = no
	}
	create_enabled = {
		always = no
	}
	declare_enabled = {
		is_ulus_chaghatay_member = yes
		scope:target = {
			has_variable = ulus_regent_seat
		}
	}
	war_goal_type = take_the_khan_wargoal
}

# THE ULUS IN DISARRAY. Handed to the regent's strongest neighbours
# when the regency reaches its third stage: a fractured ulus with a
# beg at its head is an invitation, and Tughluq Temür was not the only
# one to read it so (the Kartids of Herat raided to the Oxus in the
# same years). Ten years (add_casus_belli sets the term).
cb_ulus_in_disarray = {
	create_visible = {
		always = no
	}
	create_enabled = {
		always = no
	}
	declare_enabled = {
		scope:target = {
			has_variable = ulus_regent_seat
		}
	}
	war_goal_type = ulus_in_disarray_wargoal
}

# THE ṢĀḤIB-QIRĀN'S RECKONING. Against each amir who did not bow when
# the ulus became one man's realm: Temür's campaigns of the 1370s
# against the Jalayir and the Sufis of Khwarazm.
cb_sahib_qiran_submission = {
	create_visible = {
		always = no
	}
	create_enabled = {
		always = no
	}
	declare_enabled = {
		has_variable = ulus_sahib_qiran
	}
	war_goal_type = sahib_qiran_submission_wargoal
}

# RISING AGAINST THE KHAN. Given by the action of that name to an amir
# while the regency bar is met and no regency stands: Qazaghan against
# Qazan Khan, 1346. Take the khan's ordu and he is yours to replace.
cb_rise_against_the_khan = {
	create_visible = {
		always = no
	}
	create_enabled = {
		always = no
	}
	declare_enabled = {
		has_variable = ulus_rising
		scope:target = {
			is_ulus_chaghatay_leader = yes
			NOT = { has_variable = ulus_regent_seat }
		}
	}
	war_goal_type = take_the_khan_wargoal
}


# BREAK THE TÜMENS. After the Bulqaq, an outsider against a tümen that
# has gone its own way (given by ulus_bulqaq_open_the_pastures to every
# outside neighbour of each orphan).
cb_break_the_tumens = {
	create_visible = {
		always = no
	}
	create_enabled = {
		always = no
	}
	declare_enabled = {
		scope:target = {
			has_variable = ulus_bulqaq_orphan
		}
	}
	war_goal_type = restore_the_ulus_wargoal
}

# THE FEUD OF THE TÜMENS. Once the amirs have fallen out, a house in one
# bloc against a house in another (given by ulus_press_the_feud).
cb_tumen_feud = {
	create_visible = {
		always = no
	}
	create_enabled = {
		always = no
	}
	declare_enabled = {
		is_ulus_chaghatay_amir = yes
		ulus_regency_stage_at_least = { STAGE = 2 }
		scope:target = {
			is_ulus_chaghatay_amir = yes
		}
	}
	war_goal_type = tumen_feud_wargoal
}
