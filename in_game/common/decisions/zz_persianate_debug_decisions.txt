persianate_debug_begin_regency = { # DEBUG: the amirs raise a khan now
	decision_category = country_specific_decisions

	potential = {
		is_ai = no
		is_ulus_chaghatay_member = yes
		international_organization:ulus_chaghatay ?= {
			NOT = { has_variable = ulus_regency }
		}
	}

	allow = {
		always = yes
	}

	ai_will_do = {
		value = 0
	}

	option = {
		name = persianate_debug_begin_regency.a
		ai_chance = { value = 0 }
		effect = {
			international_organization:ulus_chaghatay ?= {
				leader_country ?= { save_scope_as = ulus_old_seat }
				ulus_begin_regency = yes
			}
		}
	}
}

persianate_debug_sahib_qiran = { # DEBUG: stage 4, a wife of the line, the gap filled five years
	decision_category = country_specific_decisions

	potential = {
		is_ai = no
		is_ulus_chaghatay_member = yes
		international_organization:ulus_chaghatay ?= {
			has_variable = ulus_regency
		}
	}

	allow = {
		# The one thing this cannot supply: a living khan in the
		# regent's keeping (enthrone one from the panel first).
		international_organization:ulus_chaghatay ?= {
			var:ulus_held_khan ?= { is_alive = yes }
		}
	}

	ai_will_do = {
		value = 0
	}

	option = {
		name = persianate_debug_sahib_qiran.a
		ai_chance = { value = 0 }
		effect = {
			international_organization:ulus_chaghatay ?= {
				set_variable = { name = ulus_regency_stage value = 4 }
				set_variable = { name = ulus_regency_progress value = 0 }
				set_variable = { name = ulus_gap value = 100 }
				set_variable = { name = ulus_gap_filled_months value = ulus_sahib_qiran_months }
				set_variable = { name = ulus_gap_empty_months value = 0 }
				leader_country ?= {
					ruler ?= { set_variable = ulus_gurkan }
				}
				ulus_chaghatay_rebuild_ledger = yes
			}
		}
	}
}

persianate_debug_cadet_flight = { # DEBUG: the cadet takes the tümen (stage 3 route)
	decision_category = country_specific_decisions

	potential = {
		is_ai = no
		is_ulus_chaghatay_amir = yes
		NOT = { has_variable = ulus_fled_chief }
	}

	allow = {
		custom_tooltip = {
			text = persianate_debug_cadet_needs_kinsman_tt
			ulus_house_has_cadet = yes
		}
	}

	ai_will_do = {
		value = 0
	}

	# The invader is the state that leads the ulus (khan or regent): the
	# cadet swears to him, with no subject tie.
	option = {
		name = persianate_debug_cadet_flight.a
		ai_chance = { value = 0 }
		effect = {
			international_organization:ulus_chaghatay ?= {
				leader_country ?= { save_scope_as = ulus_cadet_invader }
			}
			if = {
				limit = { exists = scope:ulus_cadet_invader }
				trigger_event_non_silently = amirs_cadet.1
			}
		}
	}
	# The invader is a neighbour from outside the ulus: the house becomes
	# his tributary.
	option = {
		name = persianate_debug_cadet_flight.b
		ai_chance = { value = 0 }
		effect = {
			random_neighbor_country = {
				limit = {
					NOT = { is_ulus_chaghatay_member = yes }
					is_rebel_country = no
				}
				save_scope_as = ulus_cadet_invader
			}
			if = {
				limit = { exists = scope:ulus_cadet_invader }
				trigger_event_non_silently = amirs_cadet.1
			}
		}
	}
}
