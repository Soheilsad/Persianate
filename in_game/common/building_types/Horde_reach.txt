# =====================================================================
# HORDE REACH — the khan's appointee in a settled place.
#
# The Jochid ulus was pluricentric in fact, not by approximation. There
# was no single Sarai: Sarai Batu on the lower Volga, Sarai al-Jadid
# upstream (Özbeg moved the seat there), and Saray-Jük — "little Sarai"
# — out on the Yaik on the eastern road. They named their second and
# third cities after the first. And the khan posted resident men to
# hold the settled provinces: Qutlugh-Temür governed Khwarazm from
# Urgench as Özbeg's amir, married to a Chinggisid princess; Tuluktumur
# was the amir at Solkhat in Crimea. Ibn Battuta sat with both of them
# in 1333–34. Behind them stands the whole darughachi / basqaq
# institution — a resident official holding a district for the khan,
# running the census, the assessment and the garrison liaison.
#
# Base game's `local_governor` models this and then locks it away from
# hordes twice over: `city = yes` only, and `has_road_to_capital = yes`.
# The Mongol claim is precisely that they governed without roads. This
# is that building with the road gate removed and the rank widened to
# town, because a darugha was posted to a district centre, not only to
# a metropolis — and because Bulgar/Kazan, the worst-controlled place
# in the empire at start, is rated a town.
#
# Deliberately NOT rural: this is the settled seat. The country between
# seats is the yam's job, and the yam is a different institution.
#
# CALIBRATION: the Horde had high REACH and low DENSITY. This grants
# proximity (reach) at 60 — below the governor's 80, because a khan's
# appointee is not a chancery — and no max-control of its own. The
# steppe should end up reachable, never administered.
# =====================================================================

horde_local_appointee = {
	audio_tier = 2
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	important_for_UI = yes

	employment_size = large_noble_employment

	category = government_category

	rural_settlement = no
	town = yes                  # FIX: Kazan and Chimgi-Tura are towns.
	city = yes
	megalopolis = yes

	# FIX: country-scope triggers belong in country_potential. Base game
	# does exactly this (capital_buildings.txt: `country_potential = {
	# government_type ?= government_type:monarchy }`). In
	# location_potential these would have been evaluated in location
	# scope, where `government_type` and `ruler` are not valid.
	country_potential = {
		government_type = government_type:steppe_horde
		OR = {
			ruler ?= { dynasty ?= dynasty:borjigin_dynasty }
			has_reform ?= government_reform:chinggisid_regency
		}
	}

	allow = {
		# FIX: the cap must count THIS building. As written it counted
		# `local_governor`, which a horde never builds — so the check
		# read 0 < 1 forever and the appointee was uncapped.
		# The three government seats share one budget; hordes get a
		# bigger budget via `num_local_governors` on the auto modifier.
		custom_tooltip = {
			text = horde_local_appointee_tt
			scope:actor = {
				total_building_levels_including_construction:horde_local_appointee < {
					value = modifier:num_local_governors
					subtract = total_building_levels_including_construction:lieutenancy
					subtract = total_building_levels_including_construction:local_governor
					add = 4 ### a minimum
				}
			}
		}
		owner = scope:actor #not in subjects
	}

	location_potential = {
		owner.capital != this
	}

	# The reach dies with the legitimacy that carried it — Mamai's
	# problem after 1359: real armies, no descent, and an ulus that
	# would not answer him. NOTE: this DESTROYS the buildings rather
	# than suspending them; see the note at the end of this file.
	remove_if = {
		owner = {
			NOR = {
				AND = {
					government_type = government_type:steppe_horde
					ruler ?= { dynasty ?= dynasty:borjigin_dynasty }
				}
				AND = {
					government_type = government_type:steppe_horde
					has_reform ?= government_reform:chinggisid_regency
				}
			}
		}
	}

	expensive = yes

	automation_build_allowed = no
	ai_forbid_shutdown = yes
	forbidden_for_estates = yes
	important_for_AI = yes
	ai_ignore_maintenance = yes

	build_time = government_build_time

	# FIX: renamed from `local_governor_maintenance`, which is already
	# defined by base game inside `local_governor`
	# (capital_buildings.txt:663) and carries its own global loc key in
	# goods_l_english.yml. Production-method keys are global; reusing
	# the name is a duplicate-definition collision.
	unique_production_methods = {
		# No goods_gold: a darugha is a man with a tent, a guard and a
		# tally of who owes what — not a chancery with a treasury. The
		# gold line was also unpayable in practice, since the steppe has
		# no bullion RGO of its own and imports its specie.
		horde_appointee_maintenance = {
			tools = 0.5
			masonry = 0.5
			weaponry = 1
			paper = 2
			books = 0.5
			furniture = 1
			fine_cloth = 1

			category = building_maintenance
		}
	}

	construction_demand = horde_appointee_construction

	# =================================================================
	# THE SEAT PRODUCES A MAN.
	#
	# A darugha is not an office on an organisation chart. He is a
	# specific person with a guard, a following and a name people know:
	# Qutlugh-Temür at Urgench, married to a Chinggisid princess and
	# outranking every merchant on the Amu Darya; Tuluktumur at Solkhat.
	# Ibn Battuta dined with both. So posting a seat should put an actual
	# beg into the realm's character list, and it is HE who accumulates
	# standing — not an abstraction.
	#
	# HIGH STATS ON PURPOSE. A darugha is appointed; a prince is born.
	# That asymmetry is the entire reason the begs eventually won: Mamai
	# and Edigü were formidable soldiers holding office on merit, while
	# the late Batuids inherited theirs. In the törü a beg carries no
	# blood bonus at all and must beat a prince on capability alone —
	# which a good one can, and a bad one cannot. That is the contest
	# the darugha road is supposed to create.
	#
	# `darugha_of_the_khan` is what the succession law reads to give him
	# his standing (common/heir_selections/horde_toru.txt).
	#
	# NOTE: the man OUTLIVES the post. `remove_if` destroys these
	# buildings when the Borjigin line fails, but the begs it created
	# remain — armed, experienced, and now holding nothing of anyone.
	# That is Mamai in 1359 exactly, and it is deliberate.
	# =================================================================
	# The man is created by horde_reach.20 ("A Darugha Is Appointed"),
	# which is a visible event rather than a silent spawn — the khan
	# should be told who he has just put in that town, and told in the
	# same breath what such a man eventually becomes.
	on_built = {
		location = { save_scope_as = darugha_seat }
		owner = {
			if = {
				limit = {
					government_type = government_type:steppe_horde
					exists = ruler
				}
				trigger_event_non_silently = horde_reach.20
			}
		}
	}

	# THE PRICE OF THE DARUGHA ROAD.
	#
	# A khan has two ways to hold a distant corner, and they fail
	# differently. Grant it to a son as an ulus and you have created a
	# Borjigin with a base, who can one day be khan himself — that road
	# ends in fragmentation. Post a darugha and you have created an amir
	# holding your authority who can NEVER be khan, because he has no
	# blood. His only route to power is to control the man who does.
	#
	# So every seat quietly thickens the military aristocracy inside
	# your own realm, and a khan who governs entirely through appointees
	# eventually finds that his appointees govern him. That is not a
	# design conceit: it is the office Mamai held after 1359, and Edigü
	# after 1397, and Qazaghan in Transoxiana from 1347 — beglerbegs
	# with real armies and no descent, making and unmaking khans for
	# twenty years at a stretch. Özbeg took this road deliberately when
	# he purged the princes in 1313, and it is why the Horde's FIRST
	# crisis was a regency and not a partition.
	#
	# ARITHMETIC: 0.05 a seat, stacking per instance. The auto modifier
	# grants hordes `num_local_governors = 8`, and the allow-block adds
	# a floor of 4, so a fully built-out network of ~12 seats carries
	# roughly +0.60 nobles power — major, and meant to be. Compare the
	# mod's own laws at 0.10-0.50 and the turk_vs_tazik slider at 0.50.
	# This is the dial if the amirs end up too strong or too tame.
	#
	# BOOT CHECK: raw_modifier is the block that takes country-scope
	# fields and stacks per building instance; base only ever puts
	# global estate power in `capital_country_modifier`, which is
	# useless here because a darugha is never in the capital
	# (location_potential forbids it). If the nobles-power line does not
	# appear in the country tooltip, the fallback is a tiered auto
	# modifier keyed to total_building_levels:horde_local_appointee.
	raw_modifier = {
		global_nobles_estate_power = 0.30
	}

	modifier = {
		local_proximity_source = 60
	}
}

# ---------------------------------------------------------------------
# NOTES / OPEN DECISIONS
#
# 1. `remove_if` deletes rather than suspends. When the Borjigin line
#    fails, every appointee in the realm is destroyed at once and must
#    be rebuilt from nothing by whoever restores legitimacy. That IS
#    the historical claim, and Tokhtamysh did have to reconstitute the
#    Horde — but watch it during ordinary succession gaps, where a
#    momentary non-Borjigin ruler would wipe the whole network for
#    reasons the player will read as a bug. If that happens, the
#    alternative is to drop `remove_if` and move the conditional
#    benefit onto a country modifier instead.
#
# 2. Solkhat, Sighnaq and Majar are rated RURAL by base game and so
#    cannot take this building. That excludes Crimea's administrative
#    capital — Qırım gave the peninsula its name and was Tuluktumur's
#    seat — and Sighnaq, the left wing's seat on the Syr Darya. This
#    looks like the same class of error as Sarai al-Jadid being rated
#    a town: a Jochid centre read as empty steppe. Either correct their
#    rank in the mod's 07 overrides, or let the (rural-capable) yam
#    building cover them. Left undecided on purpose.
# ---------------------------------------------------------------------
