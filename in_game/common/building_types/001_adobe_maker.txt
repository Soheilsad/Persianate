adobe_maker = {
	is_foreign = no
	max_levels = rural_building_cap
	pop_type = laborers
	
	employment_size = rural_peasant_produce_employment
	
	category = rgo_building_category
	
	rural_settlement = yes
	town = yes
	city = yes
	
	location_potential = {
		NOT = { raw_material = goods:lumber }
		OR = {
			vegetation = desert
			vegetation = sparse
			vegetation = qanat_farmland
			vegetation = crumbling_qanat
		}
	}
	
	# Adobe is local earth, straw, and wooden molds - the poor man's
	# brick precisely BECAUSE its inputs are nearly free. The old
	# numbers (0.6 clay + 0.4 tools = 1.5 cost against 1.5 lumber)
	# broke even at best and starved at Iranian clay prices. Now:
	# 0.4 clay + 0.2 tools = 0.8 cost against 1.5 - a real margin,
	# and still a real clay demand to keep the pits in business.
	unique_production_methods = {
		adobe_production = {
			clay = 0.4
			tools = 0.2
			produced = lumber
			output = 1.0
			category = building_maintenance
		}
	}
	
	build_time = rural_build_time
	construction_demand = basic_construction_needs
}
