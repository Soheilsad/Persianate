# ============================================================
# Pastoral buildings — the īl, the amīr, and the horse
#
# The base game's horse_breeder is locked to Mediterranean and
# Oceanic climates, which excludes the entire Iranian plateau,
# Khurasan, Transoxiana, and the Anatolian/Caucasus highlands —
# the actual horse-producing zone of medieval Eurasia. Worse, it
# uses peasant labor and a wheat-fed stud economy, neither of which
# fits the pastoral pattern that obtains east of the Mediterranean.
#
# Two buildings, both pastoral, both rural, both producing horses
# from grass:
#
#   tribal_pasture (tribes_estate) — the īl's own grazing range.
#     5k tribesmen per level. Heavy on labor, heavy on the location's
#     carrying capacity: the grass that feeds the herd is grass not
#     ploughed, and at full extent the ploughmen have to leave. This
#     is the building that lets a tribal confederation subsist as a
#     confederation rather than a tax base.
#
#   amirs_herd (amirs_estate) — the amīr's organized stud, the
#     formal counterpart to the īl's open range. Smaller employment
#     and a lighter carrying-capacity cost, because the amīr's
#     herd is a managed institution sitting on a defined parcel
#     rather than the open pasture of the whole tribe. Slightly
#     higher levy throughput per pop: the amīr converts horseflesh
#     into ready cavalry more reliably than the open īl does.
#
# Both gated to:
#   - the Iranian-plateau-and-neighbors region set
#   - the three plateau-and-steppe climates
#   - rural only — pasture is not a town institution
# ============================================================


# --- Tribal Pasture — the īl's own range ---------------------
tribal_pasture = {
	is_foreign = no
	max_levels = 5
	pop_type = tribesmen
	employment_size = 5.0       # 5k tribesmen per level — a tribe-scale commitment

	category = rgo_building_category


	rural_settlement = yes
	town = no
	city = no
	megalopolis = no

	build_time = estate_build_time
	destroyable_building = yes

	location_potential = {
		OR = {
			region = region:persia_region
			region = region:khorasan_region
			region = region:crescent_region
			region = region:anatolia_region
			region = region:caucasus_region
			region = region:steppes_region
		}
		OR = {
			climate = arid
			climate = cold_arid
			climate = continental
		}
		OR = {
			vegetation = grasslands
			vegetation = sparse ## will stress it more
			vegetation = seasonal_irrigated ## will stress it less
		}
		num_pop_type:tribesmen > 5.0
		local_control < 0.33


	}

	possible_production_methods = {
		tribal_pasture_grazing
		tribal_livestock_production
	}

	construction_demand = estate_construct_building

	modifier = {
		# Pasture displaces ploughland. -7.5k per level is the heavy
		# version — at full max-level extent (5 levels), the location
		# loses 37.5k carrying capacity, which on grassland is enough
		# to drive the agriculturalists out entirely.
		local_population_capacity = -7.5
		# The tribal levy: the herd IS the cavalry pool.
		local_tribes_estate_power = 1.5
		local_levy_size_modifier = 0.1
	}

	graphical_tags = { small_military }
}


### smaller, steppe variant. slightly more productive, but for thinner land.

	is_foreign = no
	max_levels = 20
	pop_type = tribesmen

	town = no
	city = no
	megalopolis = no

	build_time = estate_build_time

	location_potential = {
		OR = {
			region = region:khorasan_region
			region = region:steppes_region
		}
		OR = {
			climate = cold_arid
			climate = continental
		}
		OR = {
			vegetation = grasslands
			vegetation = sparse 
		}
		num_pop_type:tribesmen > 1
		num_pop_type:tribesmen < 5


	}

	possible_production_methods = {
		steppe_tribal_pasture_grazing
		steppe_tribal_livestock_production
	}

	construction_demand = estate_construct_building

	modifier = {

		local_population_capacity = -2.0 ### a bit more damaging to pop capacity than its cousin. but a bit more productive in horse breeding.
		# The tribal levy: the herd IS the cavalry pool.
		local_tribes_estate_power = 0.3
		local_levy_size_modifier = 0.02
	}

	graphical_tags = { small_military }
}

# =====================================================================
# REGISTERED PASTURE — the qopchur register made architectural
# (Dissolve program step 3). The il-rah routes written down, the
# flocks counted, the pasture a taxed tenancy instead of a right.
# Upgrade of a standing tribal_pasture: fewer horses come off the
# range, but the land under it is governable. The tribes know what a
# register is FOR — each one stings on the day it opens.
# Step 3 completes when EVERY rural location with >20,000 tribesmen
# carries one (detection in 02_matter_effects.txt).
# =====================================================================
registered_pasture = {
	is_foreign = no
	max_levels = 1
	pop_type = tribesmen
	employment_size = 1.0

	category = rgo_building_category

	rural_settlement = yes
	town = no
	city = no
	megalopolis = no

	build_time = estate_build_time
	destroyable_building = yes

	country_potential = {
		government_type = government_type:persianate_sultanate
		OR = {
			has_variable = matter_tork_iranization_underway
			has_variable = matter_tork_path_dissolve_done
		}
	}

	location_potential = {
		has_building = building_type:tribal_pasture
	}

	# A program step like the privileges: the estates must consent.
	allow = {
		matter_tork_estates_content = yes
	}

	unique_production_methods = {
		registered_grazing = {
			produced = horses
			output = 0.5

			category = building_maintenance
		}
	}

	modifier = {
		local_max_rural_control = 0.1
		local_tribes_estate_power = -0.5
		local_manpower = 0.005
	}

	on_built = {
		owner = {
			add_estate_satisfaction = { type = estate_type:tribes_estate value = -0.05 }
			# The register is costly work the hosts SEE: a year back.
			matter_tork_grant_time = yes
		}
	}

	construction_demand = basic_construction_needs
}

# =====================================================================
# CANTONMENT OF THE HOSTS — the Dissolve reward building. The tribal
# levy is dead (matter_hosts_of_the_state kills it); this is what
# replaces it: the registered hosts trained into state manpower, on
# state pay, under state officers.
# =====================================================================
hosts_cantonment = {
	is_foreign = no
	max_levels = 2
	pop_type = tribesmen
	employment_size = 2.0

	category = military_category

	rural_settlement = yes
	town = yes
	city = no
	megalopolis = no

	build_time = estate_build_time
	destroyable_building = yes

	country_potential = {
		government_type = government_type:persianate_sultanate
		has_variable = matter_tork_path_dissolve_done
	}

	location_potential = {
		num_pop_type:tribesmen > 5.0
	}

	modifier = {
		local_manpower = 0.02
		local_max_rural_control = 0.05
	}

	construction_demand = basic_construction_needs
}
