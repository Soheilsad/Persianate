desert_clay_pit = {
	is_foreign = no
	max_levels = rural_building_cap
	pop_type = laborers
	
	employment_size = rural_peasant_produce_employment
	
	rural_settlement = yes
	town = yes
	city = yes
	
	location_potential = {
		NOT = { raw_material = goods:clay }
		OR = {
			vegetation = desert
			vegetation = sparse
			vegetation = qanat_farmland
			vegetation = crumbling_qanat
		}
	}

	category = rgo_building_category
	
	# Economics mirror the base quarry (common_buildings.txt:295): the
	# tooled method runs a ~+0.4 margin at equilibrium prices (0.107
	# tools = 0.32 cost against 1.5 clay = 0.75), the hand-dug method
	# is the no-input fallback. The old single method (0.2 tools for
	# 1.0 clay) COST 0.6 to earn 0.5 - a building that only lived
	# while the clay crisis lived, then shed its workers and restarted
	# the crisis.
	unique_production_methods = {
		tooled_clay_pit = {
			tools = 0.107

			produced = clay
			output = 1.5

			category = building_maintenance
		}
		hand_dug_clay_pit = {
			produced = clay
			output = 0.5

			category = building_maintenance
		}
	}
	
	build_time = rural_build_time
}

REPLACE:caravanserai = {
	is_foreign = no
	pop_type = laborers
	
	max_levels = 5
	employment_size = trade_employment
	
	category = trade_category
	
	rural_settlement = yes
	town = yes
	city = yes
	
	
	
	country_potential = {
		OR = {
			has_advance = caravanserai_advance		# just to hide from those who do not get the tech.
			has_advance = persian_caravanserai_advance
		}
	}
	
	allow = {
		OR = {
			region = region:khorasan_region
			region = region:xinjiang_region
			region = region:mongolia_region
			region = region:steppes_region
			region = region:persia_region
		}
	}
	
	can_destroy = {
		NOT = { owner = { has_advance = caravanserai_advance } }
		NOT = { owner = { has_advance = persian_caravanserai_advance } }
	}
	
	build_time = trade_post_build_time
	
	modifier = {
		local_merchant_power = 0.5
		merchant_capacity_from_building = 1
		trade_range = 0.5
	}
	
	construction_demand = caravanserai_building_construction
	unique_production_methods = {
		caravanserai_maintenance = {
			jewelry = 0.01
			cloth = 0.01

			category = building_maintenance
		}
	}
}
