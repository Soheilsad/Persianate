# Erānzamīn — the operative IO for the post-Ilkhanid succession contest.
#
# Replaces the base ilkhanate IO as the real driver. Design:
#   -Flat federation: no leader, no engine-driven claimant promotion.
#   -Membership confers the "contender" passive (lower AE impact,
#     cheaper integration, modest army morale, increased unrest from
#     the contested legitimacy).
#   -Members cannot leave (can_leave_trigger = always = no below). The
#    design plan §5 still lists "abandoning claim" as an exit; that has
#    not been built — a member that kneels stays on the roll and is
#    filed as a sworn client instead.
#   -The Erānzamīn succession *situation* owns lifecycle and victory;
#     this IO auto-disbands when the situation flips its resolved flag.
#
# Starting members are seeded in
# main_menu/setup/1337/15_international_organizations.txt (the same-name
# replacement of the base file). The roster mirrors the base ilkhanate
# setup plus Sarbedars, minus a few tags that don't belong in the
# Persianate-succession frame.
#
# Panel: in_game/gui/panels/organization/eranzamin.gui. Its Overview
# cards, the "Notables" tab and the header read the ledger rebuilt by
# monthly_effect below (scripted_effects/12_eranzamin_io_effects.txt);
# the Members tab groups by the special statuses in
# international_organization_special_statuses/eranzamin.txt.

eranzamin = {

	has_target = no
	unique = yes
	expel_members_who_are_targets_of_other_members = no
	disband_if_no_leader = no
	# Flat federation: no leader country, no claimant promotion.
	has_leader_country = no

	# Diplomatic map: members tinted; the role statuses (khan-maker,
	# Chinggisid, sovereign, sworn client) override with their own
	# map_color. Colors in main_menu/common/named_colors/01_mod_colors.txt.
	show_on_diplomatic_map = yes
	member_color = map_eranzamin_member
	leader_color = map_eranzamin_member

	only_leader_country_joins_defensive_wars = no
	gives_military_access_to_all_when_at_war = no

	can_declare_war = {
		always = yes
	}

	# The "I am pursuing the throne of Erānzamīn" passive, applied to
	# every member. Eases conquest within the contested region but the
	# realm is volatile.
	modifier = {
		global_war_score_efficiency = 0.33
		antagonism_received_modifier = -0.50
		# diplomatic_annexation_cost does not exist in 1.4; the efficiency line below is the same lever.
		diplomatic_annexation_efficiency = 0.25
		pop_join_rebel_threshold = 0.1
		land_morale_modifier = 0.10
	}

	# Player-creation is gated off — the IO is seeded at game start.
	create_visible_trigger = {
		always = no
	}

	# Members in the Erānzamīn region can extend invitations to their
	# fellow regional powers. Gated on capital, not on owning a province,
	# because the latter let Mamluk-style outsiders qualify off a single
	# foothold in Anatolia or the Crescent.
	invite_visible_trigger = {
		scope:target ?= {
			capital = { is_eranzamin_region_province = yes }
			# Not the tümens of the Chagatai ulus (see can_join_trigger).
			NOT = { is_ulus_chaghatay_member = yes }
			NOT = { is_ulus_chaghatay_roster_tag = yes }
		}
		
	}

	# Organic joins: tier >= duchy, capital inside the region, sovereign.
	# Capital-based — not any_owned_location — so Mamluks (Cairo) and
	# similar adjacent powers can't slip in by holding a border province.
	can_join_trigger = {
		country_rank_level > 1

		capital = { is_eranzamin_region_province = yes }
		
		is_subject = no

		# The amir domains of Ulus Chaghatay sit in Transoxiana and
		# Tukharistan, inside khorasan_region, and since they stopped
		# being CHG's vassals they would pass every test above. They
		# answer a living Chagatayid khan; the Ilkhan's empty throne is
		# not theirs to contend for while they do.
		# (scripted_triggers/10_ulus_chaghatay_triggers.txt)
		NOT = { is_ulus_chaghatay_member = yes }
		NOT = { is_ulus_chaghatay_roster_tag = yes }
	}

	# Members cannot voluntarily abandon the claim.
	can_leave_trigger = {
		always = no
	}
	
	auto_leave_trigger = {
		always = no
	}

	# Disband when the succession situation resolves (path-victory or
	# timeout). The situation sets eranzamin_succession_resolved in
	# its on_ended.
	auto_disband_trigger = {
		has_global_variable = eranzamin_succession_resolved
	}

	on_joined = {
	}

	on_left = {
	}

	# The situation still owns lifecycle and victory. This cadence only
	# rebuilds the notables ledger the IO panel reads (heads, blocs,
	# power shares, khans in custody, key-city holders, header counts).
	monthly_effect = {
		eranzamin_rebuild_notables = yes
	}

	# Header readouts. Written by eranzamin_rebuild_notables — no
	# monthly_change of their own (the middle_kingdom
	# num_of_celestial_governors pattern).
	variables = {
		eranzamin_notables = {
			format = "ERANZAMIN_COUNT_FORMAT"
			min = 0
			max = 12
			start = 0
			monthly_change_hidden = yes
		}
		eranzamin_sovereigns = {
			format = "ERANZAMIN_COUNT_FORMAT"
			min = 0
			max = 60
			start = 0
			monthly_change_hidden = yes
		}
		eranzamin_leader_months = {
			format = "ERANZAMIN_MONTHS_FORMAT"
			min = 0
			max = 24
			start = 0
			monthly_change_hidden = yes
		}
	}

	special_statuses_implemented = {
		eranzamin_khan_maker
		eranzamin_chinggisid_claimant
		eranzamin_sovereign_shah
		eranzamin_sworn_client
	}

	ai_desire_to_join = {
		add = {
			desc = "PERSIANATE_LEGITIMACY"
			value = 50
		}
	}

	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 20
		}
	}
}
