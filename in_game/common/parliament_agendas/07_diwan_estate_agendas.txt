pa_diwan_case_embezzlement = {
	type = country
	estate = diwan_estate
	importance = 2.5
	potential = {
		country_has_estate = estate_type:diwan_estate
	}
	on_accept = {
		add_estate_satisfaction = { type = estate_type:diwan_estate value = -0.20 }
		add_estate_satisfaction = { type = estate_type:amirs_estate value = { add = { value = "estate_power(estate_type:amirs_estate)" multiply = 0.2 } } }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = { add = { value = "estate_power(estate_type:clergy_estate)" multiply = 0.2 } } }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = { add = { value = "estate_power(estate_type:burghers_estate)" multiply = 0.2 } } }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = { add = { value = "estate_power(estate_type:peasants_estate)" multiply = 0.2 } } }

		# Recover the embezzled hoard: a large share of the diwan's accumulated
		# coffers, less whatever they have sheltered in waqf. The crown takes all.
		# Inlined seizure (kept inline so the tooltip can preview the amount — a
		# scripted_effect call resolves only after the tooltip has rendered).
		"estate(estate_type:diwan_estate)" = { save_scope_as = seize_target }
		# LOCAL variable: scopes to this effect execution, not the
		# country entity — no residue, no cross-effect collisions.
		# (diwan_waqf_shelter stays a durable country variable: it is
		# set by the waqf-shelter mechanic and must persist.)
		set_local_variable = {
			name = seized_pool
			value = {
				value = scope:seize_target.estate_gold
				multiply = 0.40
			}
		}
		if = {
			limit = { has_variable = diwan_waqf_shelter }
			change_local_variable = {
				name = seized_pool
				multiply = {
					value = 1
					subtract = var:diwan_waqf_shelter
				}
			}
		}
		add_gold_to_estate = {
			estate_type = estate_type:diwan_estate
			value = {
				value = local_var:seized_pool
				multiply = -1
			}
		}
		add_gold = { value = local_var:seized_pool }

		add_country_modifier = { modifier = diwan_purge_disruption years = 3 }
		add_country_modifier = { modifier = diwan_reform_tax_efficiency years = 10 }
	}
	can_bribe = { can_bribe_default_trigger = yes }
	on_bribe = { on_bribe_default_effect = yes }
	ai_will_do = { base = 100 }
	chance = 10
}
