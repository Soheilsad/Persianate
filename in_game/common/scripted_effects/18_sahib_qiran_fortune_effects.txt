# =====================================================================
# THE ṢĀḤIB-QIRĀN'S FORTUNE (dawlat).
#
# The Amirs' Regency ends when a regent has covered for the blood he
# lacks with five years of victories (the legitimacy gap,
# scripted_effects/15_amirs_regency_effects.txt). The regency's state
# is then cleared — but the condition it measured is not. Temür after
# 1370 was still not a khan. What held the amirs of the ulus to him was
# his dawlat: the fortune a man is seen to have while he keeps winning
# and keeps giving. He campaigned almost every year for thirty-five
# years, and emptied his treasury on his followers after each one. And
# the fortune was his, not his house's: at his death in 1405 his sons
# and grandsons fought for four years over a realm none of them had
# made.
#
# So the gap is carried into the reign as sq_fortune, 0..100, on the
# Ṣāḥib-qirān's COUNTRY (TIM, when the conquest ambition releases it):
#
#   seeded     from the regency's gap at the ending (sq_fortune_seed,
#              called by ulus_end_sahib_qiran)
#   drains     every month; half as fast at war, twice as fast after
#              three years of peace (sq_fortune_tick, from
#              on_action/33_sahib_qiran_hooks.txt)
#   filled by  great battles, sieges, wars won, and largesse
#              (generic_actions/sahib_qiran_actions.txt)
#   bands      high: sq_fortune_high_mod; low: sq_fortune_low_mod
#              (main_menu/common/static_modifiers/
#              17_sahib_qiran_fortune_modifiers.txt)
#   empty      for sq_empty_years running: the amirs who swore to him at
#              the ending (sq_sworn_amir) ride away (sahib_qiran.2)
#   his death  the heir starts at sq_fortune_heir (sahib_qiran.1)
#
# Variables on the country:
#   sq_fortune          0..100
#   sq_fortune_bar      0..1, for the panel
#   sq_band_high / sq_band_low   flags, for the panel (the GUI tests
#                       IsSet; it has no verified numeric comparison)
#   sq_man              the ruler the fortune belongs to
#   sq_peace_months     months since his last war
#   sq_empty_months     running count of months at 0
# On each amir who became his vassal at the ending: sq_sworn_amir.
#
# Numbers: script_values/05_amirs_regency_values.txt, section 9.
# =====================================================================

# COUNTRY scope: the Ṣāḥib-qirān's realm, at the ending. Reads
# local_var:sq_seed if ulus_end_sahib_qiran saved the regency's gap
# there before clearing it; otherwise starts at the high line.
sq_fortune_seed = {
	if = {
		limit = { has_local_variable = sq_seed }
		set_variable = { name = sq_fortune value = local_var:sq_seed }
	}
	else = {
		set_variable = { name = sq_fortune value = sq_fortune_high_line }
	}
	set_variable = { name = sq_peace_months value = 0 }
	set_variable = { name = sq_empty_months value = 0 }
	set_variable = { name = sq_ambitions_count value = 0 }
	# The new realm starts with no accepted or tolerated cultures at all;
	# this is what he has to spend on them.
	add_political_influence = 50
	# The seat his rule began from, wherever it was: the "Conjunction
	# Holds" outcome asks that he still own it.
	if = {
		limit = { exists = capital }
		set_variable = { name = sq_seat value = capital }
	}
	if = {
		limit = { has_ruler = yes }
		ruler = { save_scope_as = sq_man_scope }
		set_variable = { name = sq_man value = scope:sq_man_scope }
	}
	sq_fortune_refresh = yes
}

# COUNTRY scope (the Ṣāḥib-qirān's realm), monthly from sq_fortune_tick.
# AN AMBITION MADE GOOD. The situation holds the region he has declared
# his conquest ambition (base: rot_conquest_ambition, changed by the
# Declare Conquest Ambition action). When he and his subjects own
# sq_ambition_share of its locations, it is counted once (list
# sq_ambitions_done, counter sq_ambitions_count), his fortune rises by
# sq_fortune_ambition, and he is told. Any region: Iran, Hindustan,
# the Qipchaq steppe, Khitai. Two of them make the Conjunction hold
# (scripted_triggers/13_sahib_qiran_triggers.txt).
sq_ambition_check = {
	if = {
		limit = {
			is_situation_active = situation:rise_of_timur
			situation:rise_of_timur = { has_variable = rot_conquest_ambition }
		}
		save_scope_as = sq_realm
		situation:rise_of_timur.var:rot_conquest_ambition = {
			save_scope_as = sq_ambition_region
		}
		if = {
			limit = {
				NOT = {
					is_target_in_variable_list = {
						name = sq_ambitions_done
						target = scope:sq_ambition_region
					}
				}
				scope:sq_ambition_region = {
					any_location_in_region = {
						percent >= sq_ambition_share
						owner ?= {
							OR = {
								this = scope:sq_realm
								is_subject_of = scope:sq_realm
							}
						}
					}
				}
			}
			add_to_variable_list = { name = sq_ambitions_done target = scope:sq_ambition_region }
			change_variable = { name = sq_ambitions_count add = 1 }
			sq_change_fortune = { AMOUNT = sq_fortune_ambition }
			trigger_event_non_silently = sahib_qiran.4
		}
	}
}

# COUNTRY scope. $AMOUNT$ a number or a script value; clamped 0..100.
sq_change_fortune = {
	if = {
		limit = { has_variable = sq_fortune }
		change_variable = { name = sq_fortune add = $AMOUNT$ }
		if = {
			limit = { var:sq_fortune < 0 }
			set_variable = { name = sq_fortune value = 0 }
		}
		if = {
			limit = { var:sq_fortune > 100 }
			set_variable = { name = sq_fortune value = 100 }
		}
		sq_fortune_refresh = yes
	}
}

# COUNTRY scope: the bar, the band flags and the band's modifier.
sq_fortune_refresh = {
	set_variable = {
		name = sq_fortune_bar
		value = {
			value = var:sq_fortune
			divide = 100
		}
	}
	if = {
		limit = { NOT = { has_country_modifier = sq_name_goes_before_him_mod } }
		add_country_modifier = {
			modifier = sq_name_goes_before_him_mod
			years = -1
		}
	}
	remove_variable = sq_band_high
	remove_variable = sq_band_low
	if = {
		limit = { var:sq_fortune >= sq_fortune_high_line }
		set_variable = sq_band_high
		if = {
			limit = { has_country_modifier = sq_fortune_low_mod }
			remove_country_modifier = sq_fortune_low_mod
		}
		if = {
			limit = { NOT = { has_country_modifier = sq_fortune_high_mod } }
			add_country_modifier = {
				modifier = sq_fortune_high_mod
				years = -1
			}
		}
	}
	else_if = {
		limit = { var:sq_fortune < sq_fortune_low_line }
		set_variable = sq_band_low
		if = {
			limit = { has_country_modifier = sq_fortune_high_mod }
			remove_country_modifier = sq_fortune_high_mod
		}
		if = {
			limit = { NOT = { has_country_modifier = sq_fortune_low_mod } }
			add_country_modifier = {
				modifier = sq_fortune_low_mod
				years = -1
			}
		}
	}
	else = {
		if = {
			limit = { has_country_modifier = sq_fortune_high_mod }
			remove_country_modifier = sq_fortune_high_mod
		}
		if = {
			limit = { has_country_modifier = sq_fortune_low_mod }
			remove_country_modifier = sq_fortune_low_mod
		}
	}
}

# COUNTRY scope, monthly.
sq_fortune_tick = {
	if = {
		limit = { at_war = yes }
		set_variable = { name = sq_peace_months value = 0 }
	}
	else = {
		change_variable = { name = sq_peace_months add = 1 }
	}

	set_local_variable = {
		name = sq_drain
		value = {
			value = sq_fortune_drain
			multiply = -1
		}
	}
	if = {
		limit = { at_war = yes }
		change_local_variable = { name = sq_drain multiply = sq_fortune_war_factor }
	}
	else_if = {
		limit = { var:sq_peace_months > sq_fortune_idle_months }
		change_local_variable = { name = sq_drain multiply = sq_fortune_idle_factor }
	}
	sq_change_fortune = { AMOUNT = local_var:sq_drain }

	if = {
		limit = { var:sq_fortune <= 0 }
		change_variable = { name = sq_empty_months add = 1 }
	}
	else = {
		set_variable = { name = sq_empty_months value = 0 }
	}

	# The reckoning. sq_reckoning_called (two months) keeps it from
	# being called again before the event has run.
	sq_ambition_check = yes

	# Under an heir (sq_heir_reign, set at every accession after the
	# founder's): months running in the low band, for the "Sons Divide"
	# outcome of the situation.
	if = {
		limit = {
			has_variable = sq_heir_reign
			has_variable = sq_band_low
		}
		change_variable = { name = sq_heir_low_months add = 1 }
	}
	else_if = {
		limit = { has_variable = sq_heir_reign }
		set_variable = { name = sq_heir_low_months value = 0 }
	}
	if = {
		limit = {
			var:sq_empty_months >= sq_empty_months_needed
			NOT = { has_variable = sq_reckoning_called }
		}
		set_variable = { name = sq_reckoning_called value = 1 months = 2 }
		trigger_event_non_silently = { id = sahib_qiran.2 days = 1 }
	}
}

# COUNTRY scope: the amirs who swore at the ending go their own way.
sq_fortune_reckoning = {
	every_subject = {
		limit = { has_variable = sq_sworn_amir }
		add_to_list = sq_leaving_amirs
	}
	every_in_list = {
		list = sq_leaving_amirs
		remove_variable = sq_sworn_amir
		# base's form (overlord = { cancel_subject = prev })
		overlord ?= { cancel_subject = prev }
	}
	# Read by the situation's "Amirs Ride Away" outcome.
	set_variable = sq_reckoned
	add_prestige = sq_reckoning_prestige
	if = {
		limit = { government_type = government_type:steppe_horde }
		add_horde_unity = sq_reckoning_unity
	}
	set_variable = { name = sq_empty_months value = 0 }
	set_variable = { name = sq_fortune value = sq_fortune_after_reckoning }
	sq_fortune_refresh = yes
}

# COUNTRY scope: a new man on the seat. The fortune was the dead man's.
sq_fortune_succession = {
	if = {
		limit = { has_ruler = yes }
		ruler = { save_scope_as = sq_man_scope }
		set_variable = { name = sq_man value = scope:sq_man_scope }
	}
	set_variable = { name = sq_fortune value = sq_fortune_heir }
	# After the Conjunction has held (sahib_qiran.3) the house has a name
	# of its own: the heir starts in the middle band, not at the bottom.
	if = {
		limit = { has_variable = sq_conjunction_held }
		set_variable = { name = sq_fortune value = sq_fortune_heir_held }
	}
	set_variable = sq_heir_reign
	set_variable = { name = sq_heir_low_months value = 0 }
	set_variable = { name = sq_empty_months value = 0 }
	sq_fortune_refresh = yes
}

# ---------------------------------------------------------------------
# THE ARMY IS LOST. COUNTRY scope: the Ṣāḥib-qirān's realm, having lost
# a war to an outsider with his fortune already in the low band;
# scope:sq_victor is the victor. Once in a game (global sq_army_lost).
#
# The Chaghatay were an army with households. This is the army gone:
# the reckoning at once (the sworn amirs ride away), and what the
# Bulqaq leads to without waiting for a Bulqaq — he and the amirs who
# were his are marked as the broken ulus's orphans, every outside
# neighbour gets the claim on them (ulus_bulqaq_open_the_pastures,
# scripted_effects/20_chaghatay_breaking_effects.txt, reads the list
# ulus_orphans built here), and the prize and the dissolution of
# on_action/36_chaghatay_breaking_hooks.txt open (global
# ulus_bulqaq_done).
# ---------------------------------------------------------------------
sq_army_is_lost = {
	set_global_variable = { name = sq_army_lost value = 1 }
	add_to_list = ulus_orphans
	every_subject = {
		limit = { has_variable = sq_sworn_amir }
		add_to_list = ulus_orphans
	}
	sq_fortune_reckoning = yes
	every_in_list = {
		list = ulus_orphans
		set_variable = ulus_bulqaq_orphan
	}
	set_global_variable = { name = ulus_bulqaq_done value = 1 }
	ulus_bulqaq_open_the_pastures = yes
	# The man who broke the army has the claim whether he borders them
	# or not.
	scope:sq_victor ?= {
		every_in_list = {
			list = ulus_orphans
			limit = {
				NOT = { this = scope:sq_victor }
				NOT = { is_subject_of = scope:sq_victor }
			}
			save_scope_as = sq_orphan
			scope:sq_victor = {
				add_casus_belli = {
					type = casus_belli:cb_break_the_tumens
					target = scope:sq_orphan
					years = 25
				}
			}
		}
		trigger_event_non_silently = sahib_qiran.6
	}
	trigger_event_non_silently = sahib_qiran.5
}

# ---------------------------------------------------------------------
# AN EMBASSY TO THE STRONG. COUNTRY scope: an AI Ṣāḥib-qirān;
# scope:sq_courted is the neighbour (sq_worth_courting). Gifts worth
# sq_court_gift_gold go to him. An AI neighbour takes them (mutual
# opinion), and if each then thinks well enough of the other, and the
# neighbour is not one to keep to himself, they ally. A PLAYER is asked
# (sahib_qiran.7) and nothing is done to him without his answer.
# ---------------------------------------------------------------------
sq_send_embassy = {
	set_variable = { name = sq_courted_recently value = 1 years = 5 }
	set_local_variable = { name = sq_gift value = sq_court_gift_gold }
	save_scope_as = sq_courting_realm
	if = {
		limit = { scope:sq_courted = { is_ai = no } }
		set_variable = { name = sq_gift_offered value = local_var:sq_gift years = 1 }
		scope:sq_courted = {
			trigger_event_non_silently = sahib_qiran.7
		}
	}
	else = {
		add_gold = {
			value = local_var:sq_gift
			multiply = -1
		}
		scope:sq_courted = {
			add_gold = local_var:sq_gift
		}
		add_opinion_mutual_effect = {
			modifier = opinion_sq_embassy_gifts
			target = scope:sq_courted
		}
		if = {
			limit = {
				opinion = {
					target = scope:sq_courted
					value >= sq_ai_court_alliance_opinion
				}
				scope:sq_courted = {
					opinion = {
						target = scope:sq_courting_realm
						value >= sq_ai_court_alliance_opinion
					}
					NOT = { ai_personality = ai_personality:ai_isolationist }
					NOT = { is_rival_of = scope:sq_courting_realm }
					at_war = no
				}
			}
			create_alliance = { target = scope:sq_courted }
		}
	}
}
