# Ulus Chaghatay — the monthly ledger behind the IO panel
# (in_game/gui/panels/organization/ulus_chaghatay.gui).
#
# Called with the IO as the CURRENT scope: from the IO's monthly_effect,
# and once from on_game_start (on_action/30_ulus_chaghatay_start.txt) so
# the panel is not empty before the first tick. Root is NOT assumed to
# be the IO — it is saved as scope:ulus_io (the eranzamin_rebuild_notables
# construction, scripted_effects/12_eranzamin_io_effects.txt).
#
# It DECIDES nothing. It records, so that the succession is never a
# surprise: the panel shows the same numbers the phase-2 poll will read.
#
#   GLOBAL LISTS (GetGlobalList in the GUI — the proven list path)
#     ulus_amirs        every junior member, strongest first
#     ulus_kingmakers   the three junior members with the highest
#                       kingmaker score, best first
#     ulus_greatest_amirs  the three junior members with the greatest
#                       WEIGHT, heaviest first (the three the amir bar
#                       is about; often not the same three)
#
#   ON EACH JUNIOR MEMBER (country)
#     ulus_kingmaker_score   script value of the same name
#     ulus_weight            0..100 across ALL members (khan included):
#                            50% military strength, 35% tax base, 15%
#                            prestige, each as a share of the ulus total
#     ulus_amir_share        0..1, its weight as a share of all the
#                            amirs' together
#
#   ON THE IO
#     ulus_amir_count   number of junior members            (header)
#     ulus_beg_power    nobles' estate power in the leader's realm,
#                       0..100                              (header)
#     ulus_beg_bar      the same against the 50% bar, 0..1
#     ulus_amir_ratio   the three greatest amirs' combined weight /
#                       the leader's own                    (header)
#     ulus_amir_bar     the same against the 2.0 bar, 0..1
#     ulus_khan_heir    the leader's heir (character), unset if none
#     ulus_khan_weight  the leader's own ulus_weight
#     ulus_khan_w_mil / _tax / _pre   its three parts (of 50, 35, 15)
#     ulus_top3_weight  the three greatest amirs' weights summed
#     ulus_crown_power  the crown's estate power in the leader's realm,
#                       0..100; ulus_crown_line = twice that
#     ulus_beg_line / ulus_amir_line   the two bars' thresholds today
#   ON THE IO, DURING A REGENCY (section 6)
#     ulus_held_khan_age    the puppet khan's age
#     ulus_heir_standing    the regent's heir's score / the best rival's
#     ulus_heir_bar         the same, capped at 1
#     ulus_heir_line        the share he needs (ulus_regent_heir_ratio)
#     ulus_gap_bar          the legitimacy gap's fill, 0..1
#     ulus_stage_is_1..4    which stage (one of the four is set)
#
# THE REGENCY BAR (design, "The amirs raise a khan"): the begs can raise
# a khan of their own when the nobles hold half the power in the khan's
# realm, or when his three greatest amirs together outweigh him twice
# over (NOT all amirs: their sum always exceeds the khan's). Both readings are kept; whichever fills first is the
# one to watch.
#
# estate_power is on a 0..1 scale (events/horde_succession.txt, verified
# there against base's rise_of_the_szlachta). military_strength is the
# country value base orders crusade and jihad candidates by.

ulus_chaghatay_rebuild_ledger = {
	save_temporary_scope_as = ulus_io

	# ---0. Wipe last month's ledger --------------------------------------
	every_in_global_list = {
		variable = ulus_amirs
		remove_variable = ulus_kingmaker_score
		remove_variable = ulus_amir_share
	}
	every_in_global_list = {
		variable = ulus_standing
		remove_variable = ulus_kingmaker_score
		remove_variable = ulus_seat_row
	}
	clear_global_variable_list = ulus_amirs
	clear_global_variable_list = ulus_kingmakers
	clear_global_variable_list = ulus_greatest_amirs
	clear_global_variable_list = ulus_standing
	remove_variable = ulus_heir_holds
	remove_variable = ulus_khan_heir

	set_local_variable = { name = ulus_n value = 0 }
	set_local_variable = { name = ulus_amir_str value = 0 }
	set_local_variable = { name = ulus_khan_str value = 0 }
	set_local_variable = { name = ulus_beg value = 0 }

	# ---1. The amirs' combined strength ----------------------------------
	# ---The weight of a realm ------------------------------------------
	# Not land. A count of locations makes a hundred empty pastures
	# outweigh Samarkand. A realm's weight in the ulus is what it can
	# put in the field, what it can tax, and the name it carries: each
	# taken as its share of the whole ulus's total (khan included), then
	#   40% military strength, 30% tax base, 20% the ruler's abilities,
	#   10% prestige (script_values/04_ulus_chaghatay_values.txt, ulus_w_*),
	# on a 0-100 scale across all members. Stored on every member as
	# ulus_weight; the bars, the amirs' shares and the kingmaker score
	# all read it.
	set_local_variable = { name = ulus_t_mil value = 0 }
	set_local_variable = { name = ulus_t_tax value = 0 }
	set_local_variable = { name = ulus_t_pre value = 0 }
	set_local_variable = { name = ulus_t_rul value = 0 }
	every_international_organization_member = {
		change_local_variable = { name = ulus_t_mil add = military_strength }
		change_local_variable = { name = ulus_t_tax add = country_tax_base }
		change_local_variable = { name = ulus_t_rul add = ulus_ruler_abilities }
		change_local_variable = {
			name = ulus_t_pre
			add = {
				value = prestige
				min = 0
			}
		}
	}
	every_international_organization_member = {
		set_variable = {
			name = ulus_weight
			value = {
				value = 0
				if = {
					limit = { local_var:ulus_t_mil > 0 }
					add = {
						value = military_strength
						divide = local_var:ulus_t_mil
						multiply = ulus_w_mil
					}
				}
				if = {
					limit = { local_var:ulus_t_tax > 0 }
					add = {
						value = country_tax_base
						divide = local_var:ulus_t_tax
						multiply = ulus_w_tax
					}
				}
				if = {
					limit = { local_var:ulus_t_rul > 0 }
					add = {
						value = ulus_ruler_abilities
						divide = local_var:ulus_t_rul
						multiply = ulus_w_rul
					}
				}
				if = {
					limit = { local_var:ulus_t_pre > 0 }
					add = {
						value = prestige
						min = 0
						divide = local_var:ulus_t_pre
						multiply = ulus_w_pre
					}
				}
			}
		}
	}

	every_international_organization_member = {
		limit = {
			NOT = { is_leader_of_international_organization = scope:ulus_io }
		}
		change_local_variable = { name = ulus_n add = 1 }
		change_local_variable = { name = ulus_amir_str add = var:ulus_weight }
	}

	# ---1b. Pledges of support (generic_actions/amirs_rising_actions.txt)
	# An amir who has pledged himself to another house (ulus_pledged_to on
	# the pledger) lends it ulus_pledge_share of his weight: it leaves his
	# own effective weight and joins the other's as ulus_backing. The
	# amirs' total is unchanged. Everything below that ranks or sums the
	# amirs reads the EFFECTIVE weight (script value ulus_realm_weight,
	# stored for the panel as ulus_weight_eff). A pledge to a house that
	# has left the ulus, or become its leader, lapses here.
	every_international_organization_member = {
		set_variable = { name = ulus_backing value = 0 }
	}
	every_international_organization_member = {
		limit = { has_variable = ulus_pledged_to }
		if = {
			limit = {
				NOT = { is_leader_of_international_organization = scope:ulus_io }
				var:ulus_pledged_to ?= {
					is_ulus_chaghatay_amir = yes
					NOT = { has_variable = ulus_pledged_to }
				}
			}
			set_local_variable = {
				name = ulus_lent
				value = {
					value = var:ulus_weight
					multiply = ulus_pledge_share
				}
			}
			var:ulus_pledged_to = {
				change_variable = { name = ulus_backing add = local_var:ulus_lent }
			}
		}
		else = {
			remove_variable = ulus_pledged_to
		}
	}
	# Pledges to the KHAN (ulus_khan_ask_pledge): the same loan, to the
	# seat. They lapse when a regency begins: the man they were given to
	# is no longer the one on the seat.
	every_international_organization_member = {
		limit = { has_variable = ulus_pledged_to_khan }
		if = {
			limit = {
				NOT = { is_leader_of_international_organization = scope:ulus_io }
				scope:ulus_io = { NOT = { has_variable = ulus_regency } }
			}
			set_local_variable = {
				name = ulus_lent
				value = {
					value = var:ulus_weight
					multiply = ulus_pledge_share
				}
			}
			scope:ulus_io.leader_country ?= {
				change_variable = { name = ulus_backing add = local_var:ulus_lent }
			}
		}
		else = {
			remove_variable = ulus_pledged_to_khan
		}
	}

	# ---1c. Friends outside the ulus --------------------------------------
	# (script_values/04_ulus_chaghatay_values.txt, FRIENDS OUTSIDE.) An
	# outsider's weight is reckoned on the ulus's own scale from his host
	# and his tax base; allies count for more than a mere backer, and a
	# backer who is also an ally is counted once, as an ally.
	every_international_organization_member = {
		set_local_variable = { name = ulus_out value = 0 }
		save_scope_as = ulus_ledger_member
		every_ally = {
			limit = {
				NOT = { is_member_of_international_organization = scope:ulus_io }
			}
			set_local_variable = { name = ulus_friend value = 0 }
			if = {
				limit = { local_var:ulus_t_mil > 0 }
				change_local_variable = {
					name = ulus_friend
					add = {
						value = military_strength
						divide = local_var:ulus_t_mil
						multiply = ulus_w_mil
					}
				}
			}
			if = {
				limit = { local_var:ulus_t_tax > 0 }
				change_local_variable = {
					name = ulus_friend
					add = {
						value = country_tax_base
						divide = local_var:ulus_t_tax
						multiply = ulus_w_tax
					}
				}
			}
			if = {
				limit = { local_var:ulus_friend > ulus_outside_friend_cap }
				set_local_variable = { name = ulus_friend value = ulus_outside_friend_cap }
			}
			change_local_variable = {
				name = ulus_out
				add = {
					value = local_var:ulus_friend
					multiply = ulus_ally_fraction
				}
			}
		}
		if = {
			limit = { has_variable = ulus_outside_backer }
			if = {
				limit = {
					var:ulus_outside_backer ?= {
						NOT = { is_member_of_international_organization = scope:ulus_io }
						var:ulus_backs_amir ?= scope:ulus_ledger_member
					}
				}
				var:ulus_outside_backer = {
					if = {
						limit = {
							NOT = { is_allied_with = { target = scope:ulus_ledger_member } }
						}
						set_local_variable = { name = ulus_friend value = 0 }
						if = {
							limit = { local_var:ulus_t_mil > 0 }
							change_local_variable = {
								name = ulus_friend
								add = {
									value = military_strength
									divide = local_var:ulus_t_mil
									multiply = ulus_w_mil
								}
							}
						}
						if = {
							limit = { local_var:ulus_t_tax > 0 }
							change_local_variable = {
								name = ulus_friend
								add = {
									value = country_tax_base
									divide = local_var:ulus_t_tax
									multiply = ulus_w_tax
								}
							}
						}
						if = {
							limit = { local_var:ulus_friend > ulus_outside_friend_cap }
							set_local_variable = { name = ulus_friend value = ulus_outside_friend_cap }
						}
						change_local_variable = {
							name = ulus_out
							add = {
								value = local_var:ulus_friend
								multiply = ulus_backer_fraction
							}
						}
					}
				}
			}
			else = {
				# The backer is gone, has joined the ulus, or backs another.
				remove_variable = ulus_outside_backer
			}
		}
		if = {
			limit = { local_var:ulus_out > ulus_outside_cap }
			set_local_variable = { name = ulus_out value = ulus_outside_cap }
		}
		set_variable = { name = ulus_outside_weight value = local_var:ulus_out }
	}

	# ---1d. Blocs ----------------------------------------------------------
	# A house's bloc is the house it has pledged to, or itself if others
	# have pledged to it (ulus_bloc_head; unset for a house that stands
	# alone). Read by ulus_press_the_feud.
	every_international_organization_member = {
		remove_variable = ulus_bloc_head
		if = {
			limit = { has_variable = ulus_pledged_to }
			set_variable = { name = ulus_bloc_head value = var:ulus_pledged_to }
		}
		else_if = {
			limit = {
				NOT = { is_leader_of_international_organization = scope:ulus_io }
				var:ulus_backing > 0
			}
			set_variable = { name = ulus_bloc_head value = this }
		}
	}

	# For the panel's Pledges card: every house that has pledged its tümen
	# (to an amir or to the khan) or has a patron outside the ulus.
	clear_global_variable_list = ulus_pledgers
	every_international_organization_member = {
		limit = {
			OR = {
				has_variable = ulus_pledged_to
				has_variable = ulus_pledged_to_khan
				has_variable = ulus_outside_backer
			}
		}
		add_to_global_variable_list = { name = ulus_pledgers target = this }
	}

	every_international_organization_member = {
		set_variable = { name = ulus_weight_eff value = ulus_realm_weight }
	}

	# ---2. The amirs, strongest first ------------------------------------
	ordered_international_organization_member = {
		limit = {
			NOT = { is_leader_of_international_organization = scope:ulus_io }
		}
		order_by = ulus_realm_weight
		max = 60
		check_range_bounds = no

		# Share first: the kingmaker score reads it.
		if = {
			limit = { local_var:ulus_amir_str > 0 }
			set_variable = {
				name = ulus_amir_share
				value = {
					value = ulus_realm_weight
					divide = local_var:ulus_amir_str
				}
			}
		}
		else = {
			set_variable = { name = ulus_amir_share value = 0 }
		}
		set_variable = { name = ulus_kingmaker_score value = ulus_kingmaker_score }
		add_to_global_variable_list = { name = ulus_amirs target = this }
	}

	# ---3. If the khan died today: the three likeliest kingmakers --------
	ordered_international_organization_member = {
		limit = {
			NOT = { is_leader_of_international_organization = scope:ulus_io }
		}
		order_by = ulus_kingmaker_score
		max = 3
		check_range_bounds = no
		add_to_global_variable_list = { name = ulus_kingmakers target = this }
		# The best amir's score, for ulus_pledge_acceptance (a house's own
		# hopes of the khan are measured against it).
		if = {
			limit = { NOT = { has_local_variable = ulus_top_set } }
			set_local_variable = { name = ulus_top_set value = 1 }
			set_local_variable = { name = ulus_top_score value = var:ulus_kingmaker_score }
		}
	}
	if = {
		limit = { has_local_variable = ulus_top_score }
		set_variable = { name = ulus_top_kingmaker_score value = local_var:ulus_top_score }
	}

	# ---4. The seat: its own strength, its begs, its heir ----------------
	# The three greatest amirs together. The khan never outweighs ALL
	# his amirs: at his best he is worth three or four of them, and with
	# a court at Almaliq he may not even be the largest. So the reading
	# that means something is the khan against his three GREATEST amirs:
	# near 1.0 he is their match, at 2.0 they outweigh him twice over.
	# The same three are listed BY WEIGHT under the bar
	# (ulus_greatest_amirs), so that they are not mistaken for the three
	# likeliest kingmakers, who are ranked by score and need not be the
	# same houses.
	set_local_variable = { name = ulus_top3 value = 0 }
	ordered_international_organization_member = {
		limit = {
			NOT = { is_leader_of_international_organization = scope:ulus_io }
		}
		order_by = ulus_realm_weight
		max = 3
		check_range_bounds = no
		change_local_variable = { name = ulus_top3 add = ulus_realm_weight }
		add_to_global_variable_list = { name = ulus_greatest_amirs target = this }
	}

	set_local_variable = { name = ulus_k_mil value = 0 }
	set_local_variable = { name = ulus_k_tax value = 0 }
	set_local_variable = { name = ulus_k_pre value = 0 }
	set_local_variable = { name = ulus_k_rul value = 0 }
	set_local_variable = { name = ulus_crown value = 0 }
	leader_country ?= {
		set_local_variable = { name = ulus_khan_str value = ulus_realm_weight }	# effective: pledged lords, friends outside
		# The seat's weight taken apart, by the same three formulas that
		# built ulus_weight above (the ulus totals are still in hand):
		# the military part out of 50, the tax part out of 35, the
		# prestige part out of 15. For the panel, so the khan's number
		# can be checked against the game's own figures.
		if = {
			limit = { local_var:ulus_t_mil > 0 }
			set_local_variable = {
				name = ulus_k_mil
				value = {
					value = military_strength
					divide = local_var:ulus_t_mil
					multiply = ulus_w_mil
				}
			}
		}
		if = {
			limit = { local_var:ulus_t_tax > 0 }
			set_local_variable = {
				name = ulus_k_tax
				value = {
					value = country_tax_base
					divide = local_var:ulus_t_tax
					multiply = ulus_w_tax
				}
			}
		}
		if = {
			limit = { local_var:ulus_t_pre > 0 }
			set_local_variable = {
				name = ulus_k_pre
				value = {
					value = prestige
					min = 0
					divide = local_var:ulus_t_pre
					multiply = ulus_w_pre
				}
			}
		}
		if = {
			limit = { local_var:ulus_t_rul > 0 }
			set_local_variable = {
				name = ulus_k_rul
				value = {
					value = ulus_ruler_abilities
					divide = local_var:ulus_t_rul
					multiply = ulus_w_rul
				}
			}
		}
		# The crown's power in the seat's realm: the other half of the
		# begs' bar (begs >= the line AND >= twice the crown).
		set_local_variable = {
			name = ulus_crown
			value = "estate_power(estate_type:crown_estate)"
		}
		# The seat's own weight on the amirs' scale, so that
		# ulus_kingmaker_score reads the same for a regent (and his
		# heir) as for the amirs he is measured against. The seat is in
		# neither list above.
		if = {
			limit = { local_var:ulus_amir_str > 0 }
			set_variable = {
				name = ulus_amir_share
				value = {
					value = var:ulus_weight
					divide = local_var:ulus_amir_str
				}
			}
		}
		else = {
			set_variable = { name = ulus_amir_share value = 0 }
		}
		if = {
			limit = { country_has_estate = estate_type:nobles_estate }
			set_local_variable = {
				name = ulus_beg
				value = "estate_power(estate_type:nobles_estate)"
			}
		}
		if = {
			limit = { has_heir = yes }
			heir = {
				scope:ulus_io = {
					set_variable = { name = ulus_khan_heir value = prev }
				}
			}
		}
	}

	# ---4b. Standing in the ulus: the seat among the three kingmakers ----
	# The khan is scored by the amirs' own formula (his ulus_amir_share
	# was set just above), plus ulus_khan_line_bonus if he is of the line,
	# and ranked with the three likeliest kingmakers. The ranking is for
	# reading; what decides a succession is the bar (ulus_heir_holds, set
	# at the end of step 5).
	leader_country ?= {
		set_variable = {
			name = ulus_kingmaker_score
			value = {
				value = ulus_kingmaker_score
				if = {
					limit = { ulus_ruler_is_of_the_line = yes }
					add = ulus_khan_line_bonus
				}
			}
		}
		set_variable = ulus_seat_row
	}
	ordered_international_organization_member = {
		limit = {
			OR = {
				is_leader_of_international_organization = scope:ulus_io
				is_target_in_global_variable_list = {
					name = ulus_kingmakers
					target = this
				}
			}
		}
		order_by = ulus_standing_score
		max = 4
		check_range_bounds = no
		add_to_global_variable_list = { name = ulus_standing target = this }
	}

	# ---5. The regency bar ------------------------------------------------
	set_variable = { name = ulus_amir_count value = local_var:ulus_n }
	set_variable = { name = ulus_khan_weight value = local_var:ulus_khan_str }
	set_variable = { name = ulus_khan_w_mil value = local_var:ulus_k_mil }
	set_variable = { name = ulus_khan_w_tax value = local_var:ulus_k_tax }
	set_variable = { name = ulus_khan_w_pre value = local_var:ulus_k_pre }
	set_variable = { name = ulus_khan_w_rul value = local_var:ulus_k_rul }
	set_variable = { name = ulus_top3_weight value = local_var:ulus_top3 }
	set_variable = {
		name = ulus_crown_power
		value = {
			value = local_var:ulus_crown
			multiply = 100
		}
	}
	# What the begs must reach on the crown's account: twice its power
	# (ulus_regency_noble_crown_ratio), as a percentage.
	set_variable = {
		name = ulus_crown_line
		value = {
			value = local_var:ulus_crown
			multiply = ulus_regency_noble_crown_ratio
			multiply = 100
		}
	}
	set_variable = {
		name = ulus_beg_power
		value = {
			value = local_var:ulus_beg
			multiply = 100
		}
	}
	set_variable = {
		name = ulus_beg_bar
		value = {
			value = local_var:ulus_beg
			divide = ulus_regency_noble_power_now
			max = 1
		}
	}
	# The two lines themselves, for the panel's "x / line" texts: they
	# move (a third lower after a child khan), so they are not literals.
	set_variable = {
		name = ulus_beg_line
		value = {
			value = ulus_regency_noble_power_now
			multiply = 100
		}
	}
	set_variable = { name = ulus_amir_line value = ulus_regency_amir_ratio_now }
	if = {
		limit = { local_var:ulus_khan_str > 0 }
		set_variable = {
			name = ulus_amir_ratio
			value = {
				value = local_var:ulus_top3
				divide = local_var:ulus_khan_str
				max = 100
			}
		}
	}
	else = {
		# A khan with no host at all: the bar is met by anyone who has one.
		if = {
			limit = { local_var:ulus_top3 > 0 }
			set_variable = { name = ulus_amir_ratio value = 100 }
		}
		else = {
			set_variable = { name = ulus_amir_ratio value = 0 }
		}
	}
	set_variable = {
		name = ulus_amir_bar
		value = {
			value = var:ulus_amir_ratio
			divide = ulus_regency_amir_ratio_now
			max = 1
		}
	}

	# Months running with the bar met and no regency: how long the amirs
	# have been able to rise (the AI's patience, amirs_rising_actions).
	if = {
		limit = {
			NOT = { has_variable = ulus_regency }
			leader_country ?= { ulus_regency_bar_met = yes }
		}
		change_variable = { name = ulus_bar_met_months add = 1 }
	}
	else = {
		set_variable = { name = ulus_bar_met_months value = 0 }
	}

	# Would the khan's heir hold the seat today? (The panel's line under
	# the standing; ulus_regency_bar_met is the poll's own test.)
	if = {
		limit = {
			leader_country ?= {
				NOT = { ulus_regency_bar_met = yes }
			}
		}
		set_variable = ulus_heir_holds
	}

	# ---6. During a regency: what the Succession card shows instead -----
	# (the state itself is kept by scripted_effects/
	# 15_amirs_regency_effects.txt; this only derives display values.)
	remove_variable = ulus_stage_is_1
	remove_variable = ulus_stage_is_2
	remove_variable = ulus_stage_is_3
	remove_variable = ulus_stage_is_4
	remove_variable = ulus_held_khan_age
	if = {
		limit = { has_variable = ulus_regency }

		# The khan's age.
		if = {
			limit = {
				var:ulus_held_khan ?= { is_alive = yes }
			}
			set_local_variable = { name = ulus_k_age value = 0 }
			var:ulus_held_khan = {
				set_local_variable = { name = ulus_k_age value = age_in_years }
			}
			set_variable = { name = ulus_held_khan_age value = local_var:ulus_k_age }
		}

		# The regent's heir against the best rival amir: the number the
		# poll at the regent's death (amirs_regency.40) will compare with
		# ulus_regent_heir_ratio.
		set_local_variable = { name = ulus_h_score value = 0 }
		set_local_variable = { name = ulus_r_score value = 0 }
		leader_country ?= {
			set_local_variable = { name = ulus_h_score value = ulus_regent_heir_score }
		}
		ordered_international_organization_member = {
			limit = {
				NOT = { is_leader_of_international_organization = scope:ulus_io }
			}
			order_by = ulus_kingmaker_score
			max = 1
			check_range_bounds = no
			set_local_variable = { name = ulus_r_score value = ulus_kingmaker_score }
		}
		if = {
			limit = { local_var:ulus_r_score > 0 }
			set_variable = {
				name = ulus_heir_standing
				value = {
					value = local_var:ulus_h_score
					divide = local_var:ulus_r_score
					max = 9.99
				}
			}
		}
		else = {
			set_variable = { name = ulus_heir_standing value = 9.99 }
		}
		set_variable = {
			name = ulus_heir_bar
			value = {
				value = var:ulus_heir_standing
				max = 1
			}
		}
		set_variable = { name = ulus_heir_line value = ulus_regent_heir_ratio }

		# The gap as a 0..1 bar, and the stage as four flags (the GUI
		# tests IsSet; it has no verified numeric comparison).
		if = {
			limit = { has_variable = ulus_gap }
			set_variable = {
				name = ulus_gap_bar
				value = {
					value = var:ulus_gap
					divide = 100
				}
			}
		}
		if = {
			limit = { has_variable = ulus_regency_stage }
			if = {
				limit = { var:ulus_regency_stage >= 4 }
				set_variable = ulus_stage_is_4
			}
			else_if = {
				limit = { var:ulus_regency_stage >= 3 }
				set_variable = ulus_stage_is_3
			}
			else_if = {
				limit = { var:ulus_regency_stage >= 2 }
				set_variable = ulus_stage_is_2
			}
			else = {
				set_variable = ulus_stage_is_1
			}
		}

		# The same readings on the REGENT'S STATE, for the disaster's own
		# panel (gui/panels/disaster/amirs_regency.gui), which can only
		# read its owner. Rewritten every month; a state that has lost
		# the khan has lost the disaster with him, so stale copies on a
		# former regent are never shown.
		leader_country ?= {
			remove_variable = ulus_d_stage_1
			remove_variable = ulus_d_stage_2
			remove_variable = ulus_d_stage_3
			remove_variable = ulus_d_stage_4
			if = {
				limit = { scope:ulus_io = { has_variable = ulus_stage_is_4 } }
				set_variable = ulus_d_stage_4
			}
			else_if = {
				limit = { scope:ulus_io = { has_variable = ulus_stage_is_3 } }
				set_variable = ulus_d_stage_3
			}
			else_if = {
				limit = { scope:ulus_io = { has_variable = ulus_stage_is_2 } }
				set_variable = ulus_d_stage_2
			}
			else = {
				set_variable = ulus_d_stage_1
			}
			if = {
				limit = { scope:ulus_io = { has_variable = ulus_regency_stage } }
				set_variable = { name = ulus_d_stage value = scope:ulus_io.var:ulus_regency_stage }
			}
			if = {
				limit = { scope:ulus_io = { has_variable = ulus_regency_progress } }
				set_variable = { name = ulus_d_progress value = scope:ulus_io.var:ulus_regency_progress }
			}
			if = {
				limit = { scope:ulus_io = { has_variable = ulus_gap } }
				set_variable = { name = ulus_d_gap value = scope:ulus_io.var:ulus_gap }
			}
			set_variable = { name = ulus_d_heir value = scope:ulus_io.var:ulus_heir_bar }
		}
	}
}
