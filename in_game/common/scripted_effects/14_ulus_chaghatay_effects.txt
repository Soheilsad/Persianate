# Ulus Chaghatay — the monthly ledger behind the IO panel
# (in_game/gui/panels/organization/ulus_chaghatay.gui).
#
# Called with the IO as the CURRENT scope: from the IO's monthly_effect,
# and once from on_game_start (on_action/30_ulus_chaghatay_start.txt) so
# the panel is not empty before the first tick. Root is NOT assumed to
# be the IO — it is saved as scope:ulus_io (the eranzamin_rebuild_notables
# construction, scripted_effects/12_eranzamin_io_effects.txt).
#
# It DECIDES nothing. It records, so that the succession is never a
# surprise: the panel shows the same numbers the phase-2 poll will read.
#
#   GLOBAL LISTS (GetGlobalList in the GUI — the proven list path)
#     ulus_amirs        every junior member, strongest first
#     ulus_kingmakers   the three junior members with the highest
#                       kingmaker score, best first
#
#   ON EACH JUNIOR MEMBER (country)
#     ulus_kingmaker_score   script value of the same name
#     ulus_weight            0..100 across ALL members (khan included):
#                            50% military strength, 35% tax base, 15%
#                            prestige, each as a share of the ulus total
#     ulus_amir_share        0..1, its weight as a share of all the
#                            amirs' together
#
#   ON THE IO
#     ulus_amir_count   number of junior members            (header)
#     ulus_beg_power    nobles' estate power in the leader's realm,
#                       0..100                              (header)
#     ulus_beg_bar      the same against the 50% bar, 0..1
#     ulus_amir_ratio   the three greatest amirs' combined weight /
#                       the leader's own                    (header)
#     ulus_amir_bar     the same against the 2.0 bar, 0..1
#     ulus_khan_heir    the leader's heir (character), unset if none
#
# THE REGENCY BAR (design, "The amirs raise a khan"): the begs can raise
# a khan of their own when the nobles hold half the power in the khan's
# realm, or when his three greatest amirs together outweigh him twice
# over (NOT all amirs: their sum always exceeds the khan's). Both readings are kept; whichever fills first is the
# one to watch.
#
# estate_power is on a 0..1 scale (events/horde_succession.txt, verified
# there against base's rise_of_the_szlachta). military_strength is the
# country value base orders crusade and jihad candidates by.

ulus_chaghatay_rebuild_ledger = {
	save_temporary_scope_as = ulus_io

	# ---0. Wipe last month's ledger --------------------------------------
	every_in_global_list = {
		variable = ulus_amirs
		remove_variable = ulus_kingmaker_score
		remove_variable = ulus_amir_share
	}
	clear_global_variable_list = ulus_amirs
	clear_global_variable_list = ulus_kingmakers
	remove_variable = ulus_khan_heir

	set_local_variable = { name = ulus_n value = 0 }
	set_local_variable = { name = ulus_amir_str value = 0 }
	set_local_variable = { name = ulus_khan_str value = 0 }
	set_local_variable = { name = ulus_beg value = 0 }

	# ---1. The amirs' combined strength ----------------------------------
	# ---The weight of a realm ------------------------------------------
	# Not land. A count of locations makes a hundred empty pastures
	# outweigh Samarkand. A realm's weight in the ulus is what it can
	# put in the field, what it can tax, and the name it carries: each
	# taken as its share of the whole ulus's total (khan included), then
	#   50% military strength, 35% tax base, 15% prestige,
	# on a 0-100 scale across all members. Stored on every member as
	# ulus_weight; the bars, the amirs' shares and the kingmaker score
	# all read it.
	set_local_variable = { name = ulus_t_mil value = 0 }
	set_local_variable = { name = ulus_t_tax value = 0 }
	set_local_variable = { name = ulus_t_pre value = 0 }
	every_international_organization_member = {
		change_local_variable = { name = ulus_t_mil add = military_strength }
		change_local_variable = { name = ulus_t_tax add = country_tax_base }
		change_local_variable = {
			name = ulus_t_pre
			add = {
				value = prestige
				min = 0
			}
		}
	}
	every_international_organization_member = {
		set_variable = {
			name = ulus_weight
			value = {
				value = 0
				if = {
					limit = { local_var:ulus_t_mil > 0 }
					add = {
						value = military_strength
						divide = local_var:ulus_t_mil
						multiply = 50
					}
				}
				if = {
					limit = { local_var:ulus_t_tax > 0 }
					add = {
						value = country_tax_base
						divide = local_var:ulus_t_tax
						multiply = 35
					}
				}
				if = {
					limit = { local_var:ulus_t_pre > 0 }
					add = {
						value = prestige
						min = 0
						divide = local_var:ulus_t_pre
						multiply = 15
					}
				}
			}
		}
	}

	every_international_organization_member = {
		limit = {
			NOT = { is_leader_of_international_organization = scope:ulus_io }
		}
		change_local_variable = { name = ulus_n add = 1 }
		change_local_variable = { name = ulus_amir_str add = var:ulus_weight }
	}

	# ---2. The amirs, strongest first ------------------------------------
	ordered_international_organization_member = {
		limit = {
			NOT = { is_leader_of_international_organization = scope:ulus_io }
		}
		order_by = ulus_realm_weight
		max = 60
		check_range_bounds = no

		# Share first: the kingmaker score reads it.
		if = {
			limit = { local_var:ulus_amir_str > 0 }
			set_variable = {
				name = ulus_amir_share
				value = {
					value = var:ulus_weight
					divide = local_var:ulus_amir_str
				}
			}
		}
		else = {
			set_variable = { name = ulus_amir_share value = 0 }
		}
		set_variable = { name = ulus_kingmaker_score value = ulus_kingmaker_score }
		add_to_global_variable_list = { name = ulus_amirs target = this }
	}

	# ---3. If the khan died today: the three likeliest kingmakers --------
	ordered_international_organization_member = {
		limit = {
			NOT = { is_leader_of_international_organization = scope:ulus_io }
		}
		order_by = ulus_kingmaker_score
		max = 3
		check_range_bounds = no
		add_to_global_variable_list = { name = ulus_kingmakers target = this }
	}

	# ---4. The seat: its own strength, its begs, its heir ----------------
	# The three greatest amirs together. The khan never outweighs ALL
	# his amirs: at his best he is worth three or four of them, and with
	# a court at Almaliq he may not even be the largest. So the reading
	# that means something is the khan against his three GREATEST amirs:
	# near 1.0 he is their match, at 2.0 they outweigh him twice over.
	set_local_variable = { name = ulus_top3 value = 0 }
	ordered_international_organization_member = {
		limit = {
			NOT = { is_leader_of_international_organization = scope:ulus_io }
		}
		order_by = ulus_realm_weight
		max = 3
		check_range_bounds = no
		change_local_variable = { name = ulus_top3 add = var:ulus_weight }
	}

	leader_country ?= {
		set_local_variable = { name = ulus_khan_str value = var:ulus_weight }
		if = {
			limit = { country_has_estate = estate_type:nobles_estate }
			set_local_variable = {
				name = ulus_beg
				value = "estate_power(estate_type:nobles_estate)"
			}
		}
		if = {
			limit = { has_heir = yes }
			heir = {
				scope:ulus_io = {
					set_variable = { name = ulus_khan_heir value = prev }
				}
			}
		}
	}

	# ---5. The regency bar ------------------------------------------------
	set_variable = { name = ulus_amir_count value = local_var:ulus_n }
	set_variable = {
		name = ulus_beg_power
		value = {
			value = local_var:ulus_beg
			multiply = 100
		}
	}
	set_variable = {
		name = ulus_beg_bar
		value = {
			value = local_var:ulus_beg
			divide = 0.5
			max = 1
		}
	}
	if = {
		limit = { local_var:ulus_khan_str > 0 }
		set_variable = {
			name = ulus_amir_ratio
			value = {
				value = local_var:ulus_top3
				divide = local_var:ulus_khan_str
				max = 100
			}
		}
	}
	else = {
		# A khan with no host at all: the bar is met by anyone who has one.
		if = {
			limit = { local_var:ulus_top3 > 0 }
			set_variable = { name = ulus_amir_ratio value = 100 }
		}
		else = {
			set_variable = { name = ulus_amir_ratio value = 0 }
		}
	}
	set_variable = {
		name = ulus_amir_bar
		value = {
			value = var:ulus_amir_ratio
			divide = 2
			max = 1
		}
	}
}
