# =====================================================================
# RISING AGAINST THE KHAN — how the war ends.

# COUNTRY scope: the amir who rose and WON. The khan is killed (Qazan,
# 1346), and the victor takes a boy of the line into his keeping: the
# regency begins with him as regent, not with whoever tops the ranking.
ulus_rising_won = {
	remove_variable = ulus_rising
	save_scope_as = ulus_new_regent
	save_scope_as = ulus_kingmaker	# ulus_pick_puppet_khan reads it
	international_organization:ulus_chaghatay ?= {
		leader_country ?= {
			save_scope_as = ulus_old_seat
			ruler ?= {
				kill_character = {
					target = this
					reason = execution
				}
			}
		}
		ulus_begin_regency_by_force = yes
	}
}

# IO scope; needs scope:ulus_new_regent (= scope:ulus_kingmaker) and
# scope:ulus_old_seat. ulus_begin_regency (15_amirs_regency_effects.txt)
# with the regent given instead of polled, and a lower starting gap.
ulus_begin_regency_by_force = {
	ulus_chaghatay_rebuild_ledger = yes
	ulus_pick_puppet_khan = yes
	ulus_install_regent = yes
	ulus_seat_khan = yes

	set_variable = ulus_regency
	set_variable = { name = ulus_regency_stage value = 1 }
	set_variable = { name = ulus_regency_progress value = 0 }
	set_variable = {
		name = ulus_gap
		value = {
			value = ulus_gap_start
			subtract = ulus_rising_gap_penalty
			min = 0
		}
	}
	set_variable = { name = ulus_gap_filled_months value = 0 }
	set_variable = { name = ulus_gap_empty_months value = 0 }
	set_variable = { name = ulus_murder_clock value = 0 }
	remove_variable = ulus_bar_reduced
	ulus_chaghatay_rebuild_ledger = yes

	scope:ulus_new_regent = {
		# (pledges are paid by ulus_install_regent)
		trigger_event_non_silently = amirs_rising.2
	}
	scope:ulus_old_seat ?= {
		trigger_event_non_silently = amirs_regency.10
	}
}

# COUNTRY scope: the amir who rose and LOST; scope:ulus_khan_seat is the
# khan's realm. The khan's purge has succeeded without a regency ever
# starting.
#   - the begs are cowed for ulus_rising_failed_peace_years (the bar
#     cannot be met; ulus_regency_bar_met reads the variable);
#   - the beaten amir pays ulus_rising_fine_months of income, loses two
#     places outside his capital, and loses standing;
#   - every house that had pledged its tümen to him pays half the fine
#     and loses one place: they backed the wrong man;
#   - the land goes to the lords who had pledged to the KHAN, turn and
#     turn about, or to the khan himself if he had none;
#   - the amir himself: one chance in two he does not leave the khan's
#     tent alive. If he lives, his own house may put him aside
#     (ulus_cadet_after_defeat, scripted_effects/17_amirs_cadet_effects.txt).
ulus_rising_lost = {
	remove_variable = ulus_rising
	save_scope_as = ulus_beaten_riser
	add_prestige = ulus_rising_lost_prestige
	if = {
		limit = { government_type = government_type:steppe_horde }
		add_horde_unity = ulus_rising_lost_unity
	}
	ulus_pay_the_khans_fine = { SHARE = 1 }
	ulus_forfeit_a_place = yes
	ulus_forfeit_a_place = yes

	international_organization:ulus_chaghatay ?= {
		set_variable = { name = ulus_khan_triumphant_peace value = 1 years = ulus_rising_failed_peace_years }
		every_international_organization_member = {
			limit = { var:ulus_pledged_to ?= scope:ulus_beaten_riser }
			remove_variable = ulus_pledged_to
			ulus_pay_the_khans_fine = { SHARE = 0.5 }
			ulus_forfeit_a_place = yes
			trigger_event_non_silently = amirs_rising.12
		}
		# The khan's own lords have chosen well.
		every_international_organization_member = {
			limit = { has_variable = ulus_pledged_to_khan }
			add_prestige = ulus_khan_loyalist_prestige
		}
		ulus_chaghatay_rebuild_ledger = yes
	}
	scope:ulus_khan_seat ?= {
		add_prestige = ulus_rising_khan_prestige
		if = {
			limit = { government_type = government_type:steppe_horde }
			add_horde_unity = ulus_rising_khan_unity
		}
		trigger_event_non_silently = amirs_rising.4
	}
	trigger_event_non_silently = amirs_rising.3

	# The man himself.
	random_list = {
		50 = {
			ruler ?= {
				kill_character = {
					target = this
					reason = execution
				}
			}
		}
		50 = {
			if = {
				limit = { exists = scope:ulus_khan_seat }
				scope:ulus_khan_seat = { save_scope_as = ulus_cadet_invader }
				ulus_cadet_after_defeat = yes
			}
		}
	}
}

# COUNTRY scope: a beaten house; scope:ulus_khan_seat is paid. $SHARE$ of
# ulus_rising_fine_months of its income from trade and tax.
ulus_pay_the_khans_fine = {
	set_local_variable = {
		name = ulus_fine
		value = {
			value = monthly_income_trade_and_tax
			multiply = ulus_rising_fine_months
			multiply = $SHARE$
			min = 0
		}
	}
	add_gold = {
		value = local_var:ulus_fine
		multiply = -1
	}
	scope:ulus_khan_seat ?= {
		add_gold = local_var:ulus_fine
	}
}

# COUNTRY scope: a beaten house gives up one place that is not its
# capital. It goes to a lord pledged to the khan who has had the least
# so far (ulus_spoils_taken counts them), or to the khan if none is.
ulus_forfeit_a_place = {
	if = {
		limit = { exists = scope:ulus_spoils_heir }
		clear_saved_scope = ulus_spoils_heir
	}
	international_organization:ulus_chaghatay ?= {
		ordered_international_organization_member = {
			limit = { has_variable = ulus_pledged_to_khan }
			order_by = ulus_spoils_order
			max = 1
			check_range_bounds = no
			save_scope_as = ulus_spoils_heir
		}
	}
	if = {
		limit = { NOT = { exists = scope:ulus_spoils_heir } }
		scope:ulus_khan_seat ?= { save_scope_as = ulus_spoils_heir }
	}
	if = {
		limit = { exists = scope:ulus_spoils_heir }
		random_owned_location = {
			limit = { is_capital = no }
			change_location_owner = scope:ulus_spoils_heir
		}
		scope:ulus_spoils_heir = {
			if = {
				limit = { has_variable = ulus_spoils_taken }
				change_variable = { name = ulus_spoils_taken add = 1 }
			}
			else = {
				set_variable = { name = ulus_spoils_taken value = 1 }
			}
		}
	}
}

# COUNTRY scope: a lesser lord who agrees to stand with the khan
# (ulus_khan_ask_pledge, or amirs_rising.9.a). scope:ulus_asking_khan is
# the khan's seat.
ulus_pledge_to_khan = {
	set_variable = ulus_pledged_to_khan
	add_prestige = 5
	add_opinion = {
		modifier = opinion_ulus_pledged_to_khan
		target = scope:ulus_asking_khan
	}
	save_scope_as = ulus_khans_lord
	scope:ulus_asking_khan = {
		add_opinion = {
			modifier = opinion_ulus_pledged_to_khan
			target = scope:ulus_khans_lord
		}
		trigger_event_non_silently = amirs_rising.10
	}
	international_organization:ulus_chaghatay ?= {
		ulus_chaghatay_rebuild_ledger = yes
	}
}

# COUNTRY scope: an OUTSIDER who agrees to back scope:ulus_asking_house
# (ulus_ask_outside_support, ulus_back_an_amir, or amirs_rising.11.a).
# One backer per house, one house per backer. The khan does not thank him.
ulus_outsider_backs_house = {
	save_scope_as = ulus_backer
	set_variable = { name = ulus_backs_amir value = scope:ulus_asking_house }
	add_opinion = {
		modifier = opinion_ulus_outside_backing
		target = scope:ulus_asking_house
	}
	scope:ulus_asking_house = {
		set_variable = { name = ulus_outside_backer value = scope:ulus_backer }
		add_opinion = {
			modifier = opinion_ulus_outside_backing
			target = scope:ulus_backer
		}
		trigger_event_non_silently = amirs_rising.13
	}
	international_organization:ulus_chaghatay ?= {
		leader_country ?= {
			if = {
				limit = { NOT = { this = scope:ulus_asking_house } }
				add_opinion = {
					modifier = opinion_ulus_meddler
					target = scope:ulus_backer
				}
			}
		}
		ulus_chaghatay_rebuild_ledger = yes
	}
}

# COUNTRY scope: a house that has just taken the khan. Every amir
# pledged to it is owed: favours with the new regent, and a name for
# having chosen well. The pledge itself stands.
ulus_pay_pledges = {
	save_scope_as = ulus_backed_house
	# An outside backer is owed too.
	if = {
		limit = {
			var:ulus_outside_backer ?= { exists = this }
		}
		var:ulus_outside_backer = {
			add_favors = { target = scope:ulus_backed_house value = ulus_pledge_favors }
			trigger_event_non_silently = amirs_rising.14
		}
	}
	international_organization:ulus_chaghatay ?= {
		every_international_organization_member = {
			limit = { var:ulus_pledged_to ?= scope:ulus_backed_house }
			add_favors = { target = scope:ulus_backed_house value = ulus_pledge_favors }
			add_prestige = ulus_pledge_prestige
			trigger_event_non_silently = amirs_rising.6
		}
	}
}

# COUNTRY scope: a house that has agreed to pledge to

ulus_pledge_to_asker = {

	scope:ulus_asking_house = {
		trigger_event_non_silently = amirs_rising.5
	}
	international_organization:ulus_chaghatay ?= {

	}
}

# ---------------------------------------------------------------------
# THE RECKONING, SHOWN. Tooltip-only effects: what a house will answer
# when asked, and each thing it weighs, as lines the player can read
# before he pays. They repeat the terms of the acceptance values
# (script_values/05_amirs_regency_values.txt, sections 12, 15, 16); if a
# term changes there, change its line here and its text in
# amirs_rising_l_english.yml.
# COUNTRY scope: the house (or outsider) that is asked.
# ---------------------------------------------------------------------

# scope:actor asks this house to pledge to him.
ulus_show_pledge_reckoning = {
	if = {
		limit = { is_ai = no }
		custom_tooltip = ulus_reckon_player_tt
	}
	else_if = {
		limit = { ulus_pledge_acceptance >= 0 }
		custom_tooltip = ulus_reckon_agree_tt
	}
	else = {
		custom_tooltip = ulus_reckon_refuse_tt
	}
	custom_tooltip = ulus_reckon_reluctance_25_tt
	ulus_show_reckoning_of_actor = yes
	if = {
		limit = {
			has_variable = ulus_weight
			scope:actor = { has_variable = ulus_weight }
		}
		if = {
			limit = {
				scope:actor.var:ulus_weight >= {
					value = var:ulus_weight
					multiply = 2
				}
			}
			custom_tooltip = ulus_reckon_twice_30_tt
		}
		else_if = {
			limit = {
				scope:actor.var:ulus_weight >= {
					value = var:ulus_weight
					multiply = 1.25
				}
			}
			custom_tooltip = ulus_reckon_heavier_15_tt
		}
		else_if = {
			limit = { scope:actor.var:ulus_weight < var:ulus_weight }
			custom_tooltip = ulus_reckon_lighter_tt
		}
		else = {
			custom_tooltip = ulus_reckon_even_tt
		}
	}
	if = {
		limit = {
			is_target_in_global_variable_list = {
				name = ulus_kingmakers
				target = this
			}
		}
		custom_tooltip = ulus_reckon_own_hopes_tt
	}
	if = {
		limit = {
			scope:actor = {
				is_target_in_global_variable_list = {
					name = ulus_kingmakers
					target = this
				}
			}
		}
		custom_tooltip = ulus_reckon_we_contend_10_tt
	}
}

# Shared: what this house makes of scope:actor himself.
ulus_show_reckoning_of_actor = {
	if = {
		limit = {
			opinion = {
				target = scope:actor
				value >= 100
			}
		}
		custom_tooltip = ulus_reckon_opinion_high_tt
	}
	else_if = {
		limit = {
			opinion = {
				target = scope:actor
				value >= 0
			}
		}
		custom_tooltip = ulus_reckon_opinion_mid_tt
	}
	else = {
		custom_tooltip = ulus_reckon_opinion_low_tt
	}
	if = {
		limit = { has_royal_marriage_with = scope:actor }
		custom_tooltip = ulus_reckon_marriage_25_tt
	}
	if = {
		limit = { is_allied_with = { target = scope:actor } }
		custom_tooltip = ulus_reckon_alliance_15_tt
	}
	if = {
		limit = { is_rival_of = scope:actor }
		custom_tooltip = ulus_reckon_rival_100_tt
	}
}

# scope:ulus_asking_house asks this OUTSIDER to back it.
ulus_show_outside_reckoning = {
	if = {
		limit = { is_ai = no }
		custom_tooltip = ulus_reckon_player_tt
	}
	else_if = {
		limit = { ulus_outside_support_acceptance >= 0 }
		custom_tooltip = ulus_reckon_agree_tt
	}
	else = {
		custom_tooltip = ulus_reckon_refuse_tt
	}
	custom_tooltip = ulus_reckon_reluctance_20_tt
	if = {
		limit = {
			opinion = {
				target = scope:ulus_asking_house
				value >= 100
			}
		}
		custom_tooltip = ulus_reckon_opinion_high_tt
	}
	else_if = {
		limit = {
			opinion = {
				target = scope:ulus_asking_house
				value >= 0
			}
		}
		custom_tooltip = ulus_reckon_opinion_mid_tt
	}
	else = {
		custom_tooltip = ulus_reckon_opinion_low_tt
	}
	if = {
		limit = { is_rival_of = scope:ulus_asking_house }
		custom_tooltip = ulus_reckon_rival_100_tt
	}
	if = {
		limit = {
			international_organization:ulus_chaghatay ?= { international_organization_has_leader = yes }
			is_allied_with = { target = international_organization:ulus_chaghatay.leader_country }
		}
		custom_tooltip = ulus_reckon_out_khans_ally_tt
	}
	else_if = {
		limit = {
			international_organization:ulus_chaghatay ?= { international_organization_has_leader = yes }
			is_rival_of = international_organization:ulus_chaghatay.leader_country
		}
		custom_tooltip = ulus_reckon_out_khans_rival_tt
	}
	if = {
		limit = {
			international_organization:ulus_chaghatay ?= { international_organization_has_leader = yes }
			is_neighbor_of = international_organization:ulus_chaghatay.leader_country
		}
		custom_tooltip = ulus_reckon_out_khans_neighbour_tt
	}
	if = {
		limit = {
			scope:ulus_asking_house = { has_variable = ulus_kingmaker_score }
			international_organization:ulus_chaghatay ?= {
				has_variable = ulus_top_kingmaker_score
				var:ulus_top_kingmaker_score > 0
			}
		}
		if = {
			limit = {
				scope:ulus_asking_house.var:ulus_kingmaker_score >= {
					value = international_organization:ulus_chaghatay.var:ulus_top_kingmaker_score
					multiply = 0.9
				}
			}
			custom_tooltip = ulus_reckon_out_too_strong_tt
		}
		else_if = {
			limit = {
				scope:ulus_asking_house.var:ulus_kingmaker_score >= {
					value = international_organization:ulus_chaghatay.var:ulus_top_kingmaker_score
					multiply = 0.5
				}
			}
			custom_tooltip = ulus_reckon_out_contender_tt
		}
		else = {
			custom_tooltip = ulus_reckon_out_too_weak_tt
		}
	}
	custom_tooltip = ulus_reckon_out_temper_tt
}
