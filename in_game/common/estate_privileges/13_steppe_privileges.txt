# The command of a tümen passes in the beg's house, and inside it his
# word is the law: he judges, he assigns pasture, he takes his due. The
# khan may confirm him; the khan does not appoint him.
hereditary_tumens = {
	estate = nobles_estate

	potential = {
		government_type = government_type:steppe_horde
	}

	country_modifier = {
		global_nobles_estate_power = 0.5
		monthly_political_influence_gain_modifier = -0.10
		nobles_estate_target_satisfaction = medium_privilege_target_satisfaction
		monthly_towards_aristocracy = societal_value_monthly_move
		global_max_control = -0.05
		global_tribesmen_enfranchisment = -0.05
		# A tümen is its beg's own host (base gives this to the nobles
		# through its private-army privilege).
		nobles_estate_allowed_private_army = yes
	}
}

	}
