on_new_country_formed = {
	on_actions = {
		correct_kurdish_tribe_newly_formed
        correct_sedentary_iranic_tribe
	}
}


correct_kurdish_tribe_newly_formed = {
	trigger = {
		culture ?= culture:kurdish_culture
        government_type ?= government_type:tribe
		}
	
	effect = {
        change_government_type = government_type:iranic_tribal_polity
        add_reform = government_reform:kurdish_emirate
			
		
	}

}


correct_sedentary_iranic_tribe = {
	trigger = {
        OR = {
            culture ?= culture:khorasani_culture
            culture ?= culture:farsi_culture
            culture ?= culture:adhari_culture
            culture ?= culture:tajik_culture
            culture ?= culture:achomi_culture
            culture ?= culture:gilak_culture
            culture ?= culture:mazanderani_culture
            culture ?= culture:talysh_culture
            culture ?= culture:dehwar_culture
            culture ?= culture:semnani_culture
        }
        government_type ?= government_type:tribe
		}
	
	effect = {
        change_government_type = government_type:persianate_sultanate
        add_reform = government_reform:persianate_bureaucracy_reform
			
		
	}

}
