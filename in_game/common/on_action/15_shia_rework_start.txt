# =====================================================================
# SHIA REWORK — game-start wiring
#
# 1. Reassign base-setup countries off the detached branch schools
#    (jafari/ismaili are no longer attached to the shia religion) via
#    set_religious_school — safer than partial setup overrides.
# 2. Seed starting Imamate aspects for states with a settled
#    confessional identity in 1337. Everyone else starts with NO
#    imamate aspect: that is post-Mongol confessional ambiguity, and
#    committing is meant to be an in-game act (30 stability per
#    aspect change, the base non-Christian price).
#
# CHB is deliberately NOT seeded: Twelver-leaning, uncommitted — the
# ambiguity showcase. Zaydi-school states keep zaidi_school (kept,
# re-read as the imamate-scholar network) and get imamate_zaydi.
# =====================================================================

on_game_start = {
	on_actions = {
		shia_rework_start
	}
}

shia_rework_start = {
	effect = {
		# --- School reassignment off detached branch labels ---
		c:CHB ?= {
			if = {
				limit = { religion = religion:shia }
				set_religious_school = religious_school:hilla_hawza_school
			}
		}
		# Base-setup Ismaili communities → the Nizari daʿwa network.
		c:GLI ?= {
			if = {
				limit = { religion = religion:shia }
				set_religious_school = religious_school:nizari_school
			}
		}
		c:HNZ ?= {
			if = {
				limit = { religion = religion:shia }
				set_religious_school = religious_school:nizari_school
			}
		}
		c:SND ?= {
			if = {
				limit = { religion = religion:shia }
				set_religious_school = religious_school:nizari_school
			}
		}
		c:DUL ?= {
			if = {
				limit = { religion = religion:shia }
				set_religious_school = religious_school:nizari_school
			}
		}

		# --- Starting Imamate aspects ---
		# Twelver, committed: the Sarbadars (coin in the Hidden Imam's
		# name = niyaba amma) and the Baduspanids.
		c:SRB ?= {
			if = {
				limit = { religion = religion:shia }
				add_religious_aspect = religious_aspect:imamate_ithna_ashari
				add_religious_aspect = religious_aspect:niyaba_amma
			}
		}
		c:BDS ?= {
			if = {
				limit = { religion = religion:shia }
				add_religious_aspect = religious_aspect:imamate_ithna_ashari
			}
		}
		# Zaydi states: Caspian highlands + the old Zaydi south.
		c:MZN ?= {
			if = {
				limit = { religion = religion:shia }
				add_religious_aspect = religious_aspect:imamate_zaydi
			}
		}
		c:TLM ?= {
			if = {
				limit = { religion = religion:shia }
				add_religious_aspect = religious_aspect:imamate_zaydi
			}
		}
		c:LCK ?= {
			if = {
				limit = { religion = religion:shia }
				add_religious_aspect = religious_aspect:imamate_zaydi
			}
		}
		c:HLL ?= {
			if = {
				limit = { religion = religion:shia }
				add_religious_aspect = religious_aspect:imamate_zaydi
			}
		}
		c:IHB ?= {
			if = {
				limit = { religion = religion:shia }
				add_religious_aspect = religious_aspect:imamate_zaydi
			}
		}
		# Ismaili communities.
		c:GLI ?= {
			if = {
				limit = { religion = religion:shia }
				add_religious_aspect = religious_aspect:imamate_ismaili
			}
		}
		c:HNZ ?= {
			if = {
				limit = { religion = religion:shia }
				add_religious_aspect = religious_aspect:imamate_ismaili
			}
		}
		c:SND ?= {
			if = {
				limit = { religion = religion:shia }
				add_religious_aspect = religious_aspect:imamate_ismaili
			}
		}
		c:DUL ?= {
			if = {
				limit = { religion = religion:shia }
				add_religious_aspect = religious_aspect:imamate_ismaili
			}
		}
	}
}
