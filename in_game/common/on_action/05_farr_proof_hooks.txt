# ============================================================
# On-action hooks for the Farr Proof of Fortune disaster.
#
# Paradox merges on_actions across files. Re-declaring an engine
# on_action here EXTENDS it — the engine still runs its built-in
# logic; we just add a farr-specific tail.
#
# Hooks:
#   on_battle_won    — small per-battle legitimacy bump
#   on_battle_lost   — small per-battle legitimacy hit
#   on_winning_war   — apply farr_conquest_glory (the +0.4/mo
#                      stacking modifier) + flat legitimacy
#   on_losing_war    — apply farr_defeat_shame (-1.0/mo) + flat
#                      legitimacy hit
#   on_location_changed_owner — capital lost (-20 + capital_lost
#                      modifier); non-capital province lost (-3)
#
# All hooks check has_variable = farr_proof_fired so they only
# fire while the disaster is active.
#
# Each hook also fires a one-shot INDICATOR event the first time
# its effect applies, gated on two variables (see farr_proof.300+):
#   farr_indicator_<situation>_seen   — per-hook one-shot
#   farr_indicators_suppressed        — master mute for the run
# ============================================================
# on_battle_won = {
# 	on_actions = {
#     	on_farr_holder_battle_win
# 		on_farr_holder_battle_win_actor
		
#     }

on_great_battle_won = {
	on_actions = {
		on_farr_holder_battle_win
		
		
	}
}



on_battle_lost = {
	on_actions = {
		farr_on_battle_lost
		
	}
}

on_winning_war = {
	on_actions = {
		farr_on_winning_war
		
	}
}

on_losing_war = {
	on_actions = {
		farr_on_losing_war
		
	}
}

on_location_changed_owner = {
	on_actions = {
		farr_on_location_changed_owner_loss
		farr_on_location_changed_owner_win
		
	}
}



# ---Battle won: scope:actor = winning unit, scope:target = losing unit.


on_farr_holder_battle_win = {

	trigger = {
		scope:actor = {
			owner = { has_variable = farr_proof_fired }
		}
	}
	effect = {

		if = {
			limit = {
				scope:actor = {
					owner = { has_variable = farr_proof_fired }
				}
			}
			scope:actor.owner = {
				if = {
					limit = {
						NOT = { has_variable = farr_indicator_battle_won_seen }
						NOT = { has_variable = farr_indicators_suppressed }
					}
					trigger_event_non_silently = farr_proof.300
				}
				else_if = {
					limit = {
						has_variable = farr_indicators_suppressed
					}
					trigger_event_silently = farr_proof.300
				}
			}
		}
	}
}

# ---Battle lost: scope:actor = losing unit, scope:target = winning unit.
farr_on_battle_lost = {


	trigger = {
		scope:actor = {
			owner = { has_variable = farr_proof_fired }
		}
	}
	effect = {
		if = {
			limit = {
				scope:actor = {
					owner = { has_variable = farr_proof_fired }
				}
			}
			scope:actor.owner = {

				if = {
					limit = {
						NOT = { has_variable = farr_indicator_battle_lost_seen } 
						NOT = { has_variable = farr_indicators_suppressed }
					}
					trigger_event_non_silently = farr_proof.301
				}
				else_if = {
					limit = {
						has_variable = farr_indicators_suppressed
					}
					trigger_event_silently = farr_proof.301
				}
			}
		}
	}
}

# # ---Winning war: scope:winner = country, scope:loser = country.
farr_on_winning_war = {

	trigger = {
		scope:winner = {
			has_variable = farr_proof_fired
		}
	}
	effect = {
		if = {
			limit = {
				scope:winner ?= { has_variable = farr_proof_fired }
			}
			scope:winner = {

				if = {
					limit = {
						NOT = { has_variable = farr_indicators_suppressed }
					}
					trigger_event_non_silently = farr_proof.302
				}
				else_if = {
					limit = {
						has_variable = farr_indicators_suppressed
					}
					trigger_event_silently = farr_proof.302
				}
			}
		}
	}
}

# # ---Losing war: heaven visibly withdrew.
farr_on_losing_war = {

	}
	effect = {
		if = {
			limit = {
				scope:loser ?= { has_variable = farr_proof_fired }
			}
			scope:loser = {
				if = {
					limit = {
						NOT = { has_variable = farr_indicators_suppressed }

					trigger_event_non_silently = farr_proof.303
				}

					limit = {
						has_variable = farr_indicators_suppressed

}

# # ---Location changed owner: this = location, scope:winner / scope:loser = countries.
# # Capital lost is the visible humiliation — Tīmūr losing Samarqand,
# # Yazīd losing Baghdād. The check uses the loser's *current* capital
# # at the moment the trigger evaluates (engine moves capital after
# # location ownership changes, so this should still catch the original).
farr_on_location_changed_owner_loss = {
	trigger = {
		scope:loser = {
			has_variable = farr_proof_fired
		}
	}
	effect = {
		# Capital lost — the seat itself has changed hands.
		if = {
			limit = {
				scope:loser ?= {
					has_variable = farr_proof_fired
					capital = root
				}
			}
			scope:loser = {

				if = {
					limit = {
						NOT = { has_variable = farr_indicators_suppressed }
					}
					trigger_event_non_silently = farr_proof.304
				}
				else_if = {
					limit = {
						has_variable = farr_indicators_suppressed
					}
					trigger_event_silently = farr_proof.304
				}
			}
		}
		# Non-capital province lost — a smaller, persistent erosion.
		else_if = {
			limit = {
				scope:loser ?= { has_variable = farr_proof_fired }
			}
			scope:loser = {
				
				if = {
					limit = {
						NOT = { has_variable = farr_indicator_province_lost_seen }
						NOT = { has_variable = farr_indicators_suppressed }
					}
					trigger_event_non_silently = farr_proof.305
				}
				else_if = {
					limit = {
						has_variable = farr_indicators_suppressed
					}
					trigger_event_silently = farr_proof.305
				}
			}
		}
	}
}

farr_on_location_changed_owner_win = {
	trigger = {
		scope:winner = {
			has_variable = farr_proof_fired
		}
	}
	effect = {
		# Capital of an enemy — the seat itself has changed hands.
		
		# Non-capital province lost — a smaller, persistent erosion.
		if = {
			limit = {
				scope:winner ?= { has_variable = farr_proof_fired }
			}
			scope:winner = {
				
				if = {
					limit = {
						NOT = { has_variable = farr_indicator_province_gained_seen }
						NOT = { has_variable = farr_indicators_suppressed }
					}
					trigger_event_non_silently = farr_proof.306
				}
				else_if = {
					limit = {
						has_variable = farr_indicators_suppressed
					}
					trigger_event_silently = farr_proof.306
				}
			}
		}
	}
}
