on_character_death = {
	on_actions = {
		on_borjigin_puppet_death
	}
}

# When the Borjigin puppet character dies, the patron country faces the
# Chinggisid crisis: end the fiction (drop chinggisid_regency reform and
# eat the legitimacy/estate consequences) or enthrone the puppet's heir
# and let the charade continue at a steeper price.
on_borjigin_puppet_death = {
	trigger = {
		scope:target = {
			exists = var:patron_country
			var:patron_country = {
				has_reform = government_reform:chinggisid_regency
			}
		}
	}
	effect = {
		scope:target = {
			var:patron_country = {
				trigger_event_non_silently = { id = ilkhanate_flavor.2 }
			}
		}
	}
}
