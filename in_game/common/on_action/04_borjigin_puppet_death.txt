on_character_death = {
	on_actions = {
		on_borjigin_puppet_death
	}
}

# When the Borjigin puppet character dies, the patron country faces the
# Chinggisid crisis: end the fiction (drop chinggisid_regency reform and
# eat the legitimacy/estate consequences) or enthrone the puppet's heir
# and let the charade continue at a steeper price.
#
# The same marker (patron_country on the man) is carried by the puppet
# khans of Ulus Chaghatay's regents (scripted_effects/
# 15_amirs_regency_effects.txt, ulus_seat_khan). Their patron holds
# beglerbeg_regency, not chinggisid_regency, and gets its own choice:
# amirs_regency.30 (enthrone another / a khan of the in-law line / rule
# without one), never ilkhanate_flavor.2.
on_borjigin_puppet_death = {
	trigger = {
		scope:target = {
			exists = var:patron_country
			var:patron_country = {
				OR = {
					AND = {
						has_reform = government_reform:chinggisid_regency
						# The Restored Yarliq ends the puppet cycle — the
						# khan is the patron's own house now.
						NOT = { has_variable = custody_restored_yarliq }
					}
					AND = {
						has_reform = government_reform:beglerbeg_regency
						has_variable = ulus_regent_seat
					}
				}
			}
		}
	}
	effect = {
		scope:target = {
			var:patron_country = {
				if = {
					limit = {
						has_reform = government_reform:beglerbeg_regency
						has_variable = ulus_regent_seat
					}
					trigger_event_non_silently = { id = amirs_regency.30 }
				}
				else = {
					trigger_event_non_silently = { id = ilkhanate_flavor.2 }
				}
			}
		}
	}
}
