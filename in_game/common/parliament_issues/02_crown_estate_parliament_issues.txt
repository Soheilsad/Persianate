# Reopening the Audit — the istīfāʾ returns to the parcels that slipped it.
# The unaudited iqṭāʿ is not a legal form, it is a lapse: the amir simply
# stopped submitting his books. This decree is the chancery's answer — the
# assessment rides out again, and a share of the lapsed grants are forced
# back onto the register (reverting to plain, audited iqṭāʿ — the base
# grant underneath is untouched). The toyul is past reach: hereditary,
# lineage-locked, and legally immune — only direct order pries that loose,
# not a decree.
audit_unaudited_iqta = {
	estate = crown_estate

	modifier_when_in_debate = {
		# The prospect of lapsed parcels returning to the register.
		tax_income_efficiency = 0.05
	}

	allow = {
		parliament_type = parliament_type:diwan_council
		any_owned_location = { has_building = building_type:unaudited_iqta }
	}

	chance = 25

	on_debate_passed = {
		# Lawful dissolution: the audit is a judgment of the maẓālim, not a
		# seizure — the buildings' on_destroyed reads this variable and cuts
		# the backlash by 75%. Timer (not manual unset) so any deferred
		# destruction resolution still sees it; it expires on its own.
		set_variable = { name = decree_lawful_dissolution value = 1 days = 2 }
		# A quarter of the lapsed grants are re-registered. The buildings'
		# own on_destroyed carries the amirs' backlash per parcel.
		every_owned_location = {
			limit = {
				has_building = building_type:unaudited_iqta
				# Where the grant has already locked into the lineage, the
				# audit finds a sealed door.
				NOT = { has_building = building_type:toyul }
			}
			random = {
				chance = 25
				destroy_building = "building(building_type:unaudited_iqta|prev)"
			}
		}
		# The register reasserted: the realm drifts back toward the Tazik
		# order (an aggregate of the per-parcel drift the slips caused),
		# and the chancery sees the crown enforce its own instrument.
		change_societal_value = { type = turk_vs_tazik value = 0.5 }
		add_estate_satisfaction = { type = estate_type:diwan_estate value = 0.05 }
		add_legitimacy = 5
	}

	on_debate_failed = {
		parliament_debate_failed_effect = yes
	}
}

# Removing the Illegal Waqf — the crown dissolves endowments that shelter
# wealth from the fisc, clawing a share back into taxable hands. The waqf's
# whole defence is inviolability; so this is possible ONLY where the right
# of muṣādara stands (access_to_musadara) AND no inviolability charter has
# been granted to any estate. Once the endowments are made inviolable, even
# the crown's hand is stayed — that is the charter's purpose, and the trap.
# Strips ~20% of the realm's waqf buildings.
purge_illegal_waqf = {
	estate = crown_estate

	modifier_when_in_debate = {
		# The prospect of reclaimed revenue while the endowed houses tremble.
		tax_income_efficiency = 0.05
	}

	allow = {
		parliament_type = parliament_type:diwan_council
		modifier:access_to_musadara = yes
		# No endowment may be inviolable — a single charter stays the hand.
		NOT = { has_estate_privilege = estate_privilege:ayyan_waqf_charter }
		NOT = { has_estate_privilege = estate_privilege:urban_waqf_charter }
		NOT = { has_estate_privilege = estate_privilege:clerical_waqf_charter }
	}

	chance = 25

	on_debate_passed = {
		# Lawful dissolution: destroyed by decree of the maẓālim, so the
		# waqf on_destroyed backlash is cut 75% while this variable stands.
		# 2-day timer, self-expiring (see audit_unaudited_iqta note).
		set_variable = { name = decree_lawful_dissolution value = 1 days = 2 }
		# A fifth of the realm's waqf is dissolved. Each location holds at
		# most one waqf; the non-present types simply no-op. The buildings'
		# own on_destroyed carries the estate backlash.
		every_owned_location = {
			limit = {
				OR = {
					has_building = building_type:ayyan_waqf
					has_building = building_type:urban_waqf
					has_building = building_type:clerical_waqf
					has_building = building_type:grand_ayyan_waqf
					has_building = building_type:grand_urban_waqf
					has_building = building_type:grand_clerical_waqf
				}
			}
			random = {
				chance = 20
				destroy_building = "building(building_type:ayyan_waqf|prev)"
				destroy_building = "building(building_type:urban_waqf|prev)"
				destroy_building = "building(building_type:clerical_waqf|prev)"
				destroy_building = "building(building_type:grand_ayyan_waqf|prev)"
				destroy_building = "building(building_type:grand_urban_waqf|prev)"
				destroy_building = "building(building_type:grand_clerical_waqf|prev)"
			}
		}
		# The crown is seen to enforce the fisc against the dead hand.
		add_legitimacy = 5
	}

	on_debate_failed = {
		parliament_debate_failed_effect = yes
	}
}
