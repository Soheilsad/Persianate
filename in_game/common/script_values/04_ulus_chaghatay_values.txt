# Script values for Ulus Chaghatay.
# Read by scripted_effects/14_ulus_chaghatay_effects.txt (the panel's
# ledger) and, from phase 2, by the succession poll and the choice of
# regent. ONE definition on purpose: the number the panel shows is the
# number that decides.

# THE KINGMAKER SCORE — who takes the khan if the throne falls vacant
# with the begs over the bar. COUNTRY scope (a junior member).
#
#   HIS REALM'S WEIGHT, FIRST    his share of all the amirs' weight
#                               (ulus_amir_share: military strength, tax
#                               base and prestige, NOT a count of
#                               locations) x400. An amir holding a
#                               quarter of the amirs' weight scores 100
#                               here; one holding 3% scores 12. The
#                               regency goes to the man with the tümens:
#                               Qazaghan led the Qara'unas, the largest
#                               host in the ulus.
#   ruler's military skill x0.5  0-50: the man the tümens will follow
#   ruler's administrative x0.2  0-20: holding it is another matter
#                               (First build scored skill x2 and x1, up
#                               to 300 points against 60 for the realm:
#                               with random rulers a petty amir with a
#                               gifted chief outranked the Yasa'uri, who
#                               field ten times his men.)
#   a wife of the line          +25: the güregen's claim, Temür's own
#
# Character abilities are 0-100 (base reads them as ruler.adm / heir.mil;
# events/ and script_values/). The historical houses get no bonus.
#   battles won                 his renown (ulus_renown on the MAN, kept
#                               by on_action/31_amirs_regency_hooks.txt
#                               from great battles and sieges won while
#                               a regency lasts) x ulus_renown_weight,
#                               capped at ulus_renown_cap. Nothing before
#                               a regency: this is where a Temür is made.
ulus_kingmaker_score = {
	value = 0
	if = {
		limit = { has_ruler = yes }
		add = {
			value = ruler.mil
			multiply = 0.5
		}
		add = {
			value = ruler.adm
			multiply = 0.2
		}
	}
	if = {
		limit = { has_variable = ulus_amir_share }
		add = {
			value = var:ulus_amir_share
			multiply = 400
		}
	}
	if = {
		limit = {
			ruler ?= {
				any_spouse = { dynasty ?= dynasty:borjigin_dynasty }
			}
		}
		add = 25
	}
	if = {
		limit = {
			ruler ?= { has_variable = ulus_renown }
		}
		add = {
			value = ruler.var:ulus_renown
			multiply = ulus_renown_weight
			max = ulus_renown_cap
		}
	}
}

# The khan on the same scale as his amirs, for the panel's "Standing in
# the ulus" ranking: an amir gets 25 for a WIFE of the line, the khan
# twice that for being of the line himself.
ulus_khan_line_bonus = 50

# The stored score, for ordering that ranking (the khan's stored score
# carries his line bonus; the script value above does not). COUNTRY scope.
ulus_standing_score = {
	value = 0
	if = {
		limit = { has_variable = ulus_kingmaker_score }
		add = var:ulus_kingmaker_score
	}
}

# A member's weight, for ordering (the stored variable; 0 before the
# first ledger run). COUNTRY scope.
# EFFECTIVE weight since pledges exist (ledger, step 1b): his own, less
# the share he has lent to another house, plus what others lend him.
ulus_realm_weight = {
	value = 0
	if = {
		limit = { has_variable = ulus_weight }
		add = var:ulus_weight
	}
	if = {
		limit = {
			OR = {
				has_variable = ulus_pledged_to
				has_variable = ulus_pledged_to_khan
			}
		}
		multiply = {
			value = 1
			subtract = ulus_pledge_share
		}
	}
	if = {
		limit = { has_variable = ulus_backing }
		add = var:ulus_backing
	}
	# Friends outside the ulus: a share of his allies' weight and of an
	# outside backer's (ledger, step 1c).
	if = {
		limit = { has_variable = ulus_outside_weight }
		add = var:ulus_outside_weight
	}
	# A passing gain in standing (timed variable): largesse given, a
	# marriage into the line (generic_actions/amirs_regency_actions.txt).
	if = {
		limit = { has_variable = ulus_weight_boost }
		add = var:ulus_weight_boost
	}
}

# THE WEIGHT OF A REALM, in parts out of 100 across the whole ulus
# (author, 2026-11-14): the men it can put in the field, what it can pay
# them, the man who leads them, and the name it carries.
ulus_w_mil = 40
ulus_w_tax = 30
ulus_w_rul = 20
ulus_w_pre = 10

# The man who leads them: his three abilities together (0-300).
# COUNTRY scope.
ulus_ruler_abilities = {
	value = 0
	if = {
		limit = { has_ruler = yes }
		add = ruler.adm
		add = ruler.dip
		add = ruler.mil
	}
}

# FRIENDS OUTSIDE. An ally who is not of the ulus counts for
# ulus_ally_fraction of what he would weigh if he were (his host and his
# tax base on the ulus's own scale; no single friend for more than
# ulus_outside_friend_cap). An outside BACKER who is not an ally
# (ulus_ask_outside_support) counts for ulus_backer_fraction. All of a
# house's outside friends together count for at most ulus_outside_cap.
ulus_ally_fraction = 0.25
ulus_backer_fraction = 0.15
ulus_outside_friend_cap = 12
ulus_outside_cap = 10

# The share of his weight a pledged amir lends the house he backs.
ulus_pledge_share = 0.5

# For sharing out a beaten rebel's land among the khan's lords: fewest
# spoils so far first (ulus_forfeit_a_place). COUNTRY scope.
ulus_spoils_order = {
	value = 0
	if = {
		limit = { has_variable = ulus_spoils_taken }
		subtract = var:ulus_spoils_taken
	}
}
