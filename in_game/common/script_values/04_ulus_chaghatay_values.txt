# Script values for Ulus Chaghatay.
# Read by scripted_effects/14_ulus_chaghatay_effects.txt (the panel's
# ledger) and, from phase 2, by the succession poll and the choice of
# regent. ONE definition on purpose: the number the panel shows is the
# number that decides.

# THE KINGMAKER SCORE — who takes the khan if the throne falls vacant
# with the begs over the bar. COUNTRY scope (a junior member).
#
#   ruler's military skill x2   the regency is taken by the man the
#                               tümens will follow: Qazaghan was the
#                               ulus's best general before he was its
#                               master
#   ruler's administrative skill  holding it is another matter, and it
#                               was the bitikchis and the settled towns
#                               that kept Qazaghan twelve years
#   domain                      his realm's share of all the amirs'
#                               weight (ulus_amir_share: military
#                               strength, tax base and prestige, NOT a
#                               count of locations) x250, capped at 60:
#                               a quarter of the amirs' weight is worth
#                               the full 60
#   a wife of the line          +25: the güregen's claim, Temür's own
#
# Character abilities are 0-100 (base reads them as ruler.adm / heir.mil;
# events/ and script_values/). The historical houses get no bonus.
#   battles won                 his renown (ulus_renown on the MAN, kept
#                               by on_action/31_amirs_regency_hooks.txt
#                               from great battles and sieges won while
#                               a regency lasts) x ulus_renown_weight,
#                               capped at ulus_renown_cap. Nothing before
#                               a regency: this is where a Temür is made.
ulus_kingmaker_score = {
	value = 0
	if = {
		limit = { has_ruler = yes }
		add = {
			value = ruler.mil
			multiply = 2
		}
		add = ruler.adm
	}
	if = {
		limit = { has_variable = ulus_amir_share }
		add = {
			value = var:ulus_amir_share
			multiply = 250
			max = 60
		}
	}
	if = {
		limit = {
			ruler ?= {
				any_spouse = { dynasty ?= dynasty:borjigin_dynasty }
			}
		}
		add = 25
	}
	if = {
		limit = {
			ruler ?= { has_variable = ulus_renown }
		}
		add = {
			value = ruler.var:ulus_renown
			multiply = ulus_renown_weight
			max = ulus_renown_cap
		}
	}
}

# A member's weight, for ordering (the stored variable; 0 before the
# first ledger run). COUNTRY scope.
ulus_realm_weight = {
	value = 0
	if = {
		limit = { has_variable = ulus_weight }
		add = var:ulus_weight
	}
}
