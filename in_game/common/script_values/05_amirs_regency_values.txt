# The Amirs' Regency — every number the succession and the regency run
# on, in one place. Tune here after watching AI runs; nothing below is
# repeated as a literal elsewhere EXCEPT the monthly percentage rolls
# (`random = { chance = N }` takes a literal in every base use), which
# sit together in scripted_effects/15_amirs_regency_effects.txt under
# "THE ROLLS".
#
# History behind the defaults (Manz, The Rise and Rule of Tamerlane):
# Qazaghan took the ulus in 1346/7, was murdered in 1358, Tughluq Temür
# invaded in 1360 and 1361, Temür took Balkh in 1370. Twenty-four
# years from the first regent to the Ṣāḥib-qirān.

# =====================================================================
# 1. THE REGENCY BAR (read by the succession poll and the panel)
# =====================================================================

# Nobles' estate power in the khan's own realm (0..1).
ulus_regency_noble_power = 0.50
# ...and the nobles must hold at least this multiple of the crown's.
ulus_regency_noble_crown_ratio = 2
# OR: the three greatest amirs' combined weight / the khan's own.
# In-game reading at the 1337 start: 5.25 on day one, 3.5 after the
# first month (author, 2026-11-14) — the khan at Almaliq is already
# outweighed, as he was: six khans in the twelve years after 1334. So
# the bar that marks the amirs' REGENCY (1346) has to sit above the
# starting disorder, or every game opens with one.
# It is only tested at a khan's death (the poll), never on its own. 6.0
# was too high (author); 4.0 sits between the two readings, so the first
# succession can go either way. Retune once the four carved tümens
# survive day one and the reading settles.
ulus_regency_amir_ratio = 4.0
# After a child khan's accession the next poll's bars are multiplied by
# this (design: "drops by a third").
ulus_regency_child_bar_factor = 0.667

# The bars as they stand today (the child-khan reduction applied).
# Any scope.
ulus_regency_noble_power_now = {
	value = ulus_regency_noble_power
	if = {
		limit = {
			international_organization:ulus_chaghatay ?= { has_variable = ulus_bar_reduced }
		}
		multiply = ulus_regency_child_bar_factor
	}
}
ulus_regency_amir_ratio_now = {
	value = ulus_regency_amir_ratio
	if = {
		limit = {
			international_organization:ulus_chaghatay ?= { has_variable = ulus_bar_reduced }
		}
		multiply = ulus_regency_child_bar_factor
	}
}

# =====================================================================
# 2. WHO TAKES THE KHAN
# =====================================================================

# The regent's heir keeps the regency if his score is at least this
# share of the best rival amir's (Qazaghan's son ʿAbdallāh did not
# clear it in 1358).
ulus_regent_heir_ratio = 0.80

# A darugha has no domain; his seat stands in for the amirs' domain
# term of ulus_kingmaker_score (which caps at 60).
ulus_darugha_seat_bonus = 20

# Renown: gained by an amir's ruler from victories while a regency
# lasts; the "battles won" term of ulus_kingmaker_score.
ulus_renown_great_battle = 3
ulus_renown_siege = 1
# Each point of renown is worth this much kingmaker score...
ulus_renown_weight = 2
# ...up to this cap (the same ceiling as the domain term).
ulus_renown_cap = 60

# A DARUGHA'S STANDING — character scope. The same shape as
# ulus_kingmaker_score (script_values/04_ulus_chaghatay_values.txt):
# sword twice, pen once, his seat for a domain, his renown.
ulus_darugha_score = {
	value = mil
	multiply = 2
	add = adm
	add = ulus_darugha_seat_bonus
	if = {
		limit = { has_variable = ulus_renown }
		add = {
			value = var:ulus_renown
			multiply = ulus_renown_weight
			max = ulus_renown_cap
		}
	}
}

# WHO MAKES THE BEST PUPPET — character scope, higher is better. A boy
# on the felt is worth more to a regent than any grown man (the younger
# the better); among grown men, the least able.
ulus_puppet_pliancy = {
	value = 300
	if = {
		limit = { is_adult = yes }
		subtract = total_abilities
	}
	else = {
		add = 100
		subtract = age_in_years
	}
}

# The regent's standing for the panel and the death poll: his state's
# kingmaker score with the HEIR's abilities in place of the ruler's.
# COUNTRY scope (the regent's state).
ulus_regent_heir_score = {
	# The kingmaker score (script_values/04, ulus_kingmaker_score) as it
	# will read on the day the heir rules: the house's share and its
	# patron stay, the ruler's skills become the heir's, and what was
	# the dead man's own (his renown, his wife of the line) is gone.
	# Until 2026-11-14 this was the first draft's formula (military x2,
	# share x250 capped at 60) and the panel disagreed with the poll.
	value = 0
	if = {
		limit = { has_heir = yes }
		add = {
			value = heir.mil
			multiply = 0.5
		}
		add = {
			value = heir.adm
			multiply = 0.2
		}
	}
	if = {
		limit = { has_variable = ulus_amir_share }
		add = {
			value = var:ulus_amir_share
			multiply = 400
		}
	}
	if = {
		limit = { ulus_has_patron_of_the_line = yes }
		add = ulus_borjigin_patron_bonus
	}
}

# =====================================================================
# 12. ASKING FOR A PLEDGE (generic_actions/amirs_rising_actions.txt,
#     ulus_ask_for_pledge). COUNTRY scope: the house that is ASKED;
#     scope:actor is the house asking. An AI house pledges at 0 or more.
#     Each term is shown to the player in the action's tooltip.
# =====================================================================
ulus_pledge_acceptance = {
	# A tümen does not give its weight away for nothing.
	value = -25
	# Already pledged elsewhere: its word is given, and it is moved only
	# if it thinks better of us than of the house it stands behind.
	if = {
		limit = { has_variable = ulus_pledged_to }
		add = -30
		if = {
			limit = { exists = scope:ulus_current_patron }
			add = {
				value = "opinion(scope:actor)"
				subtract = "opinion(scope:ulus_current_patron)"
				multiply = 0.167
			}
		}
	}
	# What they think of us: a sixth of their opinion (opinion runs to ±200).
	add = {
		value = "opinion(scope:actor)"
		multiply = 0.167
	}
	# Kin by marriage.
	if = {
		limit = { has_royal_marriage_with = scope:actor }
		add = 25
	}
	if = {
		limit = { is_allied_with = { target = scope:actor } }
		add = 15
	}
	if = {
		limit = { is_rival_of = scope:actor }
		add = -100
	}
	# The gap in weight. A house backs a stronger one, never a weaker.
	if = {
		limit = {
			has_variable = ulus_weight
			scope:actor = { has_variable = ulus_weight }
		}
		if = {
			limit = {
				scope:actor.var:ulus_weight >= {
					value = var:ulus_weight
					multiply = 2
				}
			}
			add = 30
		}
		else_if = {
			limit = {
				scope:actor.var:ulus_weight >= {
					value = var:ulus_weight
					multiply = 1.25
				}
			}
			add = 15
		}
		# The lighter house asking the heavier: up to -100, by how much
		# lighter (equal weight 0, half the weight -50, a tenth -90).
		else_if = {
			limit = {
				scope:actor.var:ulus_weight < var:ulus_weight
				var:ulus_weight > 0
			}
			add = {
				value = scope:actor.var:ulus_weight
				divide = var:ulus_weight
				subtract = 1
				multiply = 100
				min = -100
				max = 0
			}
		}
	}
	# A house that could take the khan itself does not stand behind
	# another: up to -66, by its kingmaker score against the best amir's
	# (the likeliest kingmaker of all -66, the third of the three less).
	if = {
		limit = {
			is_target_in_global_variable_list = {
				name = ulus_kingmakers
				target = this
			}
			has_variable = ulus_kingmaker_score
		}
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= {
					has_variable = ulus_top_kingmaker_score
					var:ulus_top_kingmaker_score > 0
				}
			}
			add = {
				value = var:ulus_kingmaker_score
				divide = international_organization:ulus_chaghatay.var:ulus_top_kingmaker_score
				max = 1
				min = 0
				multiply = -66
			}
		}
		else = {
			add = -66
		}
	}
	# Standing behind a likely winner.
	if = {
		limit = {
			scope:actor = {
				is_target_in_global_variable_list = {
					name = ulus_kingmakers
					target = this
				}
			}
		}
		add = 10
	}
}

# =====================================================================
# 16. OUTSIDE SUPPORT
#     COUNTRY scope: the OUTSIDER; scope:ulus_asking_house is the amir.
#     An AI outsider backs him at 0 or more. An outsider backs a
#     contender to weaken the khan, so he wants one who can make
#     trouble and cannot win without him: not too strong, not too weak.
#     The truly strong go by alliance, which binds both ways.
# =====================================================================
ulus_outside_support_acceptance = {
	value = -20
	add = {
		value = "opinion(scope:ulus_asking_house)"
		multiply = 0.167
	}
	if = {
		limit = { is_rival_of = scope:ulus_asking_house }
		add = -100
	}
	# His quarrel with the khan's seat.
	if = {
		limit = {
			international_organization:ulus_chaghatay ?= { international_organization_has_leader = yes }
			is_rival_of = international_organization:ulus_chaghatay.leader_country
		}
		add = 30
	}
	if = {
		limit = {
			international_organization:ulus_chaghatay ?= { international_organization_has_leader = yes }
			is_neighbor_of = international_organization:ulus_chaghatay.leader_country
		}
		add = 10
	}
	if = {
		limit = {
			international_organization:ulus_chaghatay ?= { international_organization_has_leader = yes }
			is_allied_with = { target = international_organization:ulus_chaghatay.leader_country }
		}
		add = -100
	}
	# The contender's standing against the best amir's.
	if = {
		limit = {
			scope:ulus_asking_house = { has_variable = ulus_kingmaker_score }
			international_organization:ulus_chaghatay ?= {
				has_variable = ulus_top_kingmaker_score
				var:ulus_top_kingmaker_score > 0
			}
		}
		if = {
			limit = {
				scope:ulus_asking_house.var:ulus_kingmaker_score >= {
					value = international_organization:ulus_chaghatay.var:ulus_top_kingmaker_score
					multiply = 0.9
				}
			}
			add = -10		# strong enough to win alone, and to forget who helped
		}
		else_if = {
			limit = {
				scope:ulus_asking_house.var:ulus_kingmaker_score >= {
					value = international_organization:ulus_chaghatay.var:ulus_top_kingmaker_score
					multiply = 0.5
				}
			}
			add = 25		# can make trouble, cannot win without us
		}
		else = {
			add = -30		# silver thrown into the grass
		}
	}
	if = {
		limit = { ai_personality = ai_personality:ai_opportunistic }
		add = 15
	}
	if = {
		limit = { ai_personality = ai_personality:ai_expansionist }
		add = 10
	}
	if = {
		limit = { ai_personality = ai_personality:ai_isolationist }
		add = -30
