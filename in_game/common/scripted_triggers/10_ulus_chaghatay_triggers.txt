# Ulus Chaghatay — membership tests. COUNTRY scope.
#
# The exists guard is there because the organisation can end (Bulqaq,
# or the Ṣāḥib-qirān folding it into his realm, in the later phases);
# international_organization:ulus_chaghatay is then an unset link.
#
# Read by: Erānzamīn's join rules and is_eranzamin_contender (the amir
# domains sit in Transoxiana and Tukharistan, inside khorasan_region,
# and are no longer anyone's subjects — without this they would qualify
# as contenders for the Ilkhan's throne), the horde succession poll, and
# the ulus ledger.

is_ulus_chaghatay_member = {
	exists = international_organization:ulus_chaghatay
	is_member_of_international_organization = international_organization:ulus_chaghatay
}

# A junior member: in the ulus, and not the seat of the khan.
is_ulus_chaghatay_amir = {
	is_ulus_chaghatay_member = yes
	NOT = { is_leader_of_international_organization = international_organization:ulus_chaghatay }
	NOT = { ulus_marked_client = yes }
}

# Any member that is not the seat: amirs and client kings alike.
is_ulus_chaghatay_junior = {
	is_ulus_chaghatay_member = yes
	NOT = { is_leader_of_international_organization = international_organization:ulus_chaghatay }
}

# A CLIENT KING of the ulus (special status ulus_client): the Shahs of
# Badakhshan, the lords of Karategin, and any settled house the ulus
# raises on a broken tümen's ground. He sends the qubchur and may pledge
# to a house, but he is no amir: he has no hope of the khan, is not
# reckoned among the amirs, and does not raid or rise.
# The mark is the country variable ulus_client; the two 1337 tags are
# named as well so that the test holds before on_game_start has run.
ulus_marked_client = {
	OR = {
		has_variable = ulus_client
		tag = BKH
		tag = KRG
	}
}
is_ulus_chaghatay_client = {
	is_ulus_chaghatay_member = yes
	NOT = { is_leader_of_international_organization = international_organization:ulus_chaghatay }
	ulus_marked_client = yes
}

# The seat: the khan's own realm, or in a regency the regent's.
is_ulus_chaghatay_leader = {
	is_ulus_chaghatay_member = yes
	is_leader_of_international_organization = international_organization:ulus_chaghatay
}

# The 1337 roll of the ulus BY TAG. Membership is seeded after countries

# In the organisation's can_declare_war (scope:attacker, scope:defender):
# the attacker holds one of the ulus's own casus belli on the defender.
ulus_attacker_holds_ulus_cb = {
	exists = scope:attacker
	exists = scope:defender
	scope:attacker = {
		OR = {
			has_casus_belli_of_type_on = {
				target = scope:defender
				type = casus_belli:cb_rise_against_the_khan
			}
			has_casus_belli_of_type_on = {
				target = scope:defender
				type = casus_belli:cb_take_the_khan
			}
			has_casus_belli_of_type_on = {
				target = scope:defender
				type = casus_belli:cb_restore_the_ulus
			}
			has_casus_belli_of_type_on = {
				target = scope:defender
				type = casus_belli:cb_tumen_feud
			}
			has_casus_belli_of_type_on = {
				target = scope:defender
				type = casus_belli:cb_break_the_tumens
			}
			has_casus_belli_of_type_on = {
				target = scope:defender
				type = casus_belli:cb_sahib_qiran_submission
			}
		}
	}
}

# A country whose capital stands on a tümen's pasture (marked at game
# start, on_action/36) while the ulus stands, or the Ṣāḥib-qirān's own
# realm. Neither belongs in the Ilkhanid succession: a horde that
# collapsed leaves a settled Tajik country on its ground, and base's
# culture count made that a Persianate sultanate which then joined
# Erānzamīn (author's test, Suldus, 2026-11-14); the Ṣāḥib-qirān's
# state was picked up the same way at its release.
ulus_ground_or_sahib_qiran = {
	OR = {
		tag = TIM
		has_variable = sq_fortune
		AND = {
			exists = international_organization:ulus_chaghatay
			capital ?= { has_variable = ulus_tumen_pasture }
		}
	}
}


# COUNTRY scope: a house of the ulus with an ally or a backer outside it
# whose ruler is of Chinggis's line (scripted_triggers/11).
ulus_has_patron_of_the_line = {
	exists = international_organization:ulus_chaghatay
	OR = {
		any_ally = {
			NOT = { is_member_of_international_organization = international_organization:ulus_chaghatay }
			ruler ?= { is_of_chinggis_line = yes }
		}
		var:ulus_outside_backer ?= {
			NOT = { is_member_of_international_organization = international_organization:ulus_chaghatay }
			ruler ?= { is_of_chinggis_line = yes }
		}
	}
}

# COUNTRY scope: a house that has put its weight behind another (a
# pledge to an amir or to the khan, or as a subject of a house): it is
# out of the reckoning for the khan itself.
ulus_has_chosen_a_side = {
	OR = {
		has_variable = ulus_pledged_to
		has_variable = ulus_pledged_to_khan
		has_variable = ulus_weight_lent_all
	}
}


# ---------------------------------------------------------------------
# WARS INSIDE THE ULUS, for the organisation's join/call rules
# (international_organizations/ulus_chaghatay.txt). root = the
# organisation, scope:actor = the caller, scope:recipient = the callee.
#
# These replace tests on scope:target, which the engine does not supply
# to an organisation with no target of its own: with it missing, "the
# enemy is not a member" read as true for everyone and "pledged to the
# enemy" never matched, so every house rode for the khan against a
# rising, the riser's own pledged tümens included (author's test,
# Jalayirs pledged to the Yasa'uri, 2026-11-14).
# A house is "against" the caller if it is at war with him or holds one
# of the ulus's own casus belli on him (which covers the days between
# the rising being proclaimed and the war being declared).
# ---------------------------------------------------------------------

# The caller is fighting, or about to fight, a house of the ulus.
ulus_caller_fights_a_house = {
	any_international_organization_member = {
		NOT = { this = scope:actor }
		OR = {
				is_at_war_with = scope:actor
				has_casus_belli_of_type_on = {
					target = scope:actor
					type = casus_belli:cb_rise_against_the_khan
				}
				has_casus_belli_of_type_on = {
					target = scope:actor
					type = casus_belli:cb_take_the_khan
				}
				has_casus_belli_of_type_on = {
					target = scope:actor
					type = casus_belli:cb_tumen_feud
				}
				has_casus_belli_of_type_on = {
					target = scope:actor
					type = casus_belli:cb_restore_the_ulus
				}
			}
	}
}

# The callee has put its tümen behind a house that is against the caller.
ulus_callee_is_pledged_against_caller = {
	scope:recipient = {
		has_variable = ulus_pledged_to
		var:ulus_pledged_to = {
			OR = {
				is_at_war_with = scope:actor
				has_casus_belli_of_type_on = {
					target = scope:actor
					type = casus_belli:cb_rise_against_the_khan
				}
				has_casus_belli_of_type_on = {
					target = scope:actor
					type = casus_belli:cb_take_the_khan
				}
				has_casus_belli_of_type_on = {
					target = scope:actor
					type = casus_belli:cb_tumen_feud
				}
				has_casus_belli_of_type_on = {
					target = scope:actor
					type = casus_belli:cb_restore_the_ulus
				}
			}
		}
	}
}

# The callee stands with the caller: pledged to him, his subject, or
# (the caller being the seat) pledged to the khan.
ulus_callee_stands_with_caller = {
	scope:recipient = {
		OR = {
			var:ulus_pledged_to ?= scope:actor
			is_subject_of = scope:actor
			AND = {
				has_variable = ulus_pledged_to_khan
				scope:actor = { is_leader_of_international_organization = root }
			}
		}
	}
}

# In a war inside the ulus nobody rides who has not taken a side; in any
# other war the ordinary rules hold.
ulus_callee_may_ride_for_caller = {
	NOT = { ulus_callee_is_pledged_against_caller = yes }
	OR = {
		NOT = { ulus_caller_fights_a_house = yes }
		ulus_callee_stands_with_caller = yes
	}
}


# LOCATION scope: a place of a broken tümen that is still loose: held by
# the broken house itself, or by a country whose capital lies inside
# that house's old ground (one of the pieces that came out of it).
# Needs scope:ulus_broken_seat_loc and scope:ulus_broken_house
# (scripted_effects/22_ulus_client_effects.txt).
ulus_is_loose_broken_ground = {
	exists = owner
	owner = {
		OR = {
			this = scope:ulus_broken_house
			AND = {
				NOT = { is_ulus_chaghatay_member = yes }
				capital ?= { var:ulus_ground_seat ?= scope:ulus_broken_seat_loc }
			}
		}
	}
}
