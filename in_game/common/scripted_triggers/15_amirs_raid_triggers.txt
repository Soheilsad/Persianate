# The tümens' raids — tests.

# COUNTRY scope (the realm to be raided); scope:actor is the raider.
# A raid on a realm of the raider's own faith needs a cause, or it costs
# him. The causes a chief could give: the realm is a patron of his rivals
# in the ulus (it backs an amir), it is his declared rival, or it is at
# war with the seat of the ulus.
ulus_raid_has_pretext = {
	OR = {
		has_variable = ulus_backs_amir
		is_rival_of = scope:actor
		AND = {
			international_organization:ulus_chaghatay ?= { international_organization_has_leader = yes }
			is_at_war_with = international_organization:ulus_chaghatay.leader_country
		}
	}
}

# COUNTRY scope (the realm to be raided); scope:actor is the raider.
# Muslim on Muslim: the case the law speaks to. Of whatever rite: a
# Sunni chief raiding a Shīʿī town was not raiding unbelievers, and
# until 2026-11-14 that case fell between this test and the ghaza and
# cost nothing. A chief who is not a Muslim has no such law on him and
# meets neither test.
ulus_raid_is_on_own_faith = {
	religion.group = religion_group:muslim
	scope:actor = { religion.group = religion_group:muslim }
}

# COUNTRY scope (the realm to be raided): unbelievers, to a Muslim raider.
ulus_raid_is_ghaza = {
	scope:actor = { religion.group = religion_group:muslim }
	NOT = { religion.group = religion_group:muslim }
}
