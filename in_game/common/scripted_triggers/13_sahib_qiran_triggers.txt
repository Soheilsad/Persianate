# Outcome tests of the Rise of Timur situation as this mod runs it
# (common/situations/rise_of_timur.txt). Any scope; all read c:TIM.

# THE CONJUNCTION HOLDS. He has made good sq_ambitions_needed declared
# conquest ambitions (ANY regions: sq_ambition_check in scripted_effects/
# 18_sahib_qiran_fortune_effects.txt counts them), still owns the seat
# his rule began from, and his fortune is not in the low band.
# Nothing sends him west. Temür took Iran first because the Iran of the
# 1380s was rich, near and without a master; he sacked Delhi in 1398 and
# died on the road to China in 1405. A Ṣāḥib-qirān of another decade,
# or one whose Iran is already united, chooses otherwise.
sq_conjunction_holds = {
	custom_tooltip = {
		text = sq_conjunction_holds_tt
		c:TIM ?= {
			has_variable = sq_ambitions_count
			var:sq_ambitions_count >= sq_ambitions_needed
			var:sq_seat ?= { owner ?= c:TIM }
			NOT = { has_variable = sq_band_low }
		}
	}
}

# THE AMIRS RIDE AWAY. The reckoning has happened: his fortune lay
# empty sq_empty_years and the sworn amirs left (sq_fortune_reckoning
# sets sq_reckoned on the realm).
sq_amirs_rode_away = {
	custom_tooltip = {
		text = sq_amirs_rode_away_tt
		c:TIM ?= { has_variable = sq_reckoned }
	}
}

# THE SONS DIVIDE. The founder is dead and an heir's fortune has stayed
# in the low band sq_sons_divide_months running (counted by
# sq_fortune_tick as sq_heir_low_months), or the realm has neither
# ruler nor heir.
sq_sons_divide = {
	custom_tooltip = {
		text = sq_sons_divide_tt
		c:TIM ?= {
			OR = {
				AND = {
					has_variable = sq_heir_low_months
					var:sq_heir_low_months >= sq_sons_divide_months
				}
				AND = {
					has_heir = no
					has_ruler = no
				}
			}
		}
	}
}

# ---------------------------------------------------------------------
# The AI's judgement (script_values/06_sahib_qiran_ai_values.txt).
# ---------------------------------------------------------------------

# COUNTRY scope: a realm that has been bled. Its muster counts for half.
# (The author's case: a united Iran that has just lost a ruinous war.)
sq_is_wounded_realm = {
	OR = {
		war_exhaustion >= sq_ai_wounded_exhaustion
		manpower_percentage < sq_ai_wounded_manpower
	}
}

# COUNTRY scope, with scope:actor: even counted so, it is much stronger
# than he is.
sq_outmatches_actor = {
	sq_effective_strength > {
		value = scope:actor.military_strength
		multiply = sq_ai_outmatched_ratio
	}
}

# COUNTRY scope (the Ṣāḥib-qirān): he or one of his subjects borders
# scope:target.
sq_reaches_target = {
	OR = {
		is_neighbor_of = scope:target
		any_subject = { is_neighbor_of = scope:target }
	}
}

# COUNTRY scope (the Ṣāḥib-qirān): a sworn amir's loyalty is failing.
sq_amirs_wavering = {
	any_subject = {
		has_variable = sq_sworn_amir
		subject_loyalty < 50
	}
}

# COUNTRY scope (a neighbour); $HIM$ = the Ṣāḥib-qirān. A realm clearly
# stronger than he is, at its strength, that he has not named: one to
# send gifts to, not riders. A wounded giant is prey, not a friend.
sq_worth_courting = {
	is_rebel_country = no
	is_subject = no
	NOT = { is_rival_of = $HIM$ }
	NOT = { is_allied_with = { target = $HIM$ } }
	NOT = { has_variable = ulus_bulqaq_orphan }
	NOT = { sq_is_wounded_realm = yes }
	military_strength > {
		value = $HIM$.military_strength
		multiply = sq_ai_court_ratio
	}
	NOT = {
		AND = {
			is_situation_active = situation:rise_of_timur
			situation:rise_of_timur = { has_variable = rot_conquest_ambition }
			capital ?= { region = situation:rise_of_timur.var:rot_conquest_ambition }
		}
	}
}
