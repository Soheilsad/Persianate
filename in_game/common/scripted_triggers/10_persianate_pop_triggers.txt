is_diwan_pop = {
	AND = {
		OR = {
			culture = { has_culture_group = culture_group:iranian_group }
			culture = { has_culture_group = culture_group:arabic_group }
			# The indigenous scribal class of the sultanate world — the
			# Indian Muslim clerks whose rise into the diwan is what
			# Baranī's complaint literature is complaining about. (The
			# Khurasani émigrés live at the CHARACTER level: estate
			# seeding hardcodes farsi diwan characters.)
			AND = {
				culture = { has_culture_group = culture_group:indian_group }
				religion.group = religion_group:muslim
			}

			AND = {
				culture = { has_culture_group = culture_group:turkic_group }
				location = {
					owner ?= { has_reform = government_reform:sultanate_of_rum }
				}
				religion.group = religion_group:muslim
			}
		}
		# Literacy gate, waived during the start year. HYPOTHESIS under test:
		# the diwan estate was missing on day one and appeared at the first
		# yearly re-sort even after the chancery check was made day-one
		# safe (has_chancery_apparatus), which leaves pop_literacy as the
		# value not yet initialised when setup pops are first sorted. If
		# the diwan exists on 1 April 1337 with this, the cause is
		# confirmed; if not, remove this OR and look at the location links.
		OR = {
			pop_literacy > 20
			current_year < 1338
		}
		# Apparatus flag only — the chancery makes the pop class, not the
		# government form (a beylik sultanate has no diwan to join).
		# Pop's own owner link, as is_amir_pop and every base estate rung
		# use it. This went through `location = { owner ?= ... }`; no base
		# rung reads a pop's location link, and the amirs (owner link)
		# existed on day one while the diwan did not.
		owner ?= {
			has_chancery_apparatus = yes
		}


		OR = {
			# Same start-year waiver as literacy, for the same reason:
			# the urban test reads the pop's location link, the other
			# value that may not be ready when setup pops are sorted.
			current_year < 1338
			location ?= {
				is_province_capital = yes
			}
			location ?= {
				is_city = yes
			}
				
			location ?= {
				percentage_pop_type_in_location:burghers > 0.05
			}
		}
	}
}

is_amir_pop = {
	# Apparatus flag only (was gov-type OR): NOTE this removes the
	# commonwealth arm — the vow-state's military class is the dervish
	# corps, not amirs (08_martyrs_auto_modifiers deliberately withholds
	# amirs_base_values from it). Amir pops return to any polity the
	# moment it acquires the chancery.
	owner ?= {
		has_chancery_apparatus = yes
	}
	# The umara is a closed MUSLIM service order in 1337 — revocable
	# iqṭāʿ rank, not hereditary right (Barani's khan–malik–amir ladder;
	# conversion was the entry ticket: Harihara and Bukka passed through
	# it at Kampili). Non-Muslim hereditary lords — rajas, meliks, Tengri
	# noyans — fall through to the NOBLES estate: immunity holders by
	# descent, the feudal-rights pole of the tripartite system.
	#
	# BUT the closure is a decision, not a law of nature: the Open
	# Muster Rolls privilege (amirs_open_muster_rolls) is the historical
	# opening — Maratha silahdars in Deccani service, Shahji Bhonsle's
	# jagirs, the Rajput mansabdars. Grant it and the corps takes any
	# creed.
	OR = {
		religion.group = religion_group:muslim
		owner ?= {
			has_estate_privilege = estate_privilege:amirs_open_muster_rolls
		}
	}
}

is_burgher_diwan_pop = {
	AND = {
		culture ?= {
			language = language:persian_language
		}
		owner = {
			# Requires highly capitalized economy to allow merchants into the high bureaucracy
			societal_value:capital_economy_vs_traditional_economy < -75
		}
		# Apparatus flag only — the chancery makes the pop class, not the
		# government form (a beylik sultanate has no diwan to join).
		owner ?= {
			has_chancery_apparatus = yes
		}
		location ?= {
			OR = {
				location_rank = location_rank:city
			}
		}
		pop_literacy > 60
	}
}

# Sufyānī notables — urban-aristocratic faction in any Commonwealth of Martyrs
# polity. Lay notables, malik families, urban patricians. The constituency is
# nobles in cities; rural nobles (where they exist at all under this regime)
# don't qualify because Sufyānī identity is bound to the urban civic order.
is_sufyani_pop = {
	owner ?= {
		government_type = government_type:commonwealth_of_martyrs
	}
	OR = {
		location ?= {
			is_province_capital = yes
		}
		location ?= {
			is_city = yes
		}
		location ?= {
			location_rank = location_rank:city
		}
		
	}
}

# Shaykhī dervishes — the armed religious order. Clergy in any Commonwealth of
# Martyrs polity qualify (khānaqāh dervishes), as do soldiers (the dervish
# corps proper). Together they form the polity's mahdist-militant base.
is_shaykhi_pop = {
	owner ?= {
		government_type = government_type:commonwealth_of_martyrs
	}
}

# The chancery apparatus, COUNTRY scope. The design keys everything on
# the has_nizam_ul_mulk_chancery flag rather than on a reform, and that
# stays the rule. But the flag is a computed country modifier, and on the
# first day of a game pops are sorted into estates before country
# modifiers are computed: no country had the flag yet, no pop qualified
# as diwan or amir, and the two estates did not exist until the next
# yearly re-sort. The reforms that CARRY the flag are plain setup data
# and are readable on day one, so they are accepted directly as well.
# Keep this list in step with every reform that sets the flag
# (grep has_nizam_ul_mulk_chancery in common/government_reforms).
has_chancery_apparatus = {
	OR = {
		modifier:has_nizam_ul_mulk_chancery = yes
		has_reform = government_reform:persianate_bureaucracy_reform
		has_reform = government_reform:timurid_empire
		has_reform = government_reform:niyaba_amma_charter_reform
		has_reform = government_reform:mamluk_government
		has_reform = government_reform:indian_sultanate_reform
		has_reform = government_reform:yemeni_chancery
	}
}
