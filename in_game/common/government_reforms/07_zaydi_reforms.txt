# =====================================================================
# THE ZAYDI IMAMATE — a government reform on base theocracy.
#
# Why a reform and not a type: the imamate is a theocracy in the plain
# sense (devotion as its currency, clergy as the default estate, an
# elective head chosen for religious qualification). What is Zaydi about
# it is the rule for choosing that head (heir_selections/zaydi_imamate.txt)
# and the tribal base — Hashid, Bakil, Khawlan under their shaykhs, the
# imam's only army. No iqta, no chancery: when an imamate takes the
# lowlands it inherits the lowland apparatus (the Sharaf al-Din and
# Qasimid cases), and that is a later reform, not this one.
# =====================================================================

zaydi_imamate = {
	age = age_1_traditions
	government = theocracy
	potential = {
		government_type = government_type:theocracy
		religion = religion:shia
	}
	locked = {
		always = yes
	}
	country_modifier = {
		tolerance_own = 1
		monthly_devotion = 0.05
		monthly_religious_influence = 0.10
		global_tribes_estate_power = 0.5
		tribes_estate_levy_size = 0.25
		peasants_estate_levy_size = 0.33 
		land_morale_modifier = 0.20
		tribes_estate_can_participate_in_parliament = yes
		global_defensive = 0.15
		global_levy_size_modifier = 0.15
		clergy_estate_target_satisfaction = small_permanent_target_satisfaction
		government_reform_slots = 1
	}

	years = 0
}


# =====================================================================
# THE SULTANIZED IMAMATE — stage two, the marker on the new sultanate.
#
# Applied by imams_house_transform (scripted_effects/11_zaydi_effects.txt)
# alongside yemeni_chancery, from zaydi_chancery.1 or proclaim_imams_house
# — there is no intermediate reform: the imamate captures the chancery and
# either becomes a sultanate on the spot (tenet already dynastic) or holds
# the diwan (variable) until the community is talked into it. The ruler
# is still Imam of his imamate (the IO persists, the sworn still pay
# and march), but the office is hereditary now and the state is a
# Persianate sultanate. The Qasimid fact, with the jurists' grievance
# carried as a 25-year modifier (sultanized_imamate_grievance) and the
# tribal base kept heavy: a sultan whose army is Hashid and Bakil.
# =====================================================================

sultanized_imamate = {
	age = age_1_traditions
	government = persianate_sultanate
	# Effect-only: imams_house_transform sets imams_house_proclaimed first,
	# so this never appears as a menu choice.
	potential = {
		government_type = government_type:persianate_sultanate
		has_variable = imams_house_proclaimed
	}
	locked = {
		always = yes
	}
	country_modifier = {
		monthly_religious_influence = 0.05
		tolerance_own = 1
		land_morale_modifier = 0.05
		global_tribes_estate_power = 0.25
		tribes_estate_levy_size = 0.25
	}

	years = 0
}
