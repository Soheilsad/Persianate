# =====================================================================
# PROCLAIM THE IMAM'S HOUSE — stage two of the road to a sultanate.
#
# Requires stage one (the imam holds a captured diwan), a son to hand it to, and — the
# load-bearing gate — the imamate's own tenet already turned dynastic:
# the community must be talked into the Imam's House before the state
# can become one. Then the theocracy becomes a Persianate sultanate on
# the user's yemeni_chancery reform with the sultanized_imamate marker,
# the elective is replaced by favoured-son succession, and the jurists
# carry their grievance for a generation. The IO survives: the sultan is
# still Imam, the sworn still pay and march. al-Mutahhar b. Sharaf
# al-Din, and the Qasimids after al-Mutawakkil Isma'il.
#
# Shape copied from create_autocephalous_patriarchate (type = religious,
# religion picker as the recipient).
# =====================================================================

proclaim_imams_house = {
	type = religious

	sound = UI_action_religion_generic

	ai_tick = monthly
	ai_tick_frequency = 6
	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			government_type = government_type:theocracy
			has_reform = government_reform:zaydi_imamate
			has_variable = holds_the_diwan
		}
	}

	allow = {
		scope:actor = {
			at_war = no
			# a son to hand the house to — not "an heir": under the elective the
			# heir is whichever sayyid scores best, which is the point
			custom_tooltip = {
				text = proclaim_imams_house_has_son_tt
				ruler ?= {
					any_child = {
						is_alive = yes
						is_female = no
					}
				}
			}
			custom_tooltip = {
				text = proclaim_imams_house_dynastic_tenet_tt
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:imamate_living
					leader_country ?= scope:actor
					international_organization_has_policy = policy:dynastic_imamate
				}
			}
		}
	}

	price = price:proclaim_imams_house

	select_trigger = {
		looking_for_a = religion
		source = actor
		target_flag = recipient
		name = "choose_religion"
		column = {
			data = name
		}
	}

	effect = {
		scope:actor = {
			custom_tooltip = {
				text = proclaim_imams_house_effect_tt
				imams_house_transform = yes
			}
		}
	}

	ai_will_do = {
		add = {
			desc = "VOTE_BASE_VALUE"
			value = -10
		}
		if = {
			limit = {
				scope:actor = {
					ruler ?= { character_age >= 50 }
				}
			}
			add = {
				desc = "THE_IMAM_IS_OLD"
				value = 30
			}
		}
	}
}
