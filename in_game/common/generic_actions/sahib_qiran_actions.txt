# ============================================================
# THE ṢĀḤIB-QIRĀN'S ACTIONS.
#
# type = situation, selected on base's Rise of Timur, so the button
# appears among that panel's Main Actions (the form of base
# generic_actions/rise_of_timur.txt). Only while that situation is
# active: a Ṣāḥib-qirān whose realm was not released as TIM (Iran
# already had a master) has the fortune but not this button.
#
# Price: common/prices/05_sahib_qiran_prices.txt.
# AI list: common/generic_action_ai_lists/sahib_qiran_list.txt.
# ============================================================

# ------------------------------------------------------------
# DIVIDE THE PLUNDER. Gold into his fortune. After each campaign Temür
# divided what had been taken among the amirs and the tümens, by rank,
# at a feast; it is what the chronicles of his reign count beside the
# victories. A year's income for sq_fortune_largesse points, once a
# year.
# ------------------------------------------------------------
sq_distribute_plunder = {
	type = situation
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = sq_distribute_plunder
		years = 1
	}

	price = price:sq_distribute_plunder_price

	potential = {
		scope:actor = {
			has_variable = sq_fortune
		}
	}

	allow = {
		scope:actor = {
			is_rebel_country = no
		}
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:rise_of_timur = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			if = {
				limit = { country_has_estate = estate_type:nobles_estate }
				add_estate_satisfaction = { type = estate_type:nobles_estate value = 0.05 }
			}
			# The amirs feel it directly: his subjects' loyalty, for a time
			# (main_menu/common/static_modifiers/
			# 17_sahib_qiran_fortune_modifiers.txt).
			add_country_modifier = {
				modifier = sq_plunder_divided_mod
				years = 3
				mode = add_and_extend
			}
			custom_tooltip = sq_distribute_plunder_tt
			hidden_effect = {
				sq_change_fortune = { AMOUNT = sq_fortune_largesse }
			}
		}
	}

	# The AI pays when his fortune is under the high line.
	# The AI opens the chest when the fortune needs it and the treasury
	# can bear it (script_values/06_sahib_qiran_ai_values.txt):
	#   never        at or over the high line with the amirs steady
	#   the gap      1 point per point of fortune under the high line
	#   in the low   +60: the amirs are a season from leaving
	#     band
	#   wavering     +40 if a sworn amir's loyalty is under 50
	#     amirs
	#   at peace     +20 past three years of it: no victories are coming
	#   the purse    -80 if paying leaves him under 18 months' income
	#                (waived in the low band: he has no choice)
	#   in debt      never
	ai_will_do = {
		add = 0
		if = {
			limit = {
				scope:actor = {
					has_variable = sq_fortune
					gold > 0
					OR = {
						var:sq_fortune < sq_fortune_high_line
						sq_amirs_wavering = yes
					}
				}
			}
			add = {
				value = sq_fortune_high_line
				subtract = scope:actor.var:sq_fortune
				min = 0
			}
			if = {
				limit = { scope:actor = { has_variable = sq_band_low } }
				add = 60
			}
			if = {
				limit = { scope:actor = { sq_amirs_wavering = yes } }
				add = 40
			}
			if = {
				limit = {
					scope:actor = {
						has_variable = sq_peace_months
						var:sq_peace_months > sq_fortune_idle_months
					}
				}
				add = 20
			}
			if = {
				limit = {
					scope:actor = {
						NOT = { has_variable = sq_band_low }
						gold < sq_plunder_reserve_gold
					}
				}
				subtract = 80
			}
		}
	}
}
