# ============================================================
# The Matter of Tork — disposition claim buttons
#
# Five buttons on the disaster panel. Each is one answer to
# "what is the Türk's place now?" — and the ONLY door to that
# answer in the whole mod. One claim per realm: any disposition
# locks the others (matter_tork_disposition_chosen, set by the
# claim events).
#
#   claim_iranization        — A. dissolved into us   (~20y; needs amirs at the diwan)
#   matter_tork_release_march — B. kept at the edge   (zone earned by conquest)
#   claim_murid_bond         — C. our instrument      (mysticism ≤ -50)
#   claim_expulsion          — D. removed from society (~10y process)
#   claim_eradication        — E. destroyed as a class (civil war + 50y)
#
# Plus the approach layer: matter_tork_change_approach (the council),
# and the per-approach actions (purchase ghulams / sponsor qanat /
# invite shaykhs / release march).
#
# The claim buttons COST nothing at the button layer — the costs
# live in the claim events (the farr/ghazi lesson about claims
# cascading does not apply here: the dispositions are mutually
# exclusive, so there is no cascade to prevent).
#
# Design: docs/design/matter_of_tork_design.md §4
# ============================================================

matter_tork_change_approach = {
	type = disaster
	show_message = no

	# The AI picks its approach once, at the council; it does not
	# second-guess itself monthly.
	ai_tick = never
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	# A council, not a toggle: re-convening takes two years.
	cooldown = {
		type = matter_tork_change_approach
		years = 2
	}

	allow = {
		scope:actor = {
			has_variable = matter_of_tork_fired
			NOT = { has_variable = matter_tork_disposition_chosen }
		}
	}

	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = { data = name }
		visible = {
			disaster_type = disaster_type:matter_of_tork
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			# Re-convene the council. The horizon clock does not reset;
			# earned gains persist; only the ambient swaps.
			trigger_event_non_silently = matter_of_tork.2
		}
	}

	ai_will_do = {
		add = 0
	}
}

# ---Approach 1: Purchase Ghulam Levies ---------------------------------
# One corps tier per purchase (I → II → III). Needs an administrator
# (a ghulam administration privilege or the reform), an implementer
# (the fostering households privilege), and — past tier I — the
# physical apparatus: tier II needs >= 1 ghulam barracks, tier III
# needs >= 3. [verify: any_owned_location count = N syntax]

matter_tork_purchase_ghulams = {
	type = disaster
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 3
	automation_tick = never
	automation_tick_frequency = 12

	price = price:matter_purchase_ghulams

	# NOTE: potential does NOT hide actions (playtest-confirmed) —
	# conditions live in allow for clear tooltips. v1.5: per-approach
	# disaster variants for true visibility control (decision 37).
	cooldown = {
		type = matter_tork_purchase_ghulams
		years = 2
	}

	allow = {
		scope:actor = {
			has_variable = matter_tork_approach
			var:matter_tork_approach = 1
			# The administrator.
			OR = {
				has_estate_privilege = estate_privilege:diwan_ghulam_administration
				has_estate_privilege = estate_privilege:amirs_ghulam_administration
				has_reform = government_reform:ghulam_system_reform
			}
			# The implementer.
			has_estate_privilege = estate_privilege:ghulam_fostering_households
			# Tier ladder gated on BARRACKS LEVELS (decision 38): the
			# corps needs its physical plant before it can grow.
			# [verify: total_effective_building_levels at country scope]
			custom_tooltip = {
				text = matter_tork_barracks_ladder_tt
				OR = {
					AND = {
						NOT = { has_variable = matter_ghulam_corps_tier }
						total_effective_building_levels:ghulam_barracks >= 1
					}
					AND = {
						var:matter_ghulam_corps_tier = 1
						total_effective_building_levels:ghulam_barracks >= 3
						has_variable = matter_slave_route
					}
					AND = {
						var:matter_ghulam_corps_tier = 2
						total_effective_building_levels:ghulam_barracks >= 6
						has_variable = matter_slave_route
					}
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = { data = name }
		visible = {
			disaster_type = disaster_type:matter_of_tork
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			capital.market = {
				add_temporary_demand = { type = demand:ghulam_purchase_demand years = 5 scale_with_pop = 0.01 }

			}
			# The purchase ripples through the slave markets.
			# The boys arrive where the barracks stand — PURCHASED
			# Caucasians, not enslaved locals (playtest fix 5: the
			# old split_pop converted Tazik peasants into slaves).
			# Grow existing slave pops only; if none exist yet, the
			# barracks employment + market demand do the attracting.
			every_owned_location = {
				limit = { has_building = building_type:ghulam_barracks }
				every_pop = {
					limit = { pop_type = pop_type:slaves }
					add_pop_size = 1
				}
			}
			# Tier up.
			if = {
				limit = { NOT = { has_variable = matter_ghulam_corps_tier } }
				set_variable = { name = matter_ghulam_corps_tier value = 1 }
				add_country_modifier = { modifier = matter_ghulam_corps_1 months = -1 }
				trigger_event_non_silently = matter_of_tork.62
			}
			else_if = {
				limit = { var:matter_ghulam_corps_tier = 1 }
				set_variable = { name = matter_ghulam_corps_tier value = 2 }
				remove_country_modifier = matter_ghulam_corps_1
				add_country_modifier = { modifier = matter_ghulam_corps_2 months = -1 }
				trigger_event_non_silently = matter_of_tork.63
			}
			else_if = {
				limit = { var:matter_ghulam_corps_tier = 2 }
				set_variable = { name = matter_ghulam_corps_tier value = 3 }
				remove_country_modifier = matter_ghulam_corps_2
				add_country_modifier = { modifier = matter_ghulam_corps_3 months = -1 }
				trigger_event_non_silently = matter_of_tork.64
			}
		}
	}

	ai_will_do = {
		add = 50
	}
}

# ---Approach 1: Declare Royal Inju ------------------------------------------
# Not demolition (the player can already raze buildings) — a POLITICAL
# act: the Tazik king declares a tribal zone crown demesne, the
# Ilkhanid inju, and pushes the hosts off it. Destroys the pasture,
# marks the land, rolls a crisis. The Eradication disposition needs
# five of these; Expulsion needs two — you cannot end or expel a
# class whose economic base still stands.

matter_tork_declare_inju = {
	type = disaster
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 6
	automation_tick = never
	automation_tick_frequency = 12

	price = price:matter_declare_inju

	cooldown = {
		type = matter_tork_declare_inju
		years = 2
	}

	allow = {
		scope:actor = {
			has_variable = matter_tork_approach
			var:matter_tork_approach = 1
			# Make the requirement legible on the button (playtest fix):
			# the declaration needs an undeclared pasture to seize.
			custom_tooltip = {
				text = matter_tork_inju_target_tt
				any_owned_location = {
					has_building = building_type:tribal_pasture
					NOT = { has_location_modifier = matter_royal_inju }
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = { data = name }
		visible = {
			disaster_type = disaster_type:matter_of_tork
			disaster_is_active = yes
		}
	}

	# Pick any pasture location; the declaration covers its whole
	# PROVINCE (playtest fix: inju is a zone-level legal act, not a
	# parcel-level one).
	select_trigger = {
		looking_for_a = location
		interaction_source_list = {
			scope:actor = {
				every_owned_location = {
					limit = { has_building = building_type:tribal_pasture }
					add_to_list = source
				}
			}
		}
		target_flag = target
		name = "choose_location"
		none_available_msg_key = "no_locations_available"
		column = { data = name }
		column = { data = population }
		visible = {
			NOT = { has_location_modifier = matter_royal_inju }
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			# The whole province becomes crown demesne: every owned
			# location marked; every pasture in it broken.
			scope:target.province = {
				every_location_in_province = {
					limit = { owner = scope:actor }
					if = {
						limit = { has_building = building_type:tribal_pasture }
						destroy_building = "building(building_type:tribal_pasture|owner)"
					}
					add_location_modifier = { modifier = matter_royal_inju years = -1 }
				}
			}
			scope:actor = {
				if = {
					limit = { NOT = { has_variable = matter_tork_inju_count } }
					set_variable = { name = matter_tork_inju_count value = 0 }
				}
				change_variable = { name = matter_tork_inju_count add = 1 }
				add_estate_satisfaction = { type = estate_type:tribes_estate value = -0.05 }
				# The crisis roll: quiet / ferment / an amir's open protest.
				random_list = {
					50 = {
						# The stewards comply. This time.
					}
					35 = {
						scope:target = {
							add_location_modifier = { modifier = matter_tork_unrest_minor years = 5 }
						}
					}
					15 = {
						trigger_event_non_silently = matter_of_tork.66
					}
				}
			}
		}
	}

	ai_will_do = {
		add = 30
	}
}

# ---Approach 1: Institutionalize the Slave Caravans -------------------------
# The Caucasus supply chain as a maintained institution, not free
# background. Requires the caravanserai network AND a fortified
# border with an unfortified non-Muslim neighbor — our castle facing
# their fort-less land (conquer the forts or demand their destruction
# in peace). Unlocks corps tiers II-III.
# TODO: explicit relative-strength check on the neighbor; market
# slave-SUPPLY injection (demand-side only in v1).

matter_tork_institutionalize_caravans = {
	type = disaster
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 6
	automation_tick = never
	automation_tick_frequency = 12

	price = price:matter_institutionalize_caravans

	allow = {
		scope:actor = {
			has_variable = matter_tork_approach
			var:matter_tork_approach = 1
			NOT = { has_variable = matter_slave_route }
			# The road infrastructure — LEVELS, not separate locations
			# (playtest fix: a single level-3 caravanserai qualifies).
			custom_tooltip = {
				text = matter_tork_caravanserai_tt
				total_effective_building_levels:caravanserai >= 3
			}
			# The fortified door onto the infidel slave frontier.
			custom_tooltip = {
				text = matter_tork_slave_frontier_tt
				any_owned_location = {
					OR = {
						has_building = building_type:castle
						has_building = building_type:fortress
					}
					any_neighbor_location = {
						exists = owner
						NOT = { owner = scope:actor }
						owner = {
							NOT = { religion.group = religion_group:muslim }
						}
						NOT = { has_building = building_type:castle }
						NOT = { has_building = building_type:fortress }
					}
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = { data = name }
		visible = {
			disaster_type = disaster_type:matter_of_tork
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			set_variable = matter_slave_route
			add_country_modifier = { modifier = matter_slave_caravans months = -1 }
		}
	}

	ai_will_do = {
		add = 40
	}
}

# ---B. Release the March (disposition B, consummated by the act) -------
# The march is EARNED by conquest, not auto-transferred (design
# decision 19; create_uc_bey is the engine-native template). The
# player picks a seat among owned tribal-pastoral locations whose
# AREA passes the checks: >= 8 owned locations [verify: count = N in
# any_* iterators] and a border with at least one country that is
# neither us nor our subject — someone to be unleashed upon.
# TODO v1.1: a pop-capacity floor on the zone (useful land, not
# desert margins) once the right location-scope trigger is settled.
# TODO v1.5: custom marcher_confederation subject type (uc_bey-like
# contract) instead of plain march — the ticking bomb in the terms.

matter_tork_release_march = {
	type = disaster
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 6
	automation_tick = never
	automation_tick_frequency = 12

	allow = {
		scope:actor = {
			has_variable = matter_tork_approach
			var:matter_tork_approach = 4
			matter_tork_no_disposition_yet = yes
			at_war = no
		}
	}

	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = { data = name }
		visible = {
			disaster_type = disaster_type:matter_of_tork
			disaster_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = location
		interaction_source_list = {
			scope:actor = {
				every_owned_location = {
					limit = { is_tribal_pastoral_location = yes }
					add_to_list = source
				}
			}
		}
		target_flag = target
		name = "choose_location"
		none_available_msg_key = "no_locations_available"
		column = { data = name }
		column = { data = population }
		visible = {
			is_tribal_pastoral_location = yes
			area = {
				# Big enough to be a real march.
				any_location_in_area = {
					count = 8
					owner = scope:actor
				}
				# Pointed at someone — the zone must touch a country
				# that is neither us nor ours.
				any_location_in_area = {
					owner = scope:actor
					any_neighbor_location = {
						exists = owner
						NOT = { owner = scope:actor }
						owner = {
							NOT = { is_subject_of = scope:actor }
						}
					}
				}
			}
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:target = {
				save_scope_as = matter_march_seat
				create_country_from_location = {
					change_country_name = MATTER_TORK_MARCH
					make_subject_of = { target = scope:actor type = subject_type:march }
					save_scope_as = matter_march_country
				}
			}
			# The rest of the zone rides out with the seat.
			scope:matter_march_seat.area = {
				every_location_in_area = {
					limit = { owner = scope:actor }
					change_location_owner = scope:matter_march_country
				}
			}
			scope:actor = {
				# The interior hosts ride out to join the march.
				set_variable = matter_tork_disposition_chosen
				set_variable = matter_tork_marcher_claimed
				# The compact ceremony — modifiers + resolution.
				trigger_event_non_silently = matter_of_tork.20
			}
		}
	}

	# The ghazi's native disposition; the farr-king's weak one.
	ai_will_do = {
		add = 20
		if = {
			limit = { scope:actor = { matter_tork_is_ghazi_line = yes } }
			add = 60
		}
	}
}
