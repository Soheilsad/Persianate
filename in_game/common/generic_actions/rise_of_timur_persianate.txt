# =====================================================================
# RISE OF TIMUR — the reform lands on the Persianate chassis
#
# REPLACE: base rot_reform_into_monarchy (generic_actions/
# rise_of_timur.txt:467) converts TIM to a generic monarchy. But the
# Timurid state IS the Persianate sultanate at maximum extension —
# Persian chancery, iqṭāʿ/soyurghal grants, tümen amīrs, a puppet
# khan for legitimacy. Land it on persianate_sultanate; the REPLACEd
# timurid_empire reform (04_timurid_reform.txt) now carries the full
# chancery block itself, so no separate bureaucracy-reform grant.
# Everything else — price, allow, select_trigger, AI — base verbatim.
# =====================================================================

REPLACE:rot_reform_into_monarchy = {
	type = situation
	potential = {
		scope:actor = {
			tag = TIM
		}
	}

	automation_tick = never
	automation_tick_frequency = 12
	price = price:rot_reform_into_monarchy_price
	allow = {
		scope:actor = {
			is_rebel_country = no
			at_war = no
			NOT = { country_type = location }
			custom_tooltip = {
				text = rot_has_selected_core_region_tt
				has_variable = rot_has_selected_core_region
			}
		}
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:rise_of_timur = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			change_country_type = location
			change_government_type = government_type:persianate_sultanate
			unlock_government_reform_effect = { type = timurid_empire }
			add_reform = government_reform:timurid_empire
			# Timur designated Pīr Muḥammad and the amīrs ignored him —
			# succession by designation-then-contest, the same chassis
			# Delhi runs.
			change_heir_selection = heir_selection:favorite_son_elective_succession
			# The language of the Ẓafarnāma and of every farmān out of
			# Samarqand.
			set_court_language = culture:farsi_culture.dialect
			set_country_rank = country_rank:rank_empire
		}
	}

	ai_will_do = {
		add = 250	#Ai should always do it asap
	}
}


# =====================================================================
# THE CONQUEST ACTIONS — base copies (1.4.7) with the marked changes.
#
# Plan Invasion offered only the Ṣāḥib-qirān's OWN neighbours (plus the
# realms of his declared ambition). But his own domain is one house's
# pastures; his frontier is the frontier of the amirs sworn to him. So
# both actions now count his subjects' borders as his.
#
# Declare Conquest Ambition: base's AI, under the historical rule, had to
# name Persia first. Nothing obliged him to go west: Iran was the nearest
# rich country without a master in the 1380s, which is a fact about the
# 1380s. A Ṣāḥib-qirān of the 1350s faces a Yuan in collapse; one whose
# Iran is already united has India. The AI now takes what is offered.
# Search "PERSIANATE WORLD" for each change; re-diff after a patch.
# =====================================================================

REPLACE:rot_declare_conquest_ambition = {
	type = situation
	potential = {
		scope:actor = {
			tag = TIM
		}
	}
	allow = {
		scope:actor = {
			is_rebel_country = no
		}
	}

	automation_tick = never
	automation_tick_frequency = 12

	ai_tick = monthly
	ai_tick_frequency = 12

	cooldown = {
		type = rot_declare_conquest_ambition
		years = 5
	}
	
	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:rise_of_timur = this
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = region
		target_flag = target
		name = "rot_declare_conquest_ambition_select_region"
		none_available_msg_key = "rot_declare_conquest_ambition_no_regions"
		column = {
			data = name
		}
		column = {
			data = population
		}
		visible = {
			scope:actor.capital.region != root
			any_area_in_region = {
				# PERSIANATE WORLD: his sworn amirs' and other subjects'
				# borders count as his own.
				save_temporary_scope_as = rot_area
				OR = {
					adjacent_to_owned_by = scope:actor
					scope:actor = {
						any_subject = {
							save_temporary_scope_as = rot_subject
							scope:rot_area = { adjacent_to_owned_by = scope:rot_subject }
						}
					}
				}
			}
		}
	}

	effect = {
		custom_tooltip = {
			text = rot_declare_conquest_ambition_tt
			scope:recipient = {
				if = {
					limit = { has_variable = rot_conquest_ambition }
					#clear up all the previous modifiers first
					var:rot_conquest_ambition = {
						every_location_in_region = {
							remove_location_modifier = rot_timur_scare
						}
					}
				}
				set_variable = {
					name = rot_conquest_ambition
					value = scope:target
				}
			}
		}
		custom_tooltip = rot_declare_conquest_ambition_at
		if = {
			limit = {
				scope:actor = {
					ruler ?= { has_character_modifier = timmy_wants_to_play }
				}
			}
			custom_tooltip = rot_declare_conquest_ambition_bt
		}
	}

	# PERSIANATE WORLD: the AI names an ambition by WEALTH AGAINST
	# STRENGTH (script_values/06_sahib_qiran_ai_values.txt,
	# sq_region_ambition_score): what the region's rulers can be taxed
	# for, against what they can put in the field, a wounded realm
	# counting for half its muster. Base named the first region offered
	# (or Persia, under the historical rule) and never changed its mind.
	#   no ambition yet        300 + the region's score: he always names
	#                          one, and it is the best on offer
	#   the ambition is made   the same: name the next
	#     good
	#   the ambition has       switches if the present one now scores
	#     turned against him   under -60 and this one beats it by 60
	#   otherwise              keeps what he has declared
	# An expansionist or aggressive ruler leans in; a cautious one weighs
	# each outmatching ruler again.
	ai_will_do = {
		value = -100
		if = {
			limit = {
				NOT = { scope:recipient = { has_variable = rot_conquest_ambition } }
			}
			value = 300
			add = scope:target.sq_region_ambition_score
		}
		else_if = {
			limit = {
				scope:actor = {
					is_target_in_variable_list = {
						name = sq_ambitions_done
						target = scope:recipient.var:rot_conquest_ambition
					}
				}
			}
			value = 300
			add = scope:target.sq_region_ambition_score
		}
		else_if = {
			limit = {
				scope:recipient.var:rot_conquest_ambition = {
					sq_region_ambition_score < -60
				}
				scope:target = {
					sq_region_ambition_score > {
						value = scope:recipient.var:rot_conquest_ambition.sq_region_ambition_score
						add = 60
					}
				}
			}
			value = scope:target.sq_region_ambition_score
		}
		if = {
			limit = {
				scope:actor = {
					OR = {
						ai_personality = ai_personality:ai_expansionist
						ai_personality = ai_personality:ai_aggressive
					}
				}
			}
			add = 15
		}
		if = {
			limit = {
				scope:actor = {
					OR = {
						ai_personality = ai_personality:ai_cautious
						ai_personality = ai_personality:ai_defensive
					}
				}
			}
			subtract = {
				value = scope:target.sq_region_outmatching_rulers
				multiply = 40
			}
		}
		# Never the region already declared, nor one already made good.
		if = {
			limit = {
				OR = {
					scope:recipient = {
						has_variable = rot_conquest_ambition
						var:rot_conquest_ambition = scope:target
					}
					scope:actor = {
						is_target_in_variable_list = {
							name = sq_ambitions_done
							target = scope:target
						}
					}
				}
			}
			subtract = 2000
		}
	}
}

REPLACE:rot_plan_invasion = {
	type = situation	
	potential = {
		scope:actor = {
			tag = TIM
		}
	}
	allow = {
		scope:actor = {
			is_rebel_country = no
		}
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:rise_of_timur = this
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			if = {
				limit = {
					scope:recipient = {
						has_variable = rot_conquest_ambition
					}
				}
				scope:recipient = {
					var:rot_conquest_ambition = {
						every_country_with_capital_in_geography = {
							limit = { this != scope:actor }
							add_to_list = source
						}
					}
				}
			}
			scope:actor = {
				every_neighbor_country = {
					add_to_list = source
				}
				# PERSIANATE WORLD: the neighbours of his subjects too. The
				# Ṣāḥib-qirān's own domain may be one tümen's pastures; his
				# frontier is his sworn amirs'.
				every_subject = {
					every_neighbor_country = {
						limit = {
							this != scope:actor
							NOT = { is_subject_of = scope:actor }
						}
						add_to_list = source
					}
				}
			}
		}
		target_flag = target
		name = "rot_plan_invasion_select_country"
		none_available_msg_key = "rot_plan_invasion_no_countries"
		column = {
			data = name
		}
		column = {
			data = expected_army_size
		}
		visible = {
			is_real_country = yes
			scope:actor = {
				can_declare_war_on = root
				NOT = { has_truce_with = root }
				NOT = { is_allied_with = { target = root } }
			}
		}
	}

	cooldown = {
		type = rot_plan_invasion
		months = 3
	}

	automation_tick = never
	automation_tick_frequency = 12

	ai_tick = monthly
	ai_tick_frequency = {
		if = {
			limit = {
				at_war = yes
			}
			value = 18
		}
		else = {
			value = 3
		}
	}

	effect = {
		scope:actor = {
			if = {
				limit = {
					exists = scope:target
				}
				if = {
					limit = {
						scope:target = {
							capital = {
								OR = {
									region = scope:actor.capital.region
									AND = {
										scope:recipient = { has_variable = rot_conquest_ambition }
										region = scope:recipient.var:rot_conquest_ambition
									}
								}
							}
						}
					}
					drop_antagonism_bomb = {
						target = scope:target.capital
						modifier = antagonism_started_invasion
					}
					declare_war_with_cb = {
						target = scope:target
						type = casus_belli:cb_timurs_conquests
					}
				}
				else = {
					add_casus_belli = {
						target = scope:target
						type = casus_belli:cb_timurs_conquests
					}
					drop_antagonism_bomb = {
						target = scope:target.capital
						modifier = antagonism_planned_invasion
					}
				}
			}
			else = {
				custom_tooltip = rot_plan_invasion_tt
				if = {
					limit = {
						scope:recipient = { has_variable = rot_conquest_ambition }
					}
					custom_tooltip = rot_plan_invasion_bt
				}
				else = {
					custom_tooltip = rot_plan_invasion_at
				}
			}
		}
	}
	# PERSIANATE WORLD: base's weighting kept as the core (home or
	# ambition region +200, less his opinion of the target, scaled by
	# relative strength), with the judgement of script_values/
	# 06_sahib_qiran_ai_values.txt on top:
	#   + wealth     20 per multiple of his own tax base, to 80
	#   + 60         a wounded realm (war exhaustion, spent manpower)
	#   FORTUNE LOW  he needs a war he can win, soon: +150 against a
	#                target under half his strength, -200 against one
	#                near his own
	#   FORTUNE HIGH he can afford a real war: no penalty up to his own
	#                strength
	#   in between   -100 against a target at 0.9 of his strength or more
	#   cautious /   a further -80 against anything over 0.8
	#     defensive
	#   NEVER        a realm that outmatches him (1.25x, wounded halved),
	#                nor one neither he nor a subject borders
	ai_will_do = {
		value = -20
		if = {
			limit = { exists = scope:target }
			if = {
				limit = {
					scope:target = {
						capital = {
							OR = {
								region = scope:actor.capital.region
								AND = {
									scope:recipient = { has_variable = rot_conquest_ambition }
									region = scope:recipient.var:rot_conquest_ambition
								}
							}
						}
					}
				}
				add = 200
			}
			add = {
				value = "scope:actor.opinion(scope:target)"
				multiply = -1
			}
			add = {
				value = "scope:actor.antagonism(scope:actor)"
				divide = -10
			}
			multiply = "scope:actor.relative_military_strength(scope:target)"

			add = {
				value = scope:target.sq_target_wealth_ratio
				multiply = 20
				max = 80
			}
			if = {
				limit = { scope:target = { sq_is_wounded_realm = yes } }
				add = 60
			}
			if = {
				limit = { scope:actor = { has_variable = sq_band_low } }
				if = {
					limit = { scope:target = { sq_target_strength_ratio < sq_ai_easy_ratio } }
					add = 150
				}
				else_if = {
					limit = { scope:target = { sq_target_strength_ratio >= 0.8 } }
					subtract = 200
				}
			}
			else_if = {
				limit = {
					NOT = { scope:actor = { has_variable = sq_band_high } }
					scope:target = { sq_target_strength_ratio >= 0.9 }
				}
				subtract = 100
			}
			if = {
				limit = {
					scope:actor = {
						OR = {
							ai_personality = ai_personality:ai_cautious
							ai_personality = ai_personality:ai_defensive
						}
					}
					scope:target = { sq_target_strength_ratio >= 0.8 }
				}
				subtract = 80
			}
			if = {
				limit = { scope:target = { sq_outmatches_actor = yes } }
				subtract = 2000
			}
			if = {
				limit = {
					NOT = { scope:actor = { sq_reaches_target = yes } }
				}
				subtract = 1000
			}
		}
	}
}
