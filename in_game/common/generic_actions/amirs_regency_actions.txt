# ============================================================
# The Amirs' Regency — actions on the organisation's panel.
#
# Organisation actions (type = internationalorganization, the base
# claim_ilkhanate construction: one select_trigger that picks the
# organisation itself as scope:recipient), because the regency belongs
# to the organisation and moves between states.
#
#   FOR THE SEAT (khan or regent)
#     ulus_recall_darughas       cut the begs' weight; angers them
#   FOR THE REGENT
#     ulus_marry_into_the_line   the güregen's marriage
#     ulus_distribute_largesse   gold into the legitimacy gap
#   FOR AN AMIR WHO DOES NOT HOLD THE KHAN
#     ulus_claim_the_khan        a casus belli on the regent (stage 2+)
#     ulus_back_the_khan         the khan's purge — Qazan's gamble
#
# The fourth way to arrest the regency, granting an ulus to a prince of
# the line, is the existing Grant a Share interaction; its effect
# (scripted_effects/00_ulus_effects.txt) now sets the clock back.
#
# Prices: common/prices/04_amirs_regency_prices.txt.
# AI list: common/generic_action_ai_lists/amirs_regency_list.txt.
# ============================================================

# ------------------------------------------------------------
# RECALL THE DARUGHAS. The seat calls its appointed governors back to
# the ordu. It was the appointed aristocracy, not the princes, that
# grew heavy enough to hold a khan (events/horde_succession.txt), and a
# ruler who sees the bar filling can thin it. The begs do not thank
# him. Available to whoever leads the ulus, khan or regent; during a
# regency it also sets the clock back.
#
# The seats themselves stand (no verified effect removes a building
# from script here); what is cut is the begs' power, for ten years.
# ------------------------------------------------------------
ulus_recall_darughas = {
	type = disaster
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = ulus_recall_darughas
		years = 10
	}

	price = price:ulus_recall_darughas_price

	potential = {
		scope:actor = {
			is_ulus_chaghatay_leader = yes
		}
	}

	allow = {
		scope:actor = {
			country_has_estate = estate_type:nobles_estate
			NOT = { has_country_modifier = ulus_darughas_recalled_mod }
		}
	}

	# On the Amirs' Regency DISASTER panel (author, 2026-11-14): these
	# are the regent's own business, and that is where a player looks
	# for them. The houses' business (pledges, the claim, raids) stays
	# on the ulus panel.
	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = {
			data = name
		}
		visible = {
			disaster_type = disaster_type:amirs_regency
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			add_country_modifier = {
				modifier = ulus_darughas_recalled_mod
				years = 10
			}
			add_estate_satisfaction = { type = estate_type:nobles_estate value = -0.15 }
			add_stability = -10
			international_organization:ulus_chaghatay ?= {
				ulus_change_progress = { AMOUNT = ulus_regency_progress_recall }
				ulus_chaghatay_rebuild_ledger = yes
			}
		}
	}

	# The AI recalls when its begs are within reach of the bar.
	# The AI regent recalls the governors when the begs of his own realm
	# are strong enough to matter and the regency is running away from
	# him; not as a habit.
	ai_will_do = {
		add = -15
		if = {
			limit = { scope:actor = { "estate_power(estate_type:nobles_estate)" >= 0.4 } }
			add = 25
		}
		if = {
			limit = { scope:actor = { "estate_power(estate_type:nobles_estate)" >= 0.5 } }
			add = 20
		}
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= {
					has_variable = ulus_regency_progress
					var:ulus_regency_progress >= 60
				}
			}
			add = 20
		}
		if = {
			limit = {
				scope:actor = {
					country_has_estate = estate_type:nobles_estate
					"estate(estate_type:nobles_estate)" = { satisfaction < 0.4 }
				}
			}
			add = -20	# the begs are already close to the knife
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_cautious } }
			add = 10
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_defensive } }
			add = 10
		}
	}
}

# ------------------------------------------------------------
# MARRY INTO THE LINE. The regent takes a woman of the khan's house to
# wife and is güregen, son-in-law of the line: Temür's own title after
# 1370, by his marriage to Saray Mulk. It buys time now (the amirs read
# it as respect for the line) and it is one of the three things the
# Ṣāḥib-qirān needs. It also binds his HOUSE to the line: when a puppet
# khan dies, a house with the tie may raise a khan of its own
# daughter's blood.
# ------------------------------------------------------------
ulus_marry_into_the_line = {
	type = disaster
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = ulus_marry_into_the_line
		years = 5
	}

	price = price:ulus_marry_into_the_line_price

	potential = {
		scope:actor = {
			has_variable = ulus_regent_seat
		}
	}

	allow = {
		scope:actor = {
			is_ulus_regent = yes
			has_ruler = yes
			custom_tooltip = {
				text = ulus_not_yet_gurkan_tt
				NOT = { ulus_ruler_is_gurkan = yes }
			}
			custom_tooltip = {
				text = ulus_holds_a_khan_tt
				international_organization:ulus_chaghatay ?= { has_variable = ulus_held_khan }
			}
		}
	}

	# On the Amirs' Regency DISASTER panel (author, 2026-11-14): these
	# are the regent's own business, and that is where a player looks
	# for them. The houses' business (pledges, the claim, raids) stays
	# on the ulus panel.
	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = {
			data = name
		}
		visible = {
			disaster_type = disaster_type:amirs_regency
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			set_variable = ulus_inlaw_house
			# The bride: a daughter of the khan's house. The marriage
			# itself is attempted (a ruler already married may be
			# refused it by the engine); the variable on the man is what
			# the regency reads either way.
			create_character = {
				dynasty = dynasty:borjigin_dynasty
				estate = estate_type:nobles_estate
				religion = scope:actor.religion
				culture = scope:actor.culture
				age = { 16 22 }
				female = yes
				save_scope_as = ulus_bride
			}
			ruler = {
				set_variable = ulus_gurkan
				marry_character = scope:ulus_bride
			}
			add_prestige = 10
			# The amirs' regard for a son-in-law of the line.
			save_scope_as = ulus_giver
			international_organization:ulus_chaghatay ?= {
				every_international_organization_member = {
					limit = { NOT = { this = scope:ulus_giver } }
					add_opinion = {
						modifier = opinion_ulus_gurkan
						target = scope:ulus_giver
					}
				}
			}
			international_organization:ulus_chaghatay ?= {
				ulus_change_progress = { AMOUNT = ulus_regency_progress_marriage }
				ulus_chaghatay_rebuild_ledger = yes
			}
		}
	}

	# The marriage is the one thing a regent cannot do without if he means
	# to end as Ṣāḥib-qirān, so the AI wants it early and wants it more
	# the later the stage. It waits only if it cannot pay.
	ai_will_do = {
		add = 40
		if = {
			limit = { scope:actor = { ulus_regency_stage_at_least = { STAGE = 3 } } }
			add = 40
		}
		if = {
			limit = { scope:actor = { gold < 0 } }
			add = -100
		}
		if = {
			limit = { scope:actor = { at_war = yes } }
			add = -15
		}
	}
}

# ------------------------------------------------------------
# DISTRIBUTE LARGESSE. Gold into the gap. A beg's followers stay
# because he gives: robes, horses, a share of everything taken. Temür
# emptied his treasury after every campaign and was the richer for it.
# A year's income for ulus_gap_largesse points, once a year.
# ------------------------------------------------------------
ulus_distribute_largesse = {
	type = disaster
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = ulus_distribute_largesse
		years = 1
	}

	price = price:ulus_largesse_price

	potential = {
		scope:actor = {
			has_variable = ulus_regent_seat
		}
	}

	allow = {
		scope:actor = {
			is_ulus_regent = yes
		}
	}

	# On the Amirs' Regency DISASTER panel (author, 2026-11-14): these
	# are the regent's own business, and that is where a player looks
	# for them. The houses' business (pledges, the claim, raids) stays
	# on the ulus panel.
	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = {
			data = name
		}
		visible = {
			disaster_type = disaster_type:amirs_regency
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			save_scope_as = ulus_giver
			add_estate_satisfaction = { type = estate_type:nobles_estate value = 0.05 }
			# Not only his own begs. The robes and the silver go to the
			# AMIRS OF THE ULUS: each thinks better of him, his own
			# subjects are the surer for it, and for two years he stands
			# taller among them (ulus_weight_boost, read by the ledger:
			# it counts in the bar of the three greatest amirs against
			# him and in his heir's standing).
			international_organization:ulus_chaghatay ?= {
				every_international_organization_member = {
					limit = { NOT = { this = scope:ulus_giver } }
					add_opinion = {
						modifier = opinion_ulus_largesse
						target = scope:ulus_giver
					}
					# A house that has taken his silver does not ride on
					# him while it is remembered (read by Claim the Khan's
					# AI, and shown in its tooltip to a player).
					set_variable = { name = ulus_took_his_silver value = 1 months = ulus_largesse_months }
				}
			}
			add_country_modifier = {
				modifier = ulus_largesse_given_mod
				months = ulus_largesse_months
				mode = add_and_extend
			}
			set_variable = { name = ulus_weight_boost value = ulus_largesse_weight months = ulus_largesse_months }
			international_organization:ulus_chaghatay ?= {
				ulus_change_gap = { AMOUNT = ulus_gap_largesse }
				ulus_chaghatay_rebuild_ledger = yes
			}
		}
	}

	# The AI pays when the gap is under the filled line.
	# The AI regent opens the treasury by how far the gap has fallen under
	# its line: nothing while it is filled, urgently once it is empty,
	# and more readily in the last stage, when the filled years are being
	# counted. Never in debt, and less in a war he is paying for.
	ai_will_do = {
		add = -10
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= {
					has_variable = ulus_gap
					var:ulus_gap < ulus_gap_filled_line
				}
			}
			add = {
				value = ulus_gap_filled_line
				subtract = international_organization:ulus_chaghatay.var:ulus_gap
				multiply = 1.5
				max = 60
			}
		}
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= {
					has_variable = ulus_gap_empty_months
					var:ulus_gap_empty_months > 0
				}
			}
			add = 40
		}
		if = {
			limit = { scope:actor = { ulus_regency_stage_at_least = { STAGE = 4 } } }
			add = 15
		}
		if = {
			limit = { scope:actor = { gold < 0 } }
			add = -100
		}
		if = {
			limit = { scope:actor = { at_war = yes } }
			add = -10
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_friendly } }
			add = 10
		}
	}
}

# ------------------------------------------------------------
# CLAIM THE KHAN. Once the amirs have fallen out (stage 2), an amir who
# does not hold the khan may claim him: a casus belli on the regent for
# ten years. Inside the ulus this is the only road to war — the
# organisation forbids war without one on a fellow member.
# ------------------------------------------------------------
ulus_claim_the_khan = {
	type = internationalorganization
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = ulus_claim_the_khan
		years = 10
	}

	price = price:ulus_claim_the_khan_price

	potential = {
		scope:actor = {
			is_ulus_chaghatay_amir = yes
		}
	}

	allow = {
		scope:actor = {
			custom_tooltip = {
				text = ulus_amirs_fallen_out_tt
				ulus_regency_stage_at_least = { STAGE = 2 }
			}
			at_war = no
			# A player is bound as the AI is inclined: no claim while the
			# regent's largesse to us is remembered.
			custom_tooltip = {
				text = ulus_took_his_silver_tt
				NOT = { has_variable = ulus_took_his_silver }
			}
		}
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = {
			data = name
		}
		visible = {
			international_organization_type = international_organization_type:ulus_chaghatay
		}
	}

	effect = {
		scope:actor = {
			international_organization:ulus_chaghatay ?= {
				leader_country ?= { save_scope_as = ulus_regent }
			}
			if = {
				limit = { exists = scope:ulus_regent }
				add_casus_belli = {
					type = casus_belli:cb_take_the_khan
					target = scope:ulus_regent
					years = 10
				}
				# Read by on_action/32_amirs_cadet_hooks.txt: a chief who
				# loses this war may lose his tümen to a kinsman.
				set_variable = { name = ulus_claimed_the_khan value = 1 years = 10 }
				# Told to the claimant (.22), the keeper (.23) and every
				# other house (.24), each with the names.
				save_scope_as = ulus_claimant
				trigger_event_non_silently = amirs_rising.22
				scope:ulus_regent = {
					trigger_event_non_silently = amirs_rising.23
				}
				international_organization:ulus_chaghatay ?= {
					every_international_organization_member = {
						limit = {
							is_ai = no
							NOT = { this = scope:ulus_claimant }
							NOT = { this = scope:ulus_regent }
						}
						trigger_event_non_silently = amirs_rising.24
					}
				}
			}
		}
	}

	# The AI claims when it stands among the three likeliest kingmakers.
	# The AI amir claims the khan when he can plausibly take him: by his
	# weight against the regent's, his own standing among the amirs, and
	# how badly the regent is doing. A house sworn to the regent does
	# not, and a cautious chief wants better odds than an aggressive one.
	ai_will_do = {
		add = -25
		if = {
			limit = {
				scope:actor = { has_variable = ulus_weight_eff }
				international_organization:ulus_chaghatay ?= {
					leader_country ?= { has_variable = ulus_weight_eff }
				}
			}
			if = {
				limit = {
					scope:actor.var:ulus_weight_eff >= {
						value = international_organization:ulus_chaghatay.leader_country.var:ulus_weight_eff
						multiply = 1.3
					}
				}
				add = 55
			}
			else_if = {
				limit = {
					scope:actor.var:ulus_weight_eff >= {
						value = international_organization:ulus_chaghatay.leader_country.var:ulus_weight_eff
						multiply = 0.9
					}
				}
				add = 30
			}
			else_if = {
				limit = {
					scope:actor.var:ulus_weight_eff < {
						value = international_organization:ulus_chaghatay.leader_country.var:ulus_weight_eff
						multiply = 0.7
					}
				}
				add = -60
			}
		}
		# His robes are still on our backs.
		if = {
			limit = { scope:actor = { has_variable = ulus_took_his_silver } }
			add = -40
		}
		# A keeper of the khan with no host is an invitation, whatever
		# his rents and his friends make him weigh.
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= {
					leader_country ?= {
						army_size < {
							value = scope:actor.army_size
							multiply = 0.34
						}
					}
				}
			}
			add = 60
		}
		else_if = {
			limit = {
				international_organization:ulus_chaghatay ?= {
					leader_country ?= {
						army_size < {
							value = scope:actor.army_size
							multiply = 0.67
						}
					}
				}
			}
			add = 30
		}
		# And one the houses already hold in sufferance.
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= { has_variable = ulus_regent_outweighed }
			}
			add = 10
		}
		if = {
			limit = {
				scope:actor = {
					is_target_in_global_variable_list = {
						name = ulus_kingmakers
						target = this
					}
				}
			}
			add = 15
		}
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= {
					has_variable = ulus_gap
					var:ulus_gap < 30
				}
			}
			add = 20	# the regent's deeds no longer cover for him
		}
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= {
					leader_country ?= { has_variable = ulus_recent_victory }
				}
			}
			add = -20
		}
		if = {
			limit = { scope:actor = { has_variable = ulus_sworn_to_regent } }
			add = -80
		}
		if = {
			limit = {
				scope:actor = {
					has_variable = ulus_outside_weight
					var:ulus_outside_weight > 3
				}
			}
			add = 10
		}
		if = {
			limit = { scope:actor = { horde_unity < 30 } }
			add = -25
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_aggressive } }
			add = 15
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_expansionist } }
			add = 10
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_cautious } }
			add = -20
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_defensive } }
			add = -10
		}
	}
}

# ------------------------------------------------------------
# BACK THE KHAN — the khan's purge, Qazan's gamble.
#
# Qazan Khan (1343-46) was the last khan to try to rule the western
# amirs rather than be ruled by them; he executed several, drove the
# rest together, and was killed by Qazaghan in battle. A grown khan
# with a soldier's skill, kept in a beg's tent, may try the same from
# the inside — but not alone. An amir who does not hold the khan sends
# him men and silver.
#
#   SUCCESS  the Khan Triumphant: the khan kills the regent's house
#            and takes the regent's state for his own (.200).
#   FAILURE  the regency moves a stage on, and the khan usually dies
#            for it (.201).
# THE ROLL (literals, as in scripted_effects/15_...: THE ROLLS):
# success 30%, or 50% if the khan's military skill is at least
# ulus_purge_good_mil; on failure the khan is killed 3 times in 4.
# ------------------------------------------------------------
ulus_back_the_khan = {
	type = internationalorganization
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 24
	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = ulus_back_the_khan
		years = 10
	}

	price = price:ulus_back_the_khan_price

	potential = {
		scope:actor = {
			is_ulus_chaghatay_amir = yes
		}
	}

	allow = {
		scope:actor = {
			ulus_regency_is_on = yes
			custom_tooltip = {
				text = ulus_khan_can_purge_tt
				international_organization:ulus_chaghatay ?= {
					var:ulus_held_khan ?= {
						is_alive = yes
						is_adult = yes
						mil >= ulus_purge_min_mil
					}
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = {
			data = name
		}
		visible = {
			international_organization_type = international_organization_type:ulus_chaghatay
		}
	}

	effect = {
		scope:actor = {
			save_scope_as = ulus_khans_backer
			international_organization:ulus_chaghatay ?= {
				leader_country ?= { save_scope_as = ulus_regent }
				var:ulus_held_khan ?= { save_scope_as = ulus_purging_khan }
			}
			if = {
				limit = {
					exists = scope:ulus_regent
					exists = scope:ulus_purging_khan
				}
				if = {
					limit = {
						scope:ulus_purging_khan = { mil >= ulus_purge_good_mil }
					}
					random_list = {
						50 = {
							scope:ulus_regent = { trigger_event_non_silently = amirs_regency.200 }
						}
						50 = {
							scope:ulus_regent = { trigger_event_non_silently = amirs_regency.201 }
						}
					}
				}
				else = {
					random_list = {
						30 = {
							scope:ulus_regent = { trigger_event_non_silently = amirs_regency.200 }
						}
						70 = {
							scope:ulus_regent = { trigger_event_non_silently = amirs_regency.201 }
						}
					}
				}
			}
		}
	}

	# The AI gambles rarely, and not when it could take the khan itself.
	# The AI amir backs the khan's purge when he has more to gain from a
	# khan than from this regent: a man of the line himself, a rival of
	# the regent, or one who thinks ill of him. A house that could take
	# the khan for itself would rather do that.
	ai_will_do = {
		add = -10
		if = {
			limit = { scope:actor = { ulus_ruler_is_of_the_line = yes } }
			add = 40
		}
		if = {
