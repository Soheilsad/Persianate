# =====================================================================
# NETWORK ACTIONS — the preacher dispatch and the patronage flow
#
# dispatch_dervish_preacher (Commonwealth of Martyrs):
#   Direct the Juri network at a chosen foreign city. Gated on Shaykhi
#   power >= 25% — only a dervish-ascendant commonwealth commands the
#   network (the Sufyani wing hates it). Each dispatch drains the
#   sending wing: a stacking -5% Shaykhi power malus for 25 years
#   (preachers_dispatched_mod). Spread the revolution and you hollow
#   it at home — the Muʾayyad normalization arc as a resource loop.
#
# patronize_tariqa_order (any Muslim ruler hosting an order's lodge):
#   Gold flows to the shrine-seat, baraka flows back. In the
#   legitimacy-starved decades after 1335 this is the cheapest
#   legitimacy on the market — which is historically exactly why the
#   orders got rich.
# =====================================================================

dispatch_dervish_preacher = {
	show_message = no
	# Situation action: renders in the sarbadar_phenomenon panel
	# (SituationView.GetActionGroups) — the situation is the home of
	# the whole ferment, and a bare panel is a dead panel.
	type = situation

	sound = UI_action_religion_generic

	ai_tick = monthly
	ai_tick_frequency = 6
	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			government_type = government_type:commonwealth_of_martyrs
		}
		is_situation_active = situation:sarbadar_phenomenon
	}

	allow = {
		scope:actor = {
			"estate_power(estate_type:shaykhi_dervishes_estate)" >= 0.25
		}
	}

	cooldown = {
		type = dispatch_dervish_preacher
		months = 24
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:sarbadar_phenomenon = this
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = location
		target_flag = target
		name = "dispatch_preacher_select_city"
		none_available_msg_key = "dispatch_preacher_no_cities"
		column = { data = name }
		visible = {
			is_city = yes
			NOT = { has_building_with_at_least_one_level = sarbadar_preachers }
			NOT = { owner ?= scope:actor }
			NOT = { owner ?= { government_type = government_type:commonwealth_of_martyrs } }
			any_pop = {
				religion.group = religion_group:muslim
			}
		}
	}

	effect = {
		scope:target = {
			construct_building = {
				building_type = building_type:sarbadar_preachers
				cost_multiplier = 0
				instant = yes
			}
		}
		scope:actor = {
			# The wing that sends its best men abroad rules less at home.
			add_country_modifier = {
				modifier = preachers_dispatched_mod
				years = 25
			}
		}
	}

	ai_will_do = {
		add = 50
	}
}

patronize_tariqa_order = {
	show_message = no
	# Situation action — lives in the sarbadar_phenomenon panel with
	# the rest of the ferment's interactions.
	type = situation

	sound = UI_action_religion_generic

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			religion.group = religion_group:muslim
			NOT = { country_type = building }
		}
	}

	allow = {
	}

	cooldown = {
		type = patronize_tariqa_order
		months = 60
	}

	price = price:patronize_tariqa_order

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:sarbadar_phenomenon = this
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = target
		name = "patronize_order_select"
		none_available_msg_key = "patronize_order_no_orders"
		column = { data = name }
		visible = {
			has_reform = government_reform:tariqa_order
			# The order must actually have a lodge in the patron's realm.
			any_owned_foreign_building = {
				building_type = building_type:tariqa_khanaqah
				location ?= {
					owner ?= scope:actor
				}
			}
		}
	}

	effect = {
		# The order receives what the patron paid (price = scaled_gold 6).
		scope:target = {
			add_gold = {
				value = scope:actor.monthly_income_total
				multiply = 6
			}
		}
		scope:actor = {
			# The shaykh's blessing, read from the minbar.
			add_legitimacy = 10
			add_stability = 5
		}
		scope:actor = {
			capital = {
				construct_building = {
					building_type = building_type:grand_tariqa_khanaqah
					owner = scope:target
					payer = scope:actor
				}
			}
		}
	}

	ai_will_do = {
		add = {
			# The legitimacy-starved reach for the shrine.
			if = {
				limit = { scope:actor = { legitimacy < 40 } }
				value = 50
			}
			if = {
				limit = { scope:actor = { legitimacy < 20 } }
				value = 50
			}
		}
	}
}
