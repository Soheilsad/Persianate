# =====================================================================
# SHATTERED RUM — actions
#
# Disaster-panel program (Seljuk Inheritance, in order):
#   rum_strike_the_sikka / rum_reopen_trunk_road / rum_restore_the_diwan
#
# Situation-panel actions (Shattered Rum, the ghazi road's late doors):
#   endow_chancery_of_rum — the expensive chancery for the ghazi who
#     conquered first and must now govern; the Seljuk road gets the
#     same apparatus cheaper through the disaster, earlier, because
#     the claimant bleeds for it.
#   petition_caliph_for_rum — the Bayezid door (1394 precedent): a
#     vindicated ghazi with a functioning chancery buys the title of
#     Sultan of Rum from the Abbasid shadow in Cairo. Sets the
#     variable REPLACE:RUM_f reads.
# =====================================================================

rum_strike_the_sikka = {
	type = disaster
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 1
	automation_tick = never
	automation_tick_frequency = 12

	price = price:rum_sikka_price

	allow = {
		scope:actor = {
			owns = location:konya
			NOT = { has_variable = rum_sikka_struck }
		}
	}

	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = {
			data = name
		}
		visible = {
			disaster_type = disaster_type:seljuk_inheritance
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			set_variable = rum_sikka_struck
			change_variable = { name = rum_program_steps add = 1 }
			add_country_modifier = {
				modifier = rum_sikka_struck_mod
				years = -1
			}
			add_country_modifier = {
				modifier = rum_program_relief_mod
				years = 5
				mode = add_and_extend
			}
			trigger_event_non_silently = rum.10
		}
	}

	ai_will_do = {
		add = 100
	}
}

endow_chancery_of_rum = {
	type = situation

	automation_tick = never
	automation_tick_frequency = 12

	price = price:rum_endow_chancery_price

	potential = {
		scope:actor = {
			government_type = government_type:persianate_sultanate
			OR = {
				has_variable = ghazi_sword_of_islam
				has_variable = ghazi_guardian_of_thughur
			}
		}
	}

	allow = {
		scope:actor = {
			is_rebel_country = no
			at_war = no
			modifier:has_nizam_ul_mulk_chancery = no
			NOT = { has_reform = government_reform:persianate_bureaucracy_reform }
		}
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:rise_of_the_ottomans = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			add_reform = government_reform:persianate_bureaucracy_reform
			# Scribes imported into a war camp: the begs watch Persian
			# clerks take over the registers of what they conquered.
			add_stability = -10
		}
	}

	ai_will_do = {
		add = 200
	}
}

petition_caliph_for_rum = {
	type = situation

	automation_tick = never
	automation_tick_frequency = 12

	price = price:rum_petition_caliph_price

	potential = {
		scope:actor = {
			OR = {
				has_variable = ghazi_sword_of_islam
				has_variable = ghazi_guardian_of_thughur
			}
		}
	}

	allow = {
		scope:actor = {
			is_rebel_country = no
			# A functioning sultanate petitions; a war camp does not.
			modifier:has_nizam_ul_mulk_chancery = yes
			custom_tooltip = {
				text = rum_petition_strongest_tt
				situation:rise_of_the_ottomans.var:strongest_beylik_variable ?= this
			}
			NOT = { has_variable = rum_caliph_sanction }
		}
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:rise_of_the_ottomans = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			set_variable = rum_caliph_sanction
			add_legitimacy = 25
			add_prestige = 10
		}
	}

	ai_will_do = {
		add = 200
	}
}
