# ============================================================
# BEFORE THE REGENCY: what an amir can DO about a khan he outweighs.
#
# The Amirs' Regency began only at a khan's natural death. That left
# an amir who had done everything right (grown his tümen, or helped
# another house grow, until the bar was met) waiting for a funeral.
# Qazaghan did not wait: in 1346 the western amirs rose under him and
# killed Qazan Khan in battle.
#
#   ulus_rise_against_the_khan   a casus belli on the khan's seat, for
#                                an amir among the likeliest kingmakers
#                                while the bar is met. Won: the khan
#                                is killed and the amir begins the
#                                regency himself. Lost: the khan has
#                                broken the begs for a decade.
#   ulus_pledge_support          lend half one's weight to another
#                                house (it counts in the three-
#                                greatest bar and in his kingmaker
#                                score); if he takes the khan he owes
#                                the favour.
#   ulus_withdraw_pledge         take it back.
#
# Organisation actions, the same shape as amirs_regency_actions.txt
# (ulus_claim_the_khan); the outcome of the war is read by
# on_action/35_amirs_rising_hooks.txt.
# ============================================================

ulus_rise_against_the_khan = {
	type = internationalorganization
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = ulus_rise_against_the_khan
		years = 10
	}

	price = price:ulus_rise_against_the_khan_price

	potential = {
		scope:actor = {
			is_ulus_chaghatay_amir = yes
		}
		international_organization:ulus_chaghatay ?= {
			NOT = { has_variable = ulus_regency }
		}
	}

	allow = {
		scope:actor = {
			at_war = no
			# The amirs do not follow a nobody.
			custom_tooltip = {
				text = ulus_rising_kingmaker_tt
				OR = {
					is_target_in_global_variable_list = {
						name = ulus_kingmakers
						target = this
					}
					# DEBUG: any amir may rise while the debug bar is on.
					international_organization:ulus_chaghatay ?= { has_variable = ulus_debug_bar_met }
				}
			}
		}
		# The begs outweigh the throne: the same bar a khan's death is
		# judged by.
		custom_tooltip = {
			text = ulus_rising_bar_tt
			international_organization:ulus_chaghatay ?= {
				leader_country ?= { ulus_regency_bar_met = yes }
			}
		}
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = {
			data = name
		}
		visible = {
			international_organization_type = international_organization_type:ulus_chaghatay
		}
	}

	effect = {
		scope:actor = {
			set_variable = { name = ulus_rising value = 1 years = 10 }
			international_organization:ulus_chaghatay ?= {
				leader_country ?= { save_scope_as = ulus_khan_seat }
			}
			if = {
				limit = { exists = scope:ulus_khan_seat }
				add_casus_belli = {
					type = casus_belli:cb_rise_against_the_khan
					target = scope:ulus_khan_seat
					years = 5
				}
				scope:ulus_khan_seat = {
					trigger_event_non_silently = amirs_rising.1
				}
			}
		}
	}

	# The AI rises when it is the likeliest kingmaker of all and the bar
	# has been met for ulus_rising_ai_patience_months.
	# The AI rises when it can win and has waited long enough to be sure
	# of the others: the likeliest kingmaker of all, the bar met for
	# ulus_rising_ai_patience_months, and weight to match the khan's own
	# (tümens pledged to it and friends outside count; so do the lords
	# pledged to the khan, against it). A child on the felt is an
	# invitation. A cautious chief waits for better odds, and nobody
	# rises with his own horde coming apart: the price of failing is his
	# land and perhaps his head.
	ai_will_do = {
		add = -40
		if = {
			limit = {
				scope:actor = {
					has_variable = ulus_kingmaker_score
					international_organization:ulus_chaghatay ?= { has_variable = ulus_top_kingmaker_score }
					var:ulus_kingmaker_score >= international_organization:ulus_chaghatay.var:ulus_top_kingmaker_score
				}
			}
			add = 25
		}
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= {
					has_variable = ulus_bar_met_months
					var:ulus_bar_met_months >= ulus_rising_ai_patience_months
				}
			}
			add = 20
		}
		if = {
			limit = {
				scope:actor = { has_variable = ulus_weight_eff }
				international_organization:ulus_chaghatay ?= {
					leader_country ?= { has_variable = ulus_weight_eff }
				}
			}
			if = {
				limit = {
					scope:actor.var:ulus_weight_eff >= {
						value = international_organization:ulus_chaghatay.leader_country.var:ulus_weight_eff
						multiply = 1.25
					}
				}
				add = 35
			}
			else_if = {
				limit = {
					scope:actor.var:ulus_weight_eff >= international_organization:ulus_chaghatay.leader_country.var:ulus_weight_eff
				}
				add = 15
			}
			else_if = {
				limit = {
					scope:actor.var:ulus_weight_eff < {
						value = international_organization:ulus_chaghatay.leader_country.var:ulus_weight_eff
						multiply = 0.8
					}
				}
				add = -60
			}
		}
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= {
					leader_country ?= {
						ruler ?= { is_adult = no }
					}
				}
			}
			add = 20
		}
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= { international_organization_has_leader = yes }
				scope:actor = { is_rival_of = international_organization:ulus_chaghatay.leader_country }
			}
			add = 15
		}
		if = {
			limit = {
				scope:actor = {
					has_variable = ulus_backing
					var:ulus_backing > 0
				}
			}
			add = 10
		}
		if = {
			limit = { scope:actor = { horde_unity < 30 } }
			add = -30
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_aggressive } }
			add = 15
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_expansionist } }
			add = 10
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_opportunistic } }
			add = 5
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_cautious } }
			add = -25
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_defensive } }
			add = -15
		}
	}
}

# ------------------------------------------------------------
# ASK FOR A PLEDGE. The other direction: an amir asks another house to
# stand behind HIM. An AI house agrees when ulus_pledge_acceptance
# (script_values/05_amirs_regency_values.txt, section 12) is 0 or more:
# what it thinks of us, a marriage or alliance between the houses, the
# gap in weight, and whether it has hopes of the khan itself. Houses
# that would refuse are greyed out in the list, with the reckoning in
# the tooltip. A player's house is asked by event (amirs_rising.7).
# ------------------------------------------------------------
ulus_ask_for_pledge = {
	type = internationalorganization
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 24
	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = ulus_ask_for_pledge
		years = 2
	}

	price = price:ulus_ask_for_pledge_price

	potential = {
		scope:actor = {
			is_ulus_chaghatay_amir = yes
		}
	}

	allow = {
		scope:actor = {
			# No chains: a house that has pledged cannot be backed.
			custom_tooltip = {
				text = ulus_not_pledged_tt
				NOR = {
					has_variable = ulus_pledged_to
					has_variable = ulus_pledged_to_khan
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = {
			data = name
		}
		visible = {
			international_organization_type = international_organization_type:ulus_chaghatay
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = target_country
		name = "choose_country"
		none_available_msg_key = "no_valid_countries"
		interaction_source_list = {
			scope:recipient = {
				every_international_organization_member = {
					add_to_list = source
				}
			}
		}
		visible = {
		}
		enabled = {
			this != scope:actor
			NOT = { is_leader_of_international_organization = scope:recipient }
			custom_tooltip = {
				text = ulus_ask_not_pledged_tt
				NOT = { has_variable = ulus_pledged_to }
			}
		}
		column = { data = name }
		column = { data = population }
		column = { data = great_power_score }
	}

	effect = {
		scope:actor = { save_scope_as = ulus_asking_house }
		scope:target_country = {
			# What they will answer, and each thing they weigh (tooltip).
			ulus_show_pledge_reckoning = yes
			if = {
				limit = { is_ai = no }
				trigger_event_non_silently = amirs_rising.7
			}
			else_if = {
				limit = { ulus_pledge_acceptance >= 0 }
				ulus_pledge_to_asker = yes
			}
			else = {
				scope:ulus_asking_house = {
					trigger_event_non_silently = amirs_rising.8
				}
			}
		}
	}

	# The AI asks when it is among the likeliest kingmakers and the bar is
	# not yet met: it is building toward a rising or a succession.
	# The AI asks for pledges when it is a contender: among the likeliest
	# kingmakers before a regency (building toward a rising or a
	# succession), and again once the amirs have fallen out and blocs
	# decide wars. A house that stands no chance does not canvass.
	ai_will_do = {
		if = {
			limit = {
				exists = scope:target_country
				scope:target_country = {
					is_ai = yes
					ulus_pledge_acceptance < 0
				}
			}
			add = -1000
		}
		add = -15
		if = {
			limit = {
				scope:actor = {
					is_target_in_global_variable_list = {
						name = ulus_kingmakers
						target = this
					}
				}
			}
			add = 25
		}
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= {
					has_variable = ulus_regency
					has_variable = ulus_regency_stage
					var:ulus_regency_stage >= 2
				}
			}
			add = 10
		}
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= {
					has_variable = ulus_amir_bar
					var:ulus_amir_bar >= 0.75
				}
			}
			add = 10	# the bar is close: a little more weight tips it
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_friendly } }
			add = 10
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_opportunistic } }
			add = 10
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_isolationist } }
			add = -20
		}
	}
}

# ============================================================
# THE KHAN'S SIDE, THE FEUD, AND FRIENDS OUTSIDE
#
#   ulus_khan_ask_pledge        the khan asks a lesser lord (one NOT among
#                               the three likeliest kingmakers) to stand
#                               with the seat: half the lord's weight is
#                               counted as the khan's. The lord is paid
#                               now, and takes the land of broken rebels.
#   ulus_press_the_feud         from stage 2, a house in one bloc gets a
#                               casus belli on a house in another: what
#                               a pledge is FOR once the amirs fall out.
#   ulus_ask_outside_support    an amir asks a power outside the ulus to
#                               back him. A backer counts for less than
#                               an ally and binds nobody; he comes in
#                               to weaken the khan.
# Acceptance values: script_values/05_amirs_regency_values.txt, 15-16.
# ============================================================

ulus_khan_ask_pledge = {
	type = internationalorganization
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = ulus_khan_ask_pledge
		years = 1
	}

	price = price:ulus_khan_ask_pledge_price

	potential = {
		scope:actor = {
			is_ulus_chaghatay_leader = yes
			NOT = { has_variable = ulus_regent_seat }
		}
		international_organization:ulus_chaghatay ?= {
			NOT = { has_variable = ulus_regency }
		}
	}

	allow = {
		always = yes
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = {
			data = name
		}
		visible = {
			international_organization_type = international_organization_type:ulus_chaghatay
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = target_country
		name = "choose_country"
		none_available_msg_key = "no_valid_countries"
		interaction_source_list = {
			scope:recipient = {
				every_international_organization_member = {
					add_to_list = source
				}
			}
		}
		visible = {
		}
		enabled = {
			this != scope:actor
			custom_tooltip = {
				text = ulus_ask_not_pledged_tt
				NOR = {
					has_variable = ulus_pledged_to
					has_variable = ulus_pledged_to_khan
				}
			}
			# The great amirs are the ones he needs the others AGAINST.
			custom_tooltip = {
				text = ulus_khan_ask_lesser_tt
				NOT = {
					is_target_in_global_variable_list = {
						name = ulus_kingmakers
						target = this
					}
				}
			}
		}
		column = { data = name }
		column = { data = population }
		column = { data = great_power_score }
	}

	effect = {
		scope:actor = { save_scope_as = ulus_asking_khan }
		scope:target_country = {
			ulus_show_khan_pledge_reckoning = yes
			if = {
				limit = { is_ai = no }
				trigger_event_non_silently = amirs_rising.9
			}
			else_if = {
				limit = { ulus_khan_pledge_acceptance >= 0 }
				ulus_pledge_to_khan = yes
			}
			else = {
				scope:ulus_asking_khan = {
					trigger_event_non_silently = amirs_rising.8
				}
			}
		}
	}

	# The AI khan gathers the lesser lords when the amirs are closing on
	# him: nothing while he is safe, eager as the bar nears its line. A
	# cautious khan starts earlier; one who cannot pay does not start.
	ai_will_do = {
		if = {
			limit = {
				exists = scope:target_country
				scope:target_country = {
					is_ai = yes
					ulus_khan_pledge_acceptance < 0
				}
			}
			add = -1000
		}
		add = -10
		if = {
			limit = {
				international_organization:ulus_chaghatay ?= { has_variable = ulus_amir_bar }
			}
			add = {
				value = international_organization:ulus_chaghatay.var:ulus_amir_bar
				multiply = 40
			}
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_cautious } }
			add = 10
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_defensive } }
			add = 10
		}
		if = {
			limit = { scope:actor = { gold < 0 } }
			add = -50
		}
	}
}

ulus_press_the_feud = {
	type = internationalorganization
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = ulus_press_the_feud
		years = 5
	}

	price = price:ulus_press_the_feud_price

	potential = {
		scope:actor = {
			is_ulus_chaghatay_amir = yes
		}
		international_organization:ulus_chaghatay ?= {
			has_variable = ulus_regency
		}
	}

	allow = {
		scope:actor = {
			at_war = no
			custom_tooltip = {
				text = ulus_amirs_fallen_out_tt
				ulus_regency_stage_at_least = { STAGE = 2 }
			}
			custom_tooltip = {
				text = ulus_feud_in_a_bloc_tt
				has_variable = ulus_bloc_head
			}
		}
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = {
			data = name
		}
		visible = {
			international_organization_type = international_organization_type:ulus_chaghatay
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = target_country
		name = "choose_country"
		none_available_msg_key = "no_valid_countries"
		interaction_source_list = {
			scope:recipient = {
				every_international_organization_member = {
					add_to_list = source
				}
			}
		}
		visible = {
		}
		enabled = {
			this != scope:actor
			NOT = { is_leader_of_international_organization = scope:recipient }
			custom_tooltip = {
				text = ulus_feud_other_bloc_tt
				has_variable = ulus_bloc_head
				NOT = { var:ulus_bloc_head = scope:actor.var:ulus_bloc_head }
			}
			scope:actor = { NOT = { has_truce_with = prev } }
		}
		column = { data = name }
		column = { data = population }
		column = { data = great_power_score }
	}

	effect = {
		scope:actor = {
			add_casus_belli = {
				type = casus_belli:cb_tumen_feud
				target = scope:target_country
				years = 5
			}
		}
	}

	# The AI presses a feud against a house it can beat: never against a
	# stronger one, readily against a much weaker one, more so if that
	# house is a rival or stands behind the bloc's chief rival for the
	# khan. Aggressive rulers do it sooner, cautious ones seldom, and
	# nobody starts one while his own horde is coming apart.
	ai_will_do = {
		add = -20
		if = {
			limit = { exists = scope:target_country }
			if = {
				limit = {
					scope:actor = { has_variable = ulus_weight_eff }
					scope:target_country = { has_variable = ulus_weight_eff }
				}
				if = {
					limit = {
						scope:actor.var:ulus_weight_eff >= {
							value = scope:target_country.var:ulus_weight_eff
							multiply = 1.5
						}
					}
					add = 35
				}
				else_if = {
					limit = {
						scope:actor.var:ulus_weight_eff >= {
							value = scope:target_country.var:ulus_weight_eff
							multiply = 1.15
						}
					}
					add = 15
				}
				else = {
					add = -60
				}
			}
			if = {
				limit = { scope:actor = { is_rival_of = scope:target_country } }
				add = 20
			}
			if = {
				limit = {
					scope:target_country = {
						var:ulus_bloc_head ?= {
							is_target_in_global_variable_list = {
								name = ulus_kingmakers
								target = this
							}
						}
					}
				}
				add = 10
			}
			add = {
				value = "scope:actor.opinion(scope:target_country)"
				multiply = -0.1
			}
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_aggressive } }
			add = 15
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_expansionist } }
			add = 10
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_cautious } }
			add = -20
		}
		if = {
			limit = { scope:actor = { horde_unity < 30 } }
			add = -30
		}
	}
}

ulus_ask_outside_support = {
	type = internationalorganization
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 24
	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = ulus_ask_outside_support
		years = 3
	}

	price = price:ulus_ask_outside_support_price

	potential = {
		scope:actor = {
			is_ulus_chaghatay_amir = yes
		}
	}

	allow = {
		scope:actor = {
			custom_tooltip = {
				text = ulus_no_outside_backer_tt
				NOT = { has_variable = ulus_outside_backer }
			}
		}
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = {
			data = name
		}
		visible = {
			international_organization_type = international_organization_type:ulus_chaghatay
		}
	}

	# Powers outside the ulus that border any of its members.
	select_trigger = {
		looking_for_a = country
		target_flag = target_country
		name = "choose_country"
		none_available_msg_key = "no_valid_countries"
		interaction_source_list = {
			scope:recipient = {
				every_international_organization_member = {
					every_neighbor_country = {
						limit = {
							is_rebel_country = no
							NOT = { is_member_of_international_organization = scope:recipient }
						}
						add_to_list = source
					}
				}
			}
		}
		visible = {
		}
		enabled = {
			custom_tooltip = {
				text = ulus_outsider_backs_nobody_tt
				NOT = { has_variable = ulus_backs_amir }
			}
			scope:actor = { NOT = { is_at_war_with = prev } }
		}
		column = { data = name }
		column = { data = population }
		column = { data = great_power_score }
	}

	effect = {
		scope:actor = { save_scope_as = ulus_asking_house }
		scope:target_country = {
			ulus_show_outside_reckoning = yes
			if = {
				limit = { is_ai = no }
				trigger_event_non_silently = amirs_rising.11
			}
			else_if = {
				limit = { ulus_outside_support_acceptance >= 0 }
				ulus_outsider_backs_house = yes
			}
			else = {
				scope:ulus_asking_house = {
					trigger_event_non_silently = amirs_rising.15
				}
			}
		}
	}

	# The AI amir looks outside when he is a contender who is not the
	# strongest: the top house does not need a patron and should not
	# want one. Not while he already has allies outside to lean on.
	ai_will_do = {
		if = {
			limit = {
				exists = scope:target_country
				scope:target_country = {
					is_ai = yes
					scope:actor = { save_temporary_scope_as = ulus_asking_house }
					ulus_outside_support_acceptance < 0
				}
			}
			add = -1000
		}
		add = -10
		if = {
			limit = {
				scope:actor = {
					is_target_in_global_variable_list = {
						name = ulus_kingmakers
						target = this
					}
				}
			}
			add = 20
		}
		if = {
			limit = {
				scope:actor = {
					has_variable = ulus_kingmaker_score
					international_organization:ulus_chaghatay ?= { has_variable = ulus_top_kingmaker_score }
					var:ulus_kingmaker_score >= international_organization:ulus_chaghatay.var:ulus_top_kingmaker_score
				}
			}
			add = -25
		}
		if = {
			limit = {
				scope:actor = {
					has_variable = ulus_outside_weight
					var:ulus_outside_weight > 3
				}
			}
			add = -15
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_opportunistic } }
			add = 10
		}
		if = {
			limit = { scope:actor = { ai_personality = ai_personality:ai_isolationist } }
			add = -20
		}
	}
