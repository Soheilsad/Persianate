# ============================================================
# The Matter of Tork — "The Place of the Sword"
#
# NOT a legitimacy disaster. By the time it fires, legitimacy is
# settled — the player came through farr_proof_of_fortune, the
# ghazi_burden_of_sword, or the mystical turn. The question this
# disaster asks is the one legitimacy does NOT answer:
#
#   You are a non-Borjigin holding the ulus. The Türk military
#   constituency's world-picture — rank denominated in proximity
#   to the golden line — has just been made irrelevant by your
#   legitimacy. They did not evaporate. They are standing in the
#   realm, armed, a third of the land and most of the cavalry.
#   WHAT IS THEIR PLACE NOW?
#
# Resolved by claiming ONE of five dispositions (panel buttons):
#   A  Co-optation / Iranization — dissolved into us  (~20y process)
#   B  Frontier Confederation — kept at the edge      (instant; March subject)
#   C  Sufi Synthesis — transformed into our instrument (instant; mysticism-gated)
#   D  Expulsion — removed from the society           (~10y process)
#   E  Eradication — destroyed as a constituency      (civil war + 50y maturation)
#
# Unresolved past 240 months → mass Türk defection (civil war).
#
# Design: docs/design/matter_of_tork_design.md
# Triggers: in_game/common/scripted_triggers/03_matter_triggers.txt
# Effects:  in_game/common/scripted_effects/02_matter_effects.txt
# Actions:  in_game/common/generic_actions/matter_of_tork_actions.txt
# ============================================================

matter_of_tork = {

	image = "gfx/interface/illustrations/disaster/matter_of_tork.dds"

	# The can_start gate is the real filter; the spawn chance just
	# keeps the wait short once the gate opens (fires ~year 50-70
	# in practice, when consolidation completes).
	monthly_spawn_chance = 0.05

	can_start = {
		matter_of_tork_can_start = yes
	}

	can_end = {
		OR = {
			# ---The proclamation route. Each path renders as its LIVE
			# underlying states (raw triggers, engine-native tooltips)
			# ORed with its completion flag - the flag holds a finished
			# path true even if the live state later drifts.
			AND = {
				custom_tooltip = {
					text = matter_tork_answer_button_tt
					has_variable = matter_tork_answer_given
				}
				OR = {
					# --- A. Dissolve: the program's real states ---
					OR = {
						custom_tooltip = {
							text = matter_tork_ce_dissolve_done_tt
							has_variable = matter_tork_path_dissolve_done
						}
						AND = {
							custom_tooltip = {
								text = matter_tork_ce_dissolve_claim_tt
								has_variable = matter_tork_iranization_claimed
							}
							custom_tooltip = {
								text = matter_tork_ce_dissolve_time_tt
								trigger_if = {
									limit = { has_variable = matter_tork_iranization_months }
									var:matter_tork_iranization_months >= 120
								}
								trigger_else = { always = no }
							}
							has_estate_privilege = estate_privilege:amirs_umara_at_the_diwan
							has_estate_privilege = estate_privilege:diwan_mavajib_cash_pay
							custom_tooltip = {
								text = matter_tork_ce_step_pasture_tt
								any_owned_location = { has_building = building_type:registered_pasture }
								NOT = {
									any_owned_location = {
										location_rank = location_rank:rural_settlement
										num_pop_type:tribesmen > 20
										NOT = { has_building = building_type:registered_pasture }
									}
								}
							}
							OR = {
								has_estate_privilege = estate_privilege:salaried_pastoral_levies
								has_estate_privilege = estate_privilege:diwan_tazik_command_of_arms
							}
							societal_value:turk_vs_tazik >= 25
							estate_satisfaction:amirs_estate >= 0.5
							estate_satisfaction:diwan_estate >= 0.5
							estate_satisfaction:tribes_estate >= 0.5
						}
					}
					# --- B. March: the theater won and released ---
					OR = {
						custom_tooltip = {
							text = matter_tork_ce_march_done_tt
							has_variable = matter_tork_path_march_done
						}
						AND = {
							custom_tooltip = {
								text = matter_tork_ce_march_theater_tt
								has_variable = matter_tork_raid_region
							}
							custom_tooltip = {
								text = matter_tork_release_need_tt
								trigger_if = {
									limit = { has_variable = matter_tork_raid_owned }
									var:matter_tork_raid_owned >= var:matter_tork_raid_need
								}
								trigger_else = { always = no }
							}
							custom_tooltip = {
								text = matter_tork_release_size_tt
								trigger_if = {
									limit = { has_variable = matter_tork_raid_owned }
									var:matter_tork_raid_owned >= 25
								}
								trigger_else = { always = no }
							}
							custom_tooltip = {
								text = matter_tork_release_tribes_tt
								trigger_if = {
									limit = { has_variable = matter_tork_raid_tribes }
									var:matter_tork_raid_tribes >= 500
								}
								trigger_else = { always = no }
							}
							custom_tooltip = {
								text = matter_tork_ce_march_release_tt
								has_variable = matter_tork_marcher_claimed
							}
						}
					}
					# --- C. Murid bond: the states of the synthesis ---
					OR = {
						custom_tooltip = {
							text = matter_tork_ce_murid_done_tt
							has_variable = matter_tork_path_murid_done
						}
						AND = {
							societal_value:mysticism_vs_jurisprudence <= -50
							custom_tooltip = {
								text = matter_tork_murid_courtship_tt
								has_variable = matter_tork_vilayat_73_fired
							}
							custom_tooltip = {
								text = matter_tork_ce_murid_sworn_tt
								has_variable = matter_tork_murid_claimed
							}
						}
					}
					# --- D. Expulsion: claimed and run to its end ---
					OR = {
						custom_tooltip = {
							text = matter_tork_ce_expulsion_done_tt
							has_variable = matter_tork_path_expulsion_done
						}
						AND = {
							custom_tooltip = {
								text = matter_tork_ce_expulsion_claim_tt
								has_variable = matter_tork_expulsion_underway
							}
							custom_tooltip = {
								text = matter_tork_ce_expulsion_time_tt
								trigger_if = {
									limit = { has_variable = matter_tork_expulsion_months }
									var:matter_tork_expulsion_months >= 120
								}
								trigger_else = { always = no }
							}
						}
					}
					# --- The empty proclamation: its scar stands ---
					custom_tooltip = {
						text = matter_tork_ce_nothing_tt
						has_country_modifier = matter_turk_defection_scar
					}
				}
			}
			# ---The blood route: eradication or defection fought out.
			custom_tooltip = {
				text = matter_tork_end_resolved_tt
				has_variable = matter_of_tork_resolved
			}
			# ---Lost the realm (live): capital outside Eranzamin or
			# the realm subjugated.
			custom_tooltip = {
				text = matter_tork_end_lost_realm_tt
				NOT = { is_eranzamin_contender = yes }
			}
		}
	}

	# The friction. An armed constituency whose world has been
	# voided, standing unhappily in the realm. About a fifth the
	# weight of the farr/ghazi wager-bleeds — this is not a claim
	# under audit, it is a question waiting for an answer. The
	# matter_answer_given offset (applied when a disposition is
	# claimed) cancels most of it.
	modifier = {
		amirs_estate_target_satisfaction = -0.10
		tribes_estate_target_satisfaction = -0.10
		monthly_amirs_estate_rebel_growth = 0.01
		monthly_tribes_estate_rebel_growth = 0.015
		monthly_legitimacy = -0.10
		stability_decay = -0.001
	}

	on_start = {
		matter_tork_init = yes
		set_variable = matter_of_tork_fired
		trigger_event_non_silently = matter_of_tork.1
	}

	on_monthly = {
		# Horizon counter + process counters (iranization / expulsion).
		matter_tork_monthly_tick = yes
		# Process completion + staged-event dispatch (Iranization
		# completion, Expulsion tranches, Eradication war-end poll,
		# defection war-end poll).
		matter_tork_process_dispatch = yes
		# Horizon timeout → mass defection.
		matter_tork_horizon_poll = yes
	}

	on_end = {
		matter_tork_clear = yes
		# One-shot per playthrough.
		set_variable = matter_of_tork_completed
	}
}
