# Zakat to the Imam — the second half of bay'a. Paid by every sworn
# member to the leader country. 2.5% of tax-and-trade income is the
# canonical rate, and the multiplier below is literally that.
# A member sliding the maintenance down is withholding zakat; the
# imams fought their own tribes over exactly this.

zaydi_zakat = {
	get_payer_list = {
		every_international_organization_member = {
			limit = {
				NOT = { is_leader_of_international_organization = root }
			}
			add_to_local_variable_list = {
				name = payers
				target = this
			}
		}
	}
	get_payee_list = {
		leader_country ?= {
			add_to_local_variable_list = {
				name = payees
				target = this
			}
		}
	}
	price = zaydi_zakat_price
	uses_maintenance = yes
	maintenance_modifier = {
		potential_trigger = { "payment_maintenance(scope:recipient|zaydi_zakat)" < 0.5 }
		scale = {
			value = 1
			subtract = {
				value = "payment_maintenance(scope:recipient|zaydi_zakat)"
				multiply = 2
			}
		}
		monthly_prestige = -0.05
	}
	maintenance_modifier = {
		potential_trigger = { "payment_maintenance(scope:recipient|zaydi_zakat)" > 0.5 }
		scale = {
			value = "payment_maintenance(scope:recipient|zaydi_zakat)"
			subtract = 0.5
			multiply = 2
		}
		monthly_prestige = 0.02
	}
	price_multiplier = {
		add = {
			desc = "ZAYDI_ZAKAT_PAYMENT_MULTIPLIER"
			every_international_organization_member = {
				limit = {
					NOT = { is_leader_of_international_organization = root }
				}
				add = monthly_income_trade_and_tax
			}
			multiply = 0.025
		}
	}
	proportion_for_payer = {
		add = {
			desc = "zaydi_zakat"
			value = monthly_income_trade_and_tax
		}
	}
	proportion_for_payee = {
		add = {
			desc = "zaydi_zakat"
			value = 1
		}
	}
}
