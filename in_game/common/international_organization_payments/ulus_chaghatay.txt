# The qubchur — the khan's share. Paid by every member that is not the
# leader, to the leader country (the pattern of the mod's zaydi_zakat and
# base tatar_yoke_leader_payments).
#
# LIGHT by decision: one part in a hundred of tax-and-trade income, the
# old qubchur rate on the herds. A light levy keeps the amirs rich and
# dangerous, which is what they were; the khan's real benefit from the
# ulus is the war call, not the silver. An amir who slides the
# maintenance down is withholding the khan's due and is seen to.

ulus_chaghatay_qubchur = {
	get_payer_list = {
		every_international_organization_member = {
			limit = {
				NOT = { is_leader_of_international_organization = root }
			}
			add_to_local_variable_list = {
				name = payers
				target = this
			}
		}
	}
	get_payee_list = {
		leader_country ?= {
			add_to_local_variable_list = {
				name = payees
				target = this
			}
		}
	}
	price = ulus_chaghatay_qubchur_price
	uses_maintenance = yes
	maintenance_modifier = {
		potential_trigger = { "payment_maintenance(scope:recipient|ulus_chaghatay_qubchur)" < 0.5 }
		scale = {
			value = 1
			subtract = {
				value = "payment_maintenance(scope:recipient|ulus_chaghatay_qubchur)"
				multiply = 2
			}
		}
		# Withholding the qubchur is defiance of the khan, not thrift:
		# at zero payment a junior loses 0.30 prestige and 0.10 horde
		# unity a month (scaled down to nothing at half payment).
		monthly_prestige = -0.30
		monthly_horde_unity = -0.10
	}
	maintenance_modifier = {
		potential_trigger = { "payment_maintenance(scope:recipient|ulus_chaghatay_qubchur)" > 0.5 }
		scale = {
			value = "payment_maintenance(scope:recipient|ulus_chaghatay_qubchur)"
			subtract = 0.5
			multiply = 2
		}
		monthly_prestige = 0.02
	}
	price_multiplier = {
		add = {
			desc = "ULUS_CHAGHATAY_QUBCHUR_PAYMENT_MULTIPLIER"
			every_international_organization_member = {
				limit = {
					NOT = { is_leader_of_international_organization = root }
				}
				add = monthly_income_trade_and_tax
			}
			multiply = 0.05 # 5% of the juniors' tax-and-trade income: at 1% the khan was insolvent
			# Under a regent the tribute falls: the amirs sent Qazaghan
			# gifts, not the khan's due (ulus_regency_tribute_factor,
			# script_values/05_amirs_regency_values.txt). root = the IO.
			if = {
				limit = { has_variable = ulus_regency }
				multiply = ulus_regency_tribute_factor
			}
		}
		# The second decree, Kebek's assessment (scripted_effects/24): the
		# houses that obey it are assessed at half as much again. (The
		# engine shares the whole sum among the payers by income, so a
		# defiant house bears a little of it too.)
		add = {
			desc = "ULUS_DECREE_ASSESSMENT_MULTIPLIER"
			every_international_organization_member = {
				limit = {
					NOT = { is_leader_of_international_organization = root }
					has_variable = ulus_obeys_2
				}
				add = monthly_income_trade_and_tax
			}
			multiply = 0.025
			if = {
				limit = { has_variable = ulus_regency }
				multiply = 0
			}
		}
	}
	proportion_for_payer = {
		add = {
			desc = "ulus_chaghatay_qubchur"
			value = monthly_income_trade_and_tax
		}
	}
	proportion_for_payee = {
		add = {
			desc = "ulus_chaghatay_qubchur"
			value = 1
		}
	}
	# The AI pays in full; refusing is a player's (or a later phase's) choice.
	ai_maintenance_value = { value = 1 }
	ai_maintenance_ignore_saving = yes
}
