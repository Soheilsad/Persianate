# Base values for the Commonwealth of Martyrs estates — mirrors
# diwan_base_values / amirs_base_values in 01_diwan_auto_modifiers.txt.
# Without a base target satisfaction the estates start hyper-rebellious
# and can be neither levied nor taxed.
#
# Gated on estate presence (not government type) so any later polity that
# acquires the estates — Mushaʿshaʿ-type rebellion-fork spawns — gets the
# floor automatically.
#
# NOTE: amirs_base_values is deliberately NOT extended to the commonwealth.
# The expropriated amirs living under Sarbadar rule SHOULD be furious —
# the anti-aristocratic foundation is the polity's whole premise.

sufyani_base_values = {
    potential_trigger = {
        government_type = government_type:commonwealth_of_martyrs
    }

    sufyani_notables_estate_target_satisfaction = 0.40
    global_sufyani_notables_estate_power = 0.05
}

shaykhi_base_values = {
    potential_trigger = {
        government_type = government_type:commonwealth_of_martyrs
    }

    shaykhi_dervishes_estate_target_satisfaction = 0.40

    # POP-WEIGHT COMPENSATION — NEEDS AN IN-GAME CHECK.
    # Before 1.4 the estate declared power_per_pop = 30. The key now lives
    # on the pop type, and the corps is SOLDIERS pops at 0.0625, so the
    # estate fell to 1/480 of its weight (30 / 0.0625 = 480).
    #
    # Designed share: the scripts treat the dervishes as a major estate —
    # collapse below 0.10 (sarbadar situation), Sufyani rivalry pivot at
    # 0.20, dominance at 0.55-0.70 (parliament issues).
    #
    # 1337 pops (thousands), 1.4 pop weights, before any other modifier:
    #   Sabzevar alone: nobles 3.634 x 25 = 90.9, burghers 60.652 x 4 =
    #     242.6, clergy 0.364 x 10 = 3.6, peasants 34.306 x 0.025 = 0.9
    #     -> 338.0 for everyone else; soldiers 5.000 x 0.0625 = 0.3125,
    #     a 0.09% share.
    #   Sabzevar-Nishapur-Tus belt (8 locations): everyone else 983.2,
    #     the same 5.000 soldiers -> 0.03%.
    # With +480 (x481): 0.3125 x 481 = 150.3, i.e. the old 5 x 30.
    #   Sabzevar alone: 150.3 / 488.3 = 31%   (1.3 design: 41%)
    #   eight locations: 150.3 / 1133.5 = 13% (1.3 design: 17.5%)
    # Both land inside the designed 0.10-0.55 band; the shortfall against
    # 1.3 is the burgher weight doubling (2 -> 4), which is intended.
    #
    # Caveats: base never goes above +5.0 on a global estate-power
    # modifier, so this may be capped or displayed oddly; and because
    # national modifiers add together, the privileges' +/-0.10-0.60 on
    # this same key are now a rounding error against 480.
    global_shaykhi_dervishes_estate_power = 480
}
